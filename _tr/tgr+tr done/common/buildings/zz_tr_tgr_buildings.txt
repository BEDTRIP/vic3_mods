# ComPatch: The Great Revision + Tech & Res
# Load order: CMF -> TGR -> T&R -> this patch.
# 2026-08-25 split: the Kuromi's AI half of this compatch moved out --
#   TGR x KAI  -> _tgr/tgr+kai done   (ai_strategy_default)
#   KAI x T&R  -> _tr/kai+tr done     (ai_strategy_resource_expansion)
# Nothing in this file depends on Kuromi's AI.
#
# T&R 1.6 still keeps a full REPLACE: on exactly two vanilla industry buildings.
# Everything else it touches (arms_industry, steel_mill, chemical_plant, ...) moved
# to INJECT: deltas, which merge with TGR's REPLACE_OR_CREATE bodies on their own --
# those are false positives in the key-level conflict report, not conflicts.
#
# Base = T&R's current body, because T&R loads last and the rest of T&R expects that
# PMG layout. The only thing carried over from TGR is building_group.
#
# Why building_group: TGR creates its own bg_industry_heavy / bg_industry_light
# (children of bg_manufacturing) and keys its industry decree
# (common/decrees/TGR_POLITICS_decree.txt), its economic-incentive laws
# (common/laws/TGR_POLITICS_economic_incentives_industry.txt) and its
# building_group_* modifier types off them. T&R's bg_heavy_industry is a *sibling*
# group, not a parent, so is_building_group does not reach it -- with T&R's line
# these two buildings silently drop out of TGR's decree and incentive laws.
# Nothing appears in error.log either way; this is the silent class of breakage.
# Price of the choice: T&R's building_group_bg_heavy_industry_throughput_add bonuses
# (environmental policy laws, two automotive companies; +-0.01..0.02) no longer reach
# these two buildings. Judged the smaller loss.
#
# 2026-08-23: resynced with T&R 1.6. Changes vs the previous patch --
#   pmg_data_optimization_heavy_industry -> pmg_data_optimization_heavy_industry_algorithmic_dispatch
#     on automotive (T&R split the group; the old name still exists, so the stale
#     reference loaded fine and just quietly withheld the newer PM row),
#   has_law -> has_law_or_variant in possible (T&R's variants of law_industry_banned
#     / law_extraction_economy did not block construction under the old line),
#   levels_per_mesh 5 -> 50 to match every other T&R building.

REPLACE:building_automotive_industry = {
	building_group = bg_industry_heavy					# TGR (see header)
	icon = "gfx/interface/icons/building_icons/vehicles_industry.dds"
	city_type = city
	levels_per_mesh = 50

	unlocking_technologies = {
		combustion_engine
	}

	# T&R layout: aeroplanes moved out to its own building_aircraft_industry,
	# so pmg_aeroplanes is intentionally absent here even though TGR still lists it.
	production_method_groups = {
		pmg_automobile_production
		pmg_tanks
		pmg_automation_building_automotive_industry
		pmg_data_optimization_heavy_industry_algorithmic_dispatch
	}

	required_construction = construction_cost_very_high

	possible = {
		owner = {
			NOT = {
				OR = {
					has_law_or_variant = law_type:law_industry_banned
					has_law_or_variant = law_type:law_extraction_economy
				}
			}
		}
	}
	ownership_type = self

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_heavy_industry.dds"
}
