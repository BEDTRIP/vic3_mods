#    ai_enact_weight_modifier into TGR's taxation rework, and an injection into a
#    sub-block nobody else names merges on its own. The old patch REPLACE:d both
#    with a hand-copied body carrying an empty `modifier = { }`, which silently
#    deleted TGR's country_capitalists_pol_str_mult = -0.25 from
#    law_proportional_taxation. Dropping the entries restores TGR's modifier and
#    loses nothing measurable. (Since the 25.08.2026 split that pair lives in

#
#  * law_extraction_economy -- KEPT, rebuilt on TGR's current body (see below).
#  * three lawgroup_colonization laws -- NEW, see the second block.
#
#  * Deliberately NOT patched: law_no_womens_rights, law_women_own_property,
#    law_child_labor_allowed, law_restricted_child_labor. T&R's UN human-rights
#    rewrite defines every key TGR defines there and differs only in numbers
#    (birth rate 0.10 -> 0.05, working adult ratio 0.10 -> 0.05). Both authors set
#    those on purpose; last mod wins is an acceptable outcome and picking TGR's
#    numbers would be a rebalance. See conflicts_tgr_vs_tr_report.md.


# ---------------------------------------------------------------------------
# law_extraction_economy
#
# T&R INJECT:s an on_activate here. on_activate is a single block, not a list, so
# the injected one takes the place of TGR's -- and TGR's on_activate is the only
# place where extraction economy forces a country out of administrative/economic
# centralism into local autonomies. That swap is TGR mechanics and it disappears
# without a trace: no error, the law just enacts and the centralization law stays.
#
# Body below is TGR's current definition verbatim, with on_activate = T&R's
# extended destroy list + TGR's centralization swap appended.
#
# Two deliberate deviations from a literal copy, both marked inline:
#   - TGR repeats `custom_tooltip = industry_banned_tt` twice; kept once.
#   - T&R's list ends with `has_building = building_ecommerce_logistics` ->
#     `remove_building = building_aircraft_industry`, a copy-paste slip that
#     removes aircraft twice and never removes the e-commerce hub. Fixed here.
# ---------------------------------------------------------------------------

REPLACE:law_extraction_economy = {
	group = lawgroup_economic_system

	icon = "gfx/interface/icons/law_icons/extraction_economy.dds"
	progressiveness = 0

	unlocking_technologies = {
		colonization
	}

	# TGR: subject relationships load after law validity is checked, so a plain
	# `is_subject = yes` would invalidate the law for everyone at the 1836 start.
	can_enact = {
		trigger_if = {
			limit = {
				game_date > 1836.1.1
			}
			is_subject = yes
		}
		trigger_else = {
			always = yes
		}
	}

	on_activate = {
		custom_tooltip = industry_banned_tt
		custom_tooltip = {
			text = industry_banned_destroy_tt
			every_scope_state = {
				# --- vanilla / TGR list ---
				if = { limit = { has_building = building_chemical_plant } remove_building = building_chemical_plant }
				if = { limit = { has_building = building_synthetics_plant } remove_building = building_synthetics_plant }
				if = { limit = { has_building = building_steel_mill } remove_building = building_steel_mill }
				if = { limit = { has_building = building_motor_industry } remove_building = building_motor_industry }
				if = { limit = { has_building = building_automotive_industry } remove_building = building_automotive_industry }
				if = { limit = { has_building = building_electrics_industry } remove_building = building_electrics_industry }
				if = { limit = { has_building = building_munition_plant } remove_building = building_munition_plant }

				# --- T&R industries ---
				if = { limit = { has_building = building_alloys_plant } remove_building = building_alloys_plant }
				if = { limit = { has_building = building_mendelejew_synthetic_rubber_factory } remove_building = building_mendelejew_synthetic_rubber_factory }
				if = { limit = { has_building = building_mendelejew_hydrogenation_plants } remove_building = building_mendelejew_hydrogenation_plants }
				if = { limit = { has_building = building_electronics_industry } remove_building = building_electronics_industry }
				if = { limit = { has_building = building_battery_plant } remove_building = building_battery_plant }
				if = { limit = { has_building = building_processors_foundry } remove_building = building_processors_foundry }
				if = { limit = { has_building = building_aircraft_industry } remove_building = building_aircraft_industry }
				# T&R writes remove_building = building_aircraft_industry on this line; typo.
				if = { limit = { has_building = building_ecommerce_logistics } remove_building = building_ecommerce_logistics }
			}
		}

		# TGR: extraction economy forces the decentralized variant
		if = {
			limit = {
				has_law = law_type:law_administrative_centralism
			}
			deactivate_law = law_type:law_administrative_centralism
			activate_law = law_type:law_local_autonomies
		}
		else_if = {
			limit = {
				has_law = law_type:law_economic_centralism
			}
			deactivate_law = law_type:law_economic_centralism
			activate_law = law_type:law_local_autonomies
		}
	}

	modifier = {
		building_group_bg_agriculture_throughput_add = 0.1
		building_group_bg_plantations_throughput_add = 0.1
		building_group_bg_extraction_throughput_add = 0.1
		state_shopkeepers_investment_pool_efficiency_mult = -0.5
		state_capitalists_investment_pool_efficiency_mult = -0.25
		state_farmers_investment_pool_efficiency_mult = -0.75
		state_clergymen_investment_pool_efficiency_mult = -0.75
		state_aristocrats_investment_pool_efficiency_mult = -0.75
		country_private_construction_allocation_mult = 0.5
		building_nationalization_investment_return_add = 0.5
		country_disable_nationalization_without_compensation_bool = yes
		country_economic_dependence_on_overlord_add = 0.5
	}

	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}


# ---------------------------------------------------------------------------
# lawgroup_colonization: colonial_exploitation / colonial_resettlement /
# frontier_colonization
#
# New in T&R 1.6: its UN human-rights system re-issues all three with a full
# REPLACE:, and it loads after TGR. T&R's bodies are otherwise TGR's bodies plus
# the UN gating and has_law_or_variant, with one thing missing -- TGR attaches an
# institution_modifier to each of them (its colonial laws feed IG political
# strength through institution_colonial_affairs). T&R has no institution_modifier
# key at all, so this is a clean carry-over rather than a contest between two
# authors' numbers.
#
# Base = T&R's body verbatim. Added: TGR's institution_modifier block.
# NOT carried over: TGR's progressiveness = 25 on resettlement and frontier
# colonization (T&R sets 0). Both authors state a value there, so last mod wins.
#
# Symptom without this file: no log line. The laws work, the institution simply
# grants no political strength and TGR's colonial politics loop goes quiet.
# ---------------------------------------------------------------------------

REPLACE:law_colonial_exploitation = {
	group = lawgroup_colonization
	icon = "gfx/interface/icons/law_icons/colonial_exploitation.dds"
	progressiveness = -25
	disallowing_laws = {
		law_sakoku
	}
	unlocking_technologies = {
		colonization
	}
	modifier = {
		building_group_bg_manufacturing_unincorporated_throughput_add = -0.1
		building_group_bg_plantations_unincorporated_throughput_add = 0.2
		building_group_bg_rubber_unincorporated_throughput_add = 0.2
		building_group_bg_logging_unincorporated_throughput_add = 0.2
		building_group_bg_mining_unincorporated_throughput_add = 0.2
		state_unincorporated_starting_wages_mult = -0.1
		building_unincorporated_subsistence_output_mult = -0.1
		country_tension_decay_mult = -0.25
	}
	institution = institution_colonial_affairs
	institution_modifier = {
		country_engineers_pol_str_mult = 0.20		# TGR
	}
	#T&R
	can_enact = {
		custom_tooltip = {
			text = ztr_un_hr_conservative_decolonization_tt
			OR = {
				NOT = {
					ztr_is_un_member = yes
				}
				NOT = {
					global_var:ztr_un_hr_decolonization > 0
				}
			}
		}
	}
	#T&R
	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_colonization_in_power_bloc_bool = yes
					has_law_or_variant = scope:law
				}
			}
			can_impose_law_default = yes
		}
		custom_tooltip = {
			text = ztr_un_hr_conservative_decolonization_tt
			OR = {
				NOT = {
					ztr_is_un_member = yes
				}
				NOT = {
					global_var:ztr_un_hr_decolonization > 0
				}
			}
		}
	}
	ai_will_do = {
		OR = {
			has_law_or_variant = law_type:law_colonial_resettlement
			navy_size >= 50
		}
	}
	ai_enact_weight_modifier = {
		#Petitions
		value = 0
		if = {
			limit = {
				ai_has_enact_weight_modifier_journal_entries = yes
			}
			add = 750
		}
		if = {
			limit = {
				has_journal_entry = je_taming_the_north
			}
			add = 500
		}
	}
	ai_impose_chance = {
		value = 0
		if = {
			limit = {
				has_law_or_variant = law_type:law_colonial_exploitation
				scope:target_country = {
					has_law_or_variant = law_type:law_no_colonial_affairs
				}
				power_bloc ?= {
					has_principle_group = principle_group:principle_group_colonial_offices
				}
			}
			add = 10
		}
		if = {
			limit = {
				scope:target_country = {
					c:BIC ?= this
					OR = {
						has_journal_entry = je_british_dictate_plantations
						has_journal_entry = je_british_dictate_military
						has_journal_entry = je_british_dictate_universities
					}
				}
			}
			multiply = 0
		}
	}
}
