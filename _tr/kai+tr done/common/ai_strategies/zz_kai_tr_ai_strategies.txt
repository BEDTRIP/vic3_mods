# ComPatch: Kuromi's AI + Tech & Res -- ai_strategy_resource_expansion
# Load order (see README): CMF -> KAI -> T&R -> this patch.
#
# Split out 2026-08-25. This entry used to live in the three-mod
# TGR + KAI + T&R compatch, but it has nothing to do with The Great Revision:
# TGR and KAI both REPLACE: the strategy and KAI loads later, so TGR's version is
# out of the picture before this file is reached, and the two authors' strategies
# are deliberately different anyway (see conflicts_tgr_vs_kai_report.md next to
# _tgr/tgr+kai done -- no patch there).
#
# What this file is for: KAI's goods_stances list predates Tech & Res's resource
# goods. Without it the AI never forms an export stance on the seven T&R
# resources -- no error in the log, the stance is simply absent. Below is KAI's
# strategy verbatim with those seven appended.
#
# Removed 2026-08-23: ai_strategy_colonial_extraction. T&R 1.6 now ships the
# merged version itself (kai_has_high_supply triggers plus the same seven T&R
# goods) and loads after KAI, so the patch entry only re-stated what T&R already
# does.
#
# Deliberately NOT patched: ai_strategy_industrial_expansion (KAI vs T&R). Both
# authors write a complete, deliberately different strategy -- merging them would
# be a rebalance, not a compatibility fix. T&R's industrial_expansion does keep
# the industry_banned / extraction_economy gate that KAI had, so nothing
# structural is lost.
#
# !! MAINTENANCE !! kai_has_high_supply is KAI's trigger and the whole
# goods_stances block below rests on it. If KAI renames it, this strategy loads
# and silently does nothing.

REPLACE:ai_strategy_resource_expansion = {
	icon = "gfx/interface/icons/ai_strategy_icons/resource_expansion.dds"

	type = administrative

	building_group_weights = {
		bg_rubber = 1.5
		bg_oil_extraction = 1.5
		bg_mining = 1.5
		bg_logging = 1.5
	}

	goods_stances = {
		# --- KAI, verbatim ---
		coal = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = coal } } }
		iron = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = iron } } }
		oil = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = oil } } }
		lead = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = lead } } }
		wood = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = wood } } }
		hardwood = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = hardwood } } }
		sulfur = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = sulfur } } }
		rubber = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = rubber } } }

		# --- T&R resource goods, same seven T&R itself uses in colonial_extraction ---
		good_uranium = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = good_uranium } } }
		bauxite = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = bauxite } } }
		copper = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = copper } } }
		commonores = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = commonores } } }
		advancedores = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = advancedores } } }
		rare_earths = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = rare_earths } } }
		gas = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = gas } } }
	}

	possible = {
		always = yes
	}

	weight = {
		value = 20

		# Scaling with literacy 10-30%
		multiply = {
			value = literacy_rate
			add = -0.10
			divide = 0.20
			min = 0
			max = 1
		}
	}
}
