# T&R rebuilds this building and keeps only its own media PMGs, dropping
# Morgenroete's manzoni_pmg_publisher / manzoni_pmg_newspaper together with
# city_type and levels_per_mesh. REPLACE (not INJECT) because we need both
# mods' PMG lists merged in a specific order, and INJECT would only append.
#
# pmg_data_optimization_heavy_industry_no_shopkeepers, NOT the plain variant:
# T&R 13.05 moved every one of its buildings to the _no_shopkeepers groups.
# The plain group still exists, so using it produced no log error - just a
# building silently running a different employment model than the rest of T&R.
REPLACE:building_manzoni_publishing_industry = {
	building_group = bg_light_industry
	icon = "gfx/interface/icons/building_icons/building_manzoni_publishing_industry.dds"
	city_type = city

	levels_per_mesh = 50

	unlocking_technologies = {
		mass_communication
	}

	production_method_groups = {
		manzoni_pmg_communication                        # Morgenroete
		manzoni_pmg_building_publishing_industry_automation # Morgenroete
		pmg_broadcast_media_industry                     # T&R
		pmg_digital_media_industry                       # T&R
		manzoni_pmg_publisher                            # Morgenroete, dropped by T&R
		manzoni_pmg_newspaper                            # Morgenroete, dropped by T&R
		pmg_data_optimization_heavy_industry_no_shopkeepers # T&R
	}

	can_build_government = {
		has_building = building_urban_center
	}

	can_build_private = {
		has_building = building_urban_center
	}

	required_construction = construction_cost_very_high

	ownership_type = self

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_light_industry.dds"
}
