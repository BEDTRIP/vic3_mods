# Morgenroete <-> Tech & Res: society technologies.
#
# WHY THIS FILE REPLACED A FILE OVERRIDE
# Until 2026-08-21 the compatch shipped an empty placeholder at
# common/technology/technologies/ztr_mr_society.txt, blanking T&R's file of that
# name wholesale. That did protect Morgenroete, but it also threw away T&R's own
# work on dubois_nature_protection_tech (environmental institution),
# malaria_prevention (colonial branch) and pharmaceuticals. Those three now stay
# in T&R's redaction; the entries below are the ones where T&R genuinely breaks
# a Morgenroete mechanic, patched one at a time.
#
# THE PATTERN
# T&R has to run standalone, so wherever a Morgenroete technology carried a
# Morgenroete-only modifier type, T&R stripped or commented it out - otherwise
# the missing type would spam error.log. With Morgenroete present those types
# exist and the mechanics behind them are alive, so the blocks go back in.
# Each entry below is T&R's body plus exactly the stripped lines, except where
# noted.
#
# Re-check on every T&R update: if T&R stops shipping one of these keys, the
# REPLACE here silently becomes a plain override of Morgenroete by itself.

# --- Modern art gate -----------------------------------------------------
# These two are Morgenroete's versions restored outright, not a merge.
#
# elgar_modern_art_tech is not meant to be researchable at all. Morgenroete
# grants it from a journal entry and uses it as the switch that guarantees only
# one of the three modern-art branches (art deco / neoclassicism / new
# objectivity) can ever be taken. T&R removed can_research = no and moved it to
# era_4, which turns the gate into an ordinary tech and lets the branches be
# researched side by side.
#
# elgar_mass_culture_tech is the far end of that same gate: Morgenroete requires
# all three branch endpoints, T&R shortcut it to elgar_modern_art_tech +
# silent_cinema. Restoring only the gate and leaving the shortcut would give a
# half-Morgenroete, half-T&R chain, so both go back together.
# elgar_art_deco_tech / elgar_neoclassicism_tech / elgar_new_objectivity_tech
# are Morgenroete-only and T&R does not touch them, so the chain reassembles
# cleanly.

# --- Panum epidemics -----------------------------------------------------

# a "#Morgenroete: Added Modifier" note in malaria_prevention - so with T&R on

#
# Bodies stay T&R's. malaria_prevention in particular keeps T&R's era_4,

# commented Morgenroete line is put back. Morgenroete's own
# country_institution_health_system_max_investment_add = -1 is deliberately NOT

# --- Vikelas: a genuine three-way collision ------------------------------
# Both mods define this key with a plain (unprefixed) block - Morgenroete in
# mr_sports_vikelas_technologies.txt, T&R in ztr_new_society.txt - so whoever
# loads last simply wins and the other version vanishes. Somebody has to decide.
#
# Chosen: Morgenroete's placement, T&R's payload.
#   era_4 and organized_sports as the only prerequisite come from Morgenroete -
#     the Olympic content is built around the 1896 date and its ai_weight below
#     pushes the AI to get there in time; T&R's era_5 plus
#     international_organizations puts it well past that window.
#   the three modifiers come from T&R - Morgenroete ships an empty modifier
#     block here, and an era_4 society tech that does literally nothing is dead
#     weight in T&R's much longer tree.
#   ai_weight is Morgenroete's full five-condition version (T&R kept only the
#     first two), which is what actually makes France and Greece chase it.
REPLACE:vikelas_international_sports_tech = {
	era = era_4
	texture = "gfx/interface/icons/invention_icons/vikelas_international_sports_tech.dds"
	category = society

	unlocking_technologies = {
		organized_sports
	}

	modifier = {
		country_influence_mult = 0.05
		country_improve_relations_speed_mult = 0.05
		country_infamy_decay_mult = 0.05
	}

	ai_weight = {
		value = 1

		if = {
			limit = {
				exists = c:FRA
				c:FRA = ROOT
			}
			add = 1 #Pierre de Coubertin
		}

		if = {
			limit = {
				exists = c:GRE
				c:GRE = ROOT
			}
			add = 1 #Greece should at least try to host the Games
		}

		if = {
			limit = {
				NOT = {
					has_global_variable = vikelas_ioc_founded_var
				}
				game_date >= 1890.1.1 #Should be keen to invent organized sports before 1896
			}
			add = 0.5
		}

		if = {
			limit = {
				NOT = {
					has_global_variable = vikelas_ioc_founded_var
				}
				game_date >= 1895.1.1 #Historical Games in 1896
			}
			add = 0.5
		}

		if = {
			limit = {
				NOT = {
					has_global_variable = vikelas_ioc_founded_var
				}
				game_date >= 1900.1.1 #Hurry!
			}
			add = 1
		}
	}
}
