### E&F + Morgenroete ComPatch (Vic3 1.13)
### Hook the Morgenroete private-ownership stock switch into the yearly pulse.
###
### on_actions are additive across mods, so this file overrides nothing from
### E&F or Morgenroete (both of them also append to on_yearly_pulse_country).

on_yearly_pulse_country = {
	on_actions = {
		zz_ef_mr_on_yearly_pulse_country
	}
}

zz_ef_mr_on_yearly_pulse_country = {
	effect = {
		mr_private_ownership_production_stocks = yes
	}
}
