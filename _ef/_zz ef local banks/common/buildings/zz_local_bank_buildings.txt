### building_local_bank — частный банк, выпускающий местную валюту
###
### Занятость взята с ванильного торгового центра (pm_trade_center):
### 800 служащих и 200 лавочников на уровень.
### Владение: государство, инвестпул частников, финансовые районы.

building_local_bank = {
	building_group = bg_local_bank
	city_type = city
	levels_per_mesh = 50

	lens = infrastructure

	required_construction = construction_cost_very_low

	icon = "gfx/interface/icons/building_icons/banks/central_bank.dds"

	production_method_groups = {
		pmg_base_building_local_bank
		pmg_mutual_funds_local_bank
	}

	can_build_government = {
		always = yes
	}

	can_build_private = {
		always = yes
	}

	ownership_type = self

	########################################################################
	# Приоритет для ИИ.
	#
	# База 1000 — как у обычного ванильного здания (шахта, электростанция).
	# Идиома "нужен хотя бы один в каждом штате" взята у building_power_plant
	# (06_urban_center.txt): пока в штате нет банка и ни один не строится,
	# приоритет подскакивает, после первого уровня возвращается к обычному.
	#
	# Скоуп ai_value — штат, поэтому до страны идём через owner.
	########################################################################
	ai_value = {
		value = 1000

		# Первый банк в штате: без него в штате нет источника местной валюты
		if = {
			limit = {
				NOR = {
					has_building = building_local_bank
					any_scope_building = {
						is_building_type = building_local_bank
						is_under_construction = yes
					}
				}
			}
			add = 5000
		}

		# Страна без денежной системы: местная валюта — единственная, что
		# может закрыть popneed_currency её попам
		if = {
			limit = {
				owner ?= { has_modifier = no_money_production }
			}
			multiply = 2.0
		}

		# У страны есть своя валюта — местная нужна куда меньше
		if = {
			limit = {
				owner ?= { NOT = { has_modifier = no_money_production } }
			}
			multiply = 0.4
		}
	}
}

### Разрешаем финансовым районам вкладываться в местные банки
TRY_INJECT:building_financial_district = {
	investment_scores = {
		bg_local_bank_score = {
			group = "bg_local_bank"
			score = {
				value = 0.5
			}
		}
	}
}
