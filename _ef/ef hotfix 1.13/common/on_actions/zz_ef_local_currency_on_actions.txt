### E&F Hotfix — пересчёт выпуска local_currency
###
### Аддитивно: on_action складываются, ничего не перекрывается.
###
### Модификатор вешается на СТРАНУ, как и no_money_production у E&F, —
### он сам раздаёт state_sell_orders_* по всем штатам поровну.
###
### Вешать по штатам отдельно нельзя: multiplier в add_modifier считается
### не в скоупе штата, а выше, и любое state_* значение возвращает none.
### Проверено запуском, лог: "Value of wrong type ... Got value of type 'none'".
###
### Пересчитываем ежемесячно: население, достаток и рынок подвижны.

on_monthly_pulse_country = {
	on_actions = {
		zz_ef_local_currency_monthly
	}
}

zz_ef_local_currency_monthly = {
	effect = {
		### Уборка. Второй блок — для сохранений с предыдущей версии,
		### где модификатор пробовали вешать на каждый штат.
		if = {
			limit = { has_modifier = zz_ef_local_currency_fix }
			remove_modifier = zz_ef_local_currency_fix
		}
		every_scope_state = {
			if = {
				limit = { has_modifier = zz_ef_local_currency_fix }
				remove_modifier = zz_ef_local_currency_fix
			}
		}

		### Выдача
		if = {
			limit = {
				zz_ef_local_currency_needs_throttle = yes
				zz_ef_local_currency_reduction > 0
			}
			add_modifier = {
				name = zz_ef_local_currency_fix
				multiplier = zz_ef_local_currency_reduction
			}
		}
	}
}
