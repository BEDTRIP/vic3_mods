#############################################
# EF.30 (2026-09-30, v2 same day) -- the central bank's quarterly step.
# There is no quarterly pulse: a monthly counter 0, 1, 2; on reaching 3 the
# rate steps (scripted_effects/zz_ef_central_bank_rate.txt) and the counter
# resets. Every country, AI and player. Also seeds the player's policy
# variable so the rate buttons never read an unset variable.
# Additive: our own on_action listed under on_monthly_pulse_country (rules,
# section 1).
#############################################

on_monthly_pulse_country = {
	on_actions = {
		zz_ef_cb_rate_monthly
	}
}

zz_ef_cb_rate_monthly = {
	effect = {
		# EF.43 + В4.3 (5.10): the risk premium's currency and deficit factors (scripted_effects/zz_ef_risk_premium.txt)
		zz_ef_risk_monthly = yes
		if = {
			limit = { NOT = { has_variable = zz_ef_rate_bias } }
			set_variable = { name = zz_ef_rate_bias value = 0 }
		}
		if = {
			limit = { has_variable = zz_ef_cb_rate_month }
			change_variable = { name = zz_ef_cb_rate_month add = 1 }
		}
		else = {
			set_variable = { name = zz_ef_cb_rate_month value = 1 }
		}
		if = {
			limit = { var:zz_ef_cb_rate_month >= 3 }
			set_variable = { name = zz_ef_cb_rate_month value = 0 }
			zz_ef_cb_rate_step = yes
		}
	}
}
