#############################################
# PROBE (6.10.2026, Д2.15) -- do companies hold variables? Monthly, the player and big economies: every company
# adds 1 to var:zz_ef_probe_cv; the country logs how many companies it has, how many hold the variable and the sum.
# If variables persist on companies the sum grows by the number of companies each month. Remove after the run.
#############################################

on_monthly_pulse_country = {
	on_actions = {
		zz_ef_probe_company_var
	}
}

zz_ef_probe_company_var = {
	effect = {
		if = {
			limit = {
				OR = {
					is_player = yes
					gdp > 20000000
				}
			}
			set_variable = { name = zz_ef_probe_n value = 0 }
			set_variable = { name = zz_ef_probe_nv value = 0 }
			set_variable = { name = zz_ef_probe_sum value = 0 }
			every_company = {
				if = {
					limit = { NOT = { has_variable = zz_ef_probe_cv } }
					set_variable = { name = zz_ef_probe_cv value = 0 }
				}
				change_variable = { name = zz_ef_probe_cv add = 1 }
				owner = {
					change_variable = { name = zz_ef_probe_n add = 1 }
				}
				if = {
					limit = { has_variable = zz_ef_probe_cv }
					owner = {
						change_variable = { name = zz_ef_probe_nv add = 1 }
						change_variable = { name = zz_ef_probe_sum add = prev.var:zz_ef_probe_cv }
					}
				}
			}
			debug_log = "EFZ|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|companies [THIS.GetCountry.MakeScope.Var('zz_ef_probe_n').GetValue|0]|with_var [THIS.GetCountry.MakeScope.Var('zz_ef_probe_nv').GetValue|0]|sum [THIS.GetCountry.MakeScope.Var('zz_ef_probe_sum').GetValue|0]"
		}
	}
}
