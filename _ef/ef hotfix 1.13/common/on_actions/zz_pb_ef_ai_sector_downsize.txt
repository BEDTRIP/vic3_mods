#############################################
# STR.2 (night G, 2026-10-03, decision D.12) -- the AI sheds construction
# sectors over the limit.
#
# Urban centres cannot be built (buildable = no, they grow with urbanisation),
# so the AI's levers against the overbuilt-economy penalty are: not building
# sectors over the limit (ai_value 0 there, buildings/zz_pb_ef_construction_sector.txt),
# a lower rate (block A) and this: taking sectors down.
#
# Pace -- no faster than the penalty (D.12: "do not jerk"): one level a quarter,
# only while the country is over the limit AND the penalty index has built up to
# 10 or more (on_actions/zz_pb_ef_overbuild_counter.txt moves it 1-2 points a
# month). The penalty cuts throughput to what the limit would make; the AI then
# trims one level per quarter until the sectors fit.
#
# Which level: the biggest sector of the country (most levels). The engine has no
# effect that lowers a building's level, so the building is removed and created
# again one level lower (create_building, level = old - 1), with its construction
# and liquidity methods put back. Cash reserves: half. Ownership of the new
# building is the engine's default for a new building -- the old owners lose all
# its levels. Decided at night, to check in a run: who owns it after; whether
# employment churns. Fallback if it misbehaves: remove_building only when level 1.
#
# Players are left alone (is_player = no): the penalty and the journal are theirs.
# Countries E&F has not initialised (no var:base_rate_percentage) are skipped.
#############################################

on_monthly_pulse_country = {
	on_actions = {
		zz_pb_ef_ai_sector_downsize
	}
}

zz_pb_ef_ai_sector_downsize = {
	trigger = {
		is_player = no
		has_variable = base_rate_percentage
	}
	effect = {
		# change_variable on an unset variable logs an error and sets nothing (run 2,
		# 3.10: the timer never started and nothing was taken down)
		if = {
			limit = { NOT = { has_variable = zz_pb_ef_ai_downsize_timer } }
			set_variable = {
				name = zz_pb_ef_ai_downsize_timer
				value = 0
			}
		}
		change_variable = {
			name = zz_pb_ef_ai_downsize_timer
			add = 1
		}
		if = {
			limit = {
				var:zz_pb_ef_ai_downsize_timer >= 3
			}
			set_variable = {
				name = zz_pb_ef_ai_downsize_timer
				value = 0
			}
			if = {
				limit = {
					has_variable = speculative_share_2
					var:speculative_share_2 >= 10
					zz_pb_ef_css_levels > building_urban_center_lvl_by_base_rate
				}
				ordered_scope_state = {
					limit = { has_building = building_construction_sector }
					order_by = zz_pb_ef_state_css_level
					zz_pb_ef_css_downsize_one = yes
				}
			}
		}
	}
}
