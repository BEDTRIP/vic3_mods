# Stage 2, night 6.10 (Ф1 + Ф2; the user's decisions М, Д2.13, Д2.20-Д2.29) -- the metal accounts.
# Effects: scripted_effects/zz_ef_metal_accounts.txt. Every unit of metal on an account was bought on the market
# (or is the start's stock, logged as such): the CB (var:gold_state_1 / silver_state_1 of its CB states, E&F's
# reserves), the banks (var:zz_ef_bankm_gold / _silver, one per country -- by company in Ф3) and the pops
# (var:zz_ef_popm_gold / _silver).
#
# Units -- E&F's reserve units. Gold: a good x 100 (gold costs 100: 1 unit = 1 money at the base price). Silver: E&F
# counts it against gold by WEIGHT, 1:15.5 (silver_to_gold_rate, ours: the market prices / 15.5), so one silver
# good (base price 50 = half a gold good) is 50 gold units = 50 x 15.5 = 775 silver units -- the value in the
# reserves equals the market value at any prices. Was x 100 for silver too (Д2.7а): a silver standard's CB got
# 7.75 times less cover than it paid for (accepted at night 6.10, check).
#############################################

zz_ef_gold_units_per_good = 100
zz_ef_silver_units_per_good = 775

# Gold per unit of money at the parity (0 = no parity): gold standard and bimetallism count their parity in gold,
# a silver standard in silver.
zz_ef_gold_per_money_par = {
	value = 1
	if = {
		limit = {
			has_variable = money_value_target_1
			var:money_value_target_1 > 0
			OR = {
				has_law = law_type:law_gold_standard
				has_law = law_type:law_bimetallism_standard
				has_law = law_type:law_silver_standard
			}
		}
		value = var:money_value_target_1
		if = {
			limit = { has_law = law_type:law_silver_standard }
			multiply = silver_to_gold_rate
		}
	}
	min = 0.00001
}

# The country keeps its metal in silver (silver standard; no CB and no gold / bimetallic standard -- Asia's silver
# economies) or in gold. Bimetallism: both, half and half.
zz_ef_metal_is_silver = {
	value = 0
	if = {
		limit = {
			OR = {
				has_law = law_type:law_silver_standard
				AND = {
					NOT = { has_modifier = has_central_bank }
					NOT = { has_law = law_type:law_gold_standard }
					NOT = { has_law = law_type:law_bimetallism_standard }
				}
			}
		}
		value = 1
	}
}
zz_ef_metal_silver_share = {
	value = zz_ef_metal_is_silver
	if = {
		limit = { has_law = law_type:law_bimetallism_standard }
		value = 0.5
	}
}

# ---- step 3: the start (Д2.23, Д2.13, Д3.3) --------------------------------
# Pops' metal at the start, a share of a year's GDP: 5% with a CB (money in notes and deposits), 20% without
# (metal is the money) -- Д3.3.
zz_ef_popm_start_share = {
	value = 0.2
	if = {
		limit = { has_modifier = has_central_bank }
		value = 0.05
	}
}
zz_ef_popm_start_money = {
	value = gdp
	min = 0
	multiply = zz_ef_popm_start_share
}

# ---- accounts in money (at the country's parity, gold per money) ------------
zz_ef_popm_gold = {
	value = 0
	if = { limit = { has_variable = zz_ef_popm_gold } value = var:zz_ef_popm_gold }
}
zz_ef_popm_silver = {
	value = 0
	if = { limit = { has_variable = zz_ef_popm_silver } value = var:zz_ef_popm_silver }
}
zz_ef_bankm_gold = {
	value = 0
	if = { limit = { has_variable = zz_ef_bankm_gold } value = var:zz_ef_bankm_gold }
}
zz_ef_bankm_silver = {
	value = 0
	if = { limit = { has_variable = zz_ef_bankm_silver } value = var:zz_ef_bankm_silver }
}
zz_ef_popm_money = {
	value = zz_ef_popm_silver
	multiply = silver_to_gold_rate
	add = zz_ef_popm_gold
	divide = zz_ef_gold_per_money_par
}
zz_ef_bankm_money = {
	value = zz_ef_bankm_silver
	multiply = silver_to_gold_rate
	add = zz_ef_bankm_gold
	divide = zz_ef_gold_per_money_par
}
zz_ef_cbm_gold = {
	value = gold_state_native_for_stockpile
}
zz_ef_cbm_silver = {
	value = silver_state_native_for_stockpile
}

# ---- step 6: the banks' reserve norm (Д2.26 а) ------------------------------
# 15% at the key rate 6.5%, 10% at 2%, 20% at 12%, linear (two pieces), 5..25%.
zz_ef_reserve_norm = {
	value = zz_ef_money_rate
	subtract = 0.065
	if = {
		limit = { zz_ef_money_rate < 0.065 }
		multiply = 1.1111
	}
	else = {
		multiply = 0.9091
	}
	add = 0.15
	min = 0.05
	max = 0.25
}

# The ceiling of the banks' credit (Д2.26 а, Д2.27 а): business credit + consumer credit <= the banks' metal at
# parity / the reserve norm, times the cover's credit multiplier (1 at 40% cover, 0 at 20%). The room under it is
# what the banks may lend more: the CB's credit to them and the new consumer credit stop at it (repayment as was).
zz_ef_credit_now = {
	value = zz_ef_bc_debt
	add = zz_ef_cc_debt
}
zz_ef_credit_limit = {
	value = zz_ef_bankm_money
	divide = zz_ef_reserve_norm
	multiply = zz_ef_cb_cover_credit_mult
	min = 0
}
zz_ef_credit_room = {
	value = zz_ef_credit_limit
	subtract = zz_ef_credit_now
	min = 0
}

# ---- step 6: savings at their norm (Д2.20, Д2.22 в) --------------------------
# The norm of pops' savings: 0.6 of a year's GDP at a 0% deposit rate, 0.3 at 6%, linear, within 0.3..0.6 (as the
# cash norm moves with the rate). Over it, 10% of the gap a week leaves the money -- into wealth, as the engine
# turns pops' leftover income into wealth (Д2.20: the same leftover came back as deposits too, twice).
zz_ef_sav_norm_gdp = {
	value = zz_ef_deposit_rate
	multiply = -5
	add = 0.6
	min = 0.3
	max = 0.6
}
zz_ef_sav_norm = {
	value = gdp
	min = 0
	multiply = zz_ef_sav_norm_gdp
}
zz_ef_sav_speed = 0.1
zz_ef_sav_to_wealth = {
	value = zz_ef_pop_savings
	subtract = zz_ef_sav_norm
	multiply = zz_ef_sav_speed
	min = 0
}

# ---- steps 4, 5, 7: what was bought and sold this week, goods a week ------------
# The CB's building(s): the method's input (1 gold / 10 silver per workforce) x (1 + our percent, zz_ef_cb_metal_buy).

				add = 1
				min = 0

		}
	}
	min = 0
}
zz_ef_cb_in_silver_goods = {

		add = {
			value = modifier:goods_input_silver_add
			multiply = {
				value = modifier:goods_input_silver_mult
				add = 1
				min = 0
			}
		}
	}
	min = 0
}
zz_ef_cb_out_gold_goods = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_bank }
		add = modifier:goods_output_gold_add
	}

zz_ef_cb_out_silver_goods = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_bank }
		add = modifier:goods_output_silver_add

zz_ef_bank_in_gold_goods = {

	every_scope_building = {

zz_ef_bank_in_silver_goods = {
	value = 0

		limit = { is_building_type = building_zz_ef_bank }

		limit = { is_building_type = building_zz_ef_bank }
		add = level
	}
}
# The states' whole consumption (pops + buildings); the pops' = it - the CB's and the banks' input (step 1).
zz_ef_state_gold_cons = {
	value = 0

		add = sg:gold.state_goods_consumption

	min = 0

# ---- step 5: the banks' purchase for next week (Д2.25) ----------------------------
# Target: the norm x the larger of the credit (+5% room to grow) and the deposits; 2% of the gap a week; a surplus
# buys nothing (sold in Ф3). In the country's metal (zz_ef_metal_silver_share).
zz_ef_bank_buy_speed = 0.02
zz_ef_bank_metal_target = {
	value = zz_ef_credit_now
	multiply = 1.05
	min = zz_ef_pop_deposits
	multiply = zz_ef_reserve_norm
}
zz_ef_bank_buy_money = {
	value = zz_ef_bank_metal_target
	subtract = zz_ef_bankm_money
	multiply = zz_ef_bank_buy_speed
	min = 0
}
zz_ef_bank_buy_gold_goods = {
	value = zz_ef_bank_buy_money
	multiply = zz_ef_gold_per_money_par
	multiply = {
		value = 1
		subtract = zz_ef_metal_silver_share
	}
	divide = zz_ef_gold_units_per_good
}
zz_ef_bank_buy_silver_goods = {
	value = zz_ef_bank_buy_money
	multiply = zz_ef_gold_per_money_par
	multiply = zz_ef_metal_silver_share
	multiply = gold_to_silver_rate
	divide = zz_ef_silver_units_per_good
}
# BUILDING scope: this bank's share of the country's purchase, by its levels (an inline value in the effect read
# `level` as none, run r1006_132739; owner.var: on a building -- none too, r1006_134435: the country via state.owner)
zz_ef_bank_buy_g_bld = {
	value = level
	multiply = state.owner.var:zz_ef_mt_bank_g
	divide = state.owner.var:zz_ef_mt_bank_levels
}
zz_ef_bank_buy_s_bld = {
	value = level
	multiply = state.owner.var:zz_ef_mt_bank_s
	divide = state.owner.var:zz_ef_mt_bank_levels
}

# ---- step 7: the CB as an account (Д2.28 а) ----------------------------------------
# Its building's purchase, percent on the method's input: (40% - cover) x 1500 -- +300% at 20% cover, the method's
# own at 40%, none from 46.7%. Metal standards with a CB.
zz_ef_cb_buy_mult = {
	value = zz_ef_cover_normal
	subtract = zz_ef_cb_cover
	multiply = 1500
	min = -100
	max = 300
}
# The sale of a surplus: 2% a week of the metal over 60% cover, in goods (bimetallism sells gold).
zz_ef_cb_sell_cover = 0.6
zz_ef_cb_sell_units = {
	value = 0
	if = {
		limit = { zz_ef_cb_cover > zz_ef_cb_sell_cover }
		value = zz_ef_cb_cover
		subtract = zz_ef_cb_sell_cover
		multiply = money_supply
		multiply = zz_ef_cb_valuation
		multiply = 0.02
		# the cover counts foreign currency too: never more than 2% of the standard's own metal a week
		max = {
			value = zz_ef_cb_metal
			multiply = 0.02
		}
	}
	min = 0
}
# The buy-back from pops under 40% cover: at most 1% of their metal of the standard a week, at most the gap; paid
# at parity + 2%, new money into their savings.
zz_ef_cb_buyback_units = {
	value = 0
	if = {
		limit = { zz_ef_cb_cover < zz_ef_cover_normal }
		value = zz_ef_cover_normal
		subtract = zz_ef_cb_cover
		multiply = money_supply
		multiply = zz_ef_cb_valuation
		if = {
			limit = { has_law = law_type:law_silver_standard }
			max = {
				value = zz_ef_popm_silver
				multiply = 0.01
			}
		}
		else = {
			max = {
				value = zz_ef_popm_gold
				multiply = 0.01
			}
		}
	}
	min = 0
}
zz_ef_cb_buyback_premium = 1.02
