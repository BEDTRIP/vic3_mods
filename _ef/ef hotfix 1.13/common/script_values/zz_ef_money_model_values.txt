# EF.39 / EF.44 / EF.48 (2026-09-30) -- the money model as a flow matrix
# between accounts, all in the ENGINE'S money. COUNTRY scope. Per WEEK, as
# the engine's budget counts (the user, 2026-09-30: a monthly step against
# weekly budget lines did not add up); rates are fractions (a year). Steps:
# scripted_effects/zz_ef_money_model.txt;
# the tooltip: tools/regen_ef_money_supply_loc.py. Decisions: План проекта.md,
# EF.48.
#
# Decided by the user 2026-09-30 (evening, after a new game to 1838): every
# account in the engine's own money. E&F labels that money with the national
# currency, but its "central bank reserves" (money_supply_state) and the
# currency-good flows are counts of the liquidity_currency GOOD (x 4), not
# money -- they are shown apart. Pops hold no money in the engine: their
# income goes to taxes, purchases and pool contributions, and what is left
# becomes wealth (an index), i.e. leaves the money -- so the engine does not
# conserve money, and pops are a transit account, not a stock.
#
# MONEY (all exact engine numbers, no model stock)
#   M0 = treasury (gold_reserves > 0)
#   M1 = M0 + businesses' cash (vanilla credit limit - defines)
#   M2 = M1 + the investment pool (banks)
# E&F's money_supply is replaced by M2, so the currency value reads it.
#
# WHAT THE MODEL MOVES
#   Bank credit is borrowed from the central bank at the key rate: the pool is
#   steered to (monthly contributions) x credit months (12 at 2%, 3 at 12%);
#   below it banks borrow 1.15% of the gap a week (~5% a month; debt
#   var:zz_ef_bank_cb_debt), above it they repay as much (at most the debt);
#   interest at the key rate pool -> CB
#   -> treasury (the CB's profit). CB credit creates money, repayment removes
#   it. Everything else is the engine's own and only shown.
# (Steps 2-3 of the afternoon -- pops' savings from flows, deposits and their
# interest -- are gone: without money at pops there is nothing to hold them.)
#############################################

	value = 0.065

		limit = { has_variable = base_rate_percentage }
		value = var:base_rate_percentage

	min = 0
	max = 0.3

# Vanilla `credit` is the credit LIMIT = buildings' cash reserves +
# COUNTRY_MIN_CREDIT_BASE + COUNTRY_MIN_CREDIT_SCALED x GDP (defines; Grey's
# Soft Econ Adjustments, zzz_MoG_SEA_def.txt, sets 10000 / 0.01 in this
# playset, vanilla 100000 / 0.5). The cash part (the 1838 save: equal to the
# sum of the buildings' cash_reserves):
zz_ef_building_cash = {
	value = credit
	subtract = define:NEconomy|COUNTRY_MIN_CREDIT_BASE
	subtract = {
		value = gdp
		multiply = define:NEconomy|COUNTRY_MIN_CREDIT_SCALED
	}
	min = 0
}

# ---- the CB and abroad, one account (EF.48 item 6) ------------------------

# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of E&F's national_capacity
# (script_values/00_economic_scripted_value.txt): without
# var:trade_balance_in_gold_fixe. E&F added its cumulative trade balance in
# gold to the reserves; now the flows abroad move the CB's metal itself
# (zz_ef_cb_hume_step), so counting both would count trade twice.
REPLACE_OR_CREATE:national_capacity = {
	value = 1
	if = {
		limit = {
			or = {
				has_law = law_type:law_silver_standard
				has_law = law_type:law_bimetallism_standard
				has_law = law_type:law_gold_standard
			}
		}
		add = central_bank_reserves
		add = gold_loaned_to_central_bank_in_national_metal
	}
	min = 0.01
}

# yes when the metal moves with the flows abroad: a central bank on a metal
# standard (gold, silver, bimetallism; not fiat, not gold exchange).
zz_ef_hume_on = {
	value = 0
	if = {
		limit = {
			has_modifier = has_central_bank
			OR = {
				has_law = law_type:law_gold_standard
				has_law = law_type:law_silver_standard
				has_law = law_type:law_bimetallism_standard
			}
		}
		value = 1
	}
}
# The metal the CB pays out from (silver standard: silver; else gold, for
# bimetallism in gold terms).
zz_ef_cb_metal = {
	value = gold_state_native_for_stockpile
	if = {
		limit = { has_law = law_type:law_silver_standard }
		value = silver_state_native_for_stockpile
	}
}
