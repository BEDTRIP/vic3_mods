#############################################
# EF.39 + EF.44 (2026-09-30) -- the money model: savings, deposits, bank credit.
# COUNTRY scope. Money in the game's currency units; rates are fractions.
#
# WHAT E&F DOES (4.1.7)
# - "Pop savings" (pop_savings = var:pop_capital_loss_fix +
#   var:circulating_<currency>_c_var_1) only ever grows: every month += 4 x the
#   market's buy orders of the currency good. By 1848 France held 1960M at a
#   GDP of 85M. It is part of the money supply, so the money supply swells
#   forever.
# - Once a year (investement_pool_borrowing) the pool gets up to +10% of that
#   money supply at a low rate: France 561M in the pool (31 years of its
#   private construction), Britain 686M; Belgium, Spain, Sweden ~0.
#
# WHAT THIS DOES (decided by the user 2026-09-30, План проекта.md, EF.39)
# 1. Savings S keep E&F's inflow and get an outflow: what pops spend on goods
#    and pay in taxes. The engine does not give scripts the pops' spending
#    (the Population overview reads it from the GUI only), so:
#      pop outlays = pops' income - what they put into the pool
#                  ~ GDP / 12 - investment_pool_gross_income x 52 / 12.
#    Outflow saturates: outlays x S / (S + K), K = zz_ef_savings_months
#    months of outlays. A large stock is spent from fully, a small one is
#    hardly touched, so S settles instead of growing forever.
#    Applied through var:pop_capital_loss_fix (E&F's free offset of
#    pop_savings; only the crash writes it, adding the lost capital).
# 2. Deposits D (var:zz_ef_deposits): a target share of the pops' money
#    (S + D) sits in banks = access x rate x trust; every month
#    zz_ef_deposit_speed of the gap moves between S and D, both ways.
# 3. The pool is steered towards D x the credit multiplier (3 at 2%,
#    1.2 at 12%): banks lend more per deposit when money is cheap. Below the
#    target the pool is topped up (bank credit), above it the excess slowly
#    drains. Construction spending and the pops' own pool contributions are
#    vanilla and untouched. Replaces E&F's yearly +10% and the private credit
#    limits (scripted_effects/zz_ef_money_model.txt).
# Money supply = M2 (EF.44): CB reserves + treasury + S + buildings' cash +
# pool; E&F's government_loan counter is out (see COMPOSITION below). The
# currency value is read from it; the gold parity is re-anchored once
# (scripted_effects/zz_ef_money_model.txt).
#############################################

# Months of outlays pops keep in hand before they spend from savings in
# full. Start value, to be tuned by a run.
zz_ef_savings_months = 24

# Share of the S-D gap that moves per month.
zz_ef_deposit_speed = 0.04
# Share of the gap to the pool target closed per month: up, down.
zz_ef_pool_up_speed = 0.05
zz_ef_pool_down_speed = 0.02

# The rate the model reads: the central bank's, else E&F's neutral 6.5%.
zz_ef_money_rate = {
	value = 0.065
	if = {
		limit = { has_variable = base_rate_percentage }
		value = var:base_rate_percentage
	}
	min = 0
	max = 0.3
}

# S, never negative.
zz_ef_savings = {
	value = pop_savings
	min = 0
}

# Pops' spending on goods + taxes per month (see 1. above).
zz_ef_pop_outlays_month = {
	value = gdp
	subtract = {
		value = investment_pool_gross_income
		multiply = 52
	}
	divide = 12
	min = 0
}

zz_ef_savings_k = {
	value = zz_ef_pop_outlays_month
	multiply = zz_ef_savings_months
	min = 1
}

# Monthly outflow from savings: outlays x S / (S + K). The ratio first:
# S x outlays (Britain 60M x 6.6M = 4e14) overflows the engine's fixed point
# (64 bit, 5 decimals: ~9.2e13) -- 2026-09-30 it came out as 150K for 1.8M.
zz_ef_savings_outflow = {
	value = zz_ef_savings
	divide = {
		value = zz_ef_savings
		add = zz_ef_savings_k
	}
	multiply = zz_ef_pop_outlays_month
	min = 0
}

zz_ef_deposits = {
	value = 0
	if = {
		limit = { has_variable = zz_ef_deposits }
		value = var:zz_ef_deposits
	}
	min = 0
}

# Target share of the pops' money in banks = access x rate x trust.
# Access: literacy, and a central bank (a banking system to deposit in).
# Banks as buildings (EF.42) will replace this proxy.
zz_ef_deposit_access = {
	value = 0.15
	add = {
		value = literacy_rate
		multiply = 0.5
	}
	if = {
		limit = { has_modifier = has_central_bank }
		add = 0.15
	}
}
# Rate: a higher rate makes cash in hand dearer (Keynes, Baumol-Tobin):
# 0.8 at 2%, 1.05 at 7%, 1.3 at 12%.
zz_ef_deposit_rate_factor = {
	value = zz_ef_money_rate
	multiply = 5
	add = 0.7
	max = 1.5
}
# Trust: halved in a crisis (depositors run).
zz_ef_deposit_trust = {
	value = 1
	if = {
		limit = {
			OR = {
				has_modifier = economic_instability
				has_modifier = financial_crash
			}
		}
		value = 0.5
	}
}
zz_ef_deposit_share_target = {
	value = zz_ef_deposit_access
	multiply = zz_ef_deposit_rate_factor
	multiply = zz_ef_deposit_trust
	min = 0.05
	max = 0.8
}

# Monthly move S -> D (negative: withdrawals D -> S).
zz_ef_deposit_flow = {
	value = zz_ef_savings
	add = zz_ef_deposits
	multiply = zz_ef_deposit_share_target
	subtract = zz_ef_deposits
	multiply = zz_ef_deposit_speed
}

# Credit multiplier: 3 at 2% and below, 1.2 at 12% and above.
zz_ef_credit_multiplier = {
	value = zz_ef_money_rate
	subtract = 0.02
	multiply = -18
	add = 3
	min = 1.2
	max = 3
}

# Where the pool is steered: deposits x multiplier.
zz_ef_pool_target = {
	value = zz_ef_deposits
	multiply = zz_ef_credit_multiplier
}

# Bank credit = pool above deposits (display, M2 - M1 split, EF.44).
zz_ef_bank_credit = {
	value = investment_pool
	subtract = zz_ef_deposits
}

# This month's change of the pool (positive: new credit).
zz_ef_pool_adjust = {
	value = zz_ef_pool_target
	subtract = investment_pool
	if = {
		limit = { investment_pool < zz_ef_pool_target }
		multiply = zz_ef_pool_up_speed
	}
	else = {
		multiply = zz_ef_pool_down_speed
	}
}

# INFLOW BY MARKET (run 2026-09-30, 1836-1840). E&F's inflow
# (var:circulating_<currency>_c_var_2, pop_savings_monthly) is the whole
# market's buy orders of the currency good, written to the market OWNER only:
# Britain +2.13M a month, British India, Ontario, Bavaria 0 -- members' savings
# melted to ~0 while owners got the whole market's. Split it by GDP share:
# the owner keeps its share, each member gets its share of the owner's.
zz_ef_market_gdp_share = {
	value = 1
	if = {
		limit = { market.gdp > 0 }
		value = gdp
		divide = market.gdp
	}
	min = 0
	max = 1
}
# Monthly correction of S: negative for the owner (gives away the members'
# part), positive for a member.
zz_ef_savings_market_shift = {
	value = 0
	if = {
		limit = { market.owner = this }
		value = pop_savings_monthly
		multiply = {
			value = 1
			subtract = zz_ef_market_gdp_share
		}
		multiply = -1
	}
	else = {
		value = market.owner.pop_savings_monthly
		multiply = zz_ef_market_gdp_share
	}
}

zz_ef_savings_cap = {

	multiply = 2
}
zz_ef_savings_over_cap = {

	subtract = zz_ef_savings_cap
	min = 0
}

# COMPOSITION OF THE MONEY SUPPLY + MONTHLY LEDGER (2026-09-30, user's
# decisions after the 1848 run; План проекта.md, EF.39 / EF.44)
#
# E&F's money_circulating = pop savings + pool + building_saving +
# government_loan. Two of those are not money:
# - government_loan ("held by the government") is a counter: + E&F's
#   minting each month (the same sum vanilla already pays into the treasury)
#   + the government's loans from the central bank, - repayments. Spending
#   never lowers it, so minted money was counted twice (in the treasury's
#   spending, which lands with pops and buildings, and here). Loans from the
#   central bank are debt: E&F keeps them as var:credit_at_central_bank
#   (gold), shown apart.
# - building_saving = vanilla `credit`, the credit LIMIT = buildings' cash
#   reserves + COUNTRY_MIN_CREDIT_BASE + COUNTRY_MIN_CREDIT_SCALED x GDP
#   (defines; Grey's Soft Econ Adjustments, zzz_MoG_SEA_def.txt, sets 10000 /
#   0.01 in this playset, vanilla 100000 / 0.5).
#   The cash part is read back with define: (zz_ef_building_cash).
# The treasury (gold_reserves, when positive) was not counted at all; it is
# cash held by the state -> M0.
#   M0 = central bank reserves + treasury + pops' savings in hand
#   M1 = M0 + buildings' cash
#   M2 = M1 + investment pool (deposits + bank credit) = money_supply
# Government debt is not money: vanilla principal + debt to the central bank.
#############################################

zz_ef_treasury = {
	value = gold_reserves
	min = 0
}
zz_ef_building_cash = {
	value = credit
	subtract = define:NEconomy|COUNTRY_MIN_CREDIT_BASE
	subtract = {
		value = gdp
		multiply = define:NEconomy|COUNTRY_MIN_CREDIT_SCALED
	}
	min = 0
}

# EF.48: trade balance of the month in the national currency, from E&F's
# trade_balance_in_gold_delta (gold; E&F's tooltip shows it as "per month"; its
# var:..._fix copies are set only under a debt condition and read 0) /
# money_value_in_gold (gold per unit of currency). Positive: exports over
# imports.
zz_ef_trade_month = {
	value = 0
	if = {
		limit = { money_value_in_gold > 0 }
		value = trade_balance_in_gold_delta
		divide = money_value_in_gold
	}
}

# EF.48: the treasury card's "other": its change over the month minus the whole
# budget (all income - all expenses, week x 4.333); every budget line is listed
# in the card, so what is left is timing and loans.
zz_ef_other_treasury_budget = {
	value = zz_ef_v_d_treasury
	subtract = {
		value = total_income
		subtract = total_expenses
		multiply = zz_ef_weeks_per_month
	}
}
