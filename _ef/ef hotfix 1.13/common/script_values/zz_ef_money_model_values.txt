#############################################
# EF.39 / EF.44 / EF.48 (2026-09-30) -- the money model as a flow matrix
# between accounts. COUNTRY scope. Money in the game's currency units, per
# month; rates are fractions. Monthly step: scripted_effects/
# zz_ef_money_model.txt; the tooltip: tools/regen_ef_money_supply_loc.py.
# Decisions: План проекта.md, EF.48.
#
# ACCOUNTS
#   pops' savings S  (model: pops hold no money in the engine; kept in E&F's
#                     pop_savings through its free offset var:pop_capital_loss_fix)
#   treasury         (engine: gold_reserves > 0)
#   businesses       (engine: buildings' cash = vanilla credit limit - defines)
#   banks            (engine: the investment pool; deposits D and the banks'
#                     debt to the central bank are model claims on it)
#   central bank     (E&F: money_supply_state)
#   abroad           (no stock; flows only)
# M0 = CB reserves + treasury + S; M1 = M0 + business cash; M2 = M1 + pool
# = money_supply, the base of E&F's currency value.
#
# WHAT THE MODEL MOVES (everything else is the engine's own and only shown)
# 1. S towards k months of pops' income (Cambridge k, Baumol-Tobin): income
#    ~ GDP / 12 (wages, dividends, state pay); purchases are what is left, so
#    E&F's inflow (market demand for the currency good) no longer feeds S.
# 2. Deposits D: S <-> pool towards a target share (access x rate x trust).
#    Vanilla's pool contributions are pops' saving into banks too (shown as
#    such), but are spent on construction at once, so D is only the
#    withdrawable part.
# 3. Interest on deposits: pool -> S at the deposit rate (key - margin).
# 4. Bank credit is borrowed from the central bank at the key rate: the pool
#    is steered to D x credit multiplier; below it banks borrow (debt to the
#    CB), above it they repay; interest pool -> CB, remitted to the treasury
#    (CB profit). Replaces credit "from nowhere" (E&F's yearly +10% of the
#    money supply, then our multiplier alone).
#############################################

# ---- parameters -----------------------------------------------------------

# Months of income pops hold in cash (Cambridge k), and the share of the gap
# closed per month.
zz_ef_cash_months = 4
zz_ef_cash_speed = 0.05
# Share of the S-D gap that moves per month.
zz_ef_deposit_speed = 0.04
# Deposit rate = key rate - margin, not below the floor.
zz_ef_deposit_margin = 0.015
zz_ef_deposit_rate_floor = 0.005
# Share of the gap to the pool target borrowed / repaid per month.
zz_ef_borrow_speed = 0.05
zz_ef_repay_speed = 0.05
zz_ef_weeks_per_month = 4.333

# ---- rates ----------------------------------------------------------------

# The rate the model reads: the central bank's, else E&F's neutral 6.5%.
zz_ef_money_rate = {
	value = 0.065
	if = {
		limit = { has_variable = base_rate_percentage }
		value = var:base_rate_percentage
	}
	min = 0
	max = 0.3
}
zz_ef_deposit_rate = {
	value = zz_ef_money_rate
	subtract = zz_ef_deposit_margin
	min = zz_ef_deposit_rate_floor
}

# ---- stocks ---------------------------------------------------------------

zz_ef_savings = {
	value = pop_savings
	min = 0
}
# S as the model left it at the last step (E&F keeps adding its inflow to
# pop_savings between steps; the step takes it back out).
zz_ef_savings_base = {
	value = zz_ef_savings
	if = {
		limit = { has_variable = zz_ef_prev_savings }
		value = var:zz_ef_prev_savings
	}
	min = 0
}
zz_ef_deposits = {
	value = 0
	if = {
		limit = { has_variable = zz_ef_deposits }
		value = var:zz_ef_deposits
	}
	min = 0
}
zz_ef_bank_cb_debt = {
	value = 0
	if = {
		limit = { has_variable = zz_ef_bank_cb_debt }
		value = var:zz_ef_bank_cb_debt
	}
	min = 0
}
zz_ef_pool = {
	value = investment_pool
}
zz_ef_treasury = {
	value = gold_reserves
	min = 0
}
# Vanilla `credit` is the credit LIMIT = buildings' cash reserves +
# COUNTRY_MIN_CREDIT_BASE + COUNTRY_MIN_CREDIT_SCALED x GDP (defines; Grey's
# Soft Econ Adjustments, zzz_MoG_SEA_def.txt, sets 10000 / 0.01 in this
# playset, vanilla 100000 / 0.5). The cash part:
zz_ef_building_cash = {
	value = credit
	subtract = define:NEconomy|COUNTRY_MIN_CREDIT_BASE
	subtract = {
		value = gdp
		multiply = define:NEconomy|COUNTRY_MIN_CREDIT_SCALED
	}
	min = 0
}

# E&F's money_circulating = pop savings + pool + building_saving (the whole
# credit limit) + government_loan (a counter: cumulative minting, already paid
# into the treasury, + loans from the CB, which are debt). Now: S + pool +
# buildings' cash + treasury.
# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of E&F's money_circulating
# (script_values/00_economic_scripted_value.txt).
REPLACE_OR_CREATE:money_circulating = {
	value = 0
	add = pop_savings
	add = private_bank_funds
	add = zz_ef_building_cash
	add = zz_ef_treasury
	min = 1
}

zz_ef_m0 = {
	value = money_supply_state
	add = zz_ef_treasury
	add = zz_ef_savings
}
zz_ef_m1 = {
	value = zz_ef_m0
	add = zz_ef_building_cash
}
zz_ef_m2_to_m0 = {
	value = money_supply
	divide = {
		value = zz_ef_m0
		min = 1
	}
}

# ---- 1. pops' cash --------------------------------------------------------

zz_ef_gdp_month = {
	value = gdp
	divide = 12
}
zz_ef_cash_target = {
	value = zz_ef_gdp_month
	multiply = zz_ef_cash_months
}
# This month's move of S towards its target (purchases absorb the rest).
zz_ef_cash_adjust = {
	value = zz_ef_cash_target
	subtract = zz_ef_savings_base
	multiply = zz_ef_cash_speed
}

# ---- 2-3. deposits --------------------------------------------------------

# Target share of the pops' money in banks = access x rate x trust.
# Access: literacy, and a central bank (a banking system to deposit in).
# Banks as buildings (EF.42) will replace this proxy.
zz_ef_deposit_access = {
	value = 0.15
	add = {
		value = literacy_rate
		multiply = 0.5
	}
	if = {
		limit = { has_modifier = has_central_bank }
		add = 0.15
	}
}
# A higher rate makes cash in hand dearer (Keynes, Baumol-Tobin):
# 0.8 at 2%, 1.05 at 7%, 1.3 at 12%.
zz_ef_deposit_rate_factor = {
	value = zz_ef_money_rate
	multiply = 5
	add = 0.7
	max = 1.5
}
# Trust: halved in a crisis (depositors run).
zz_ef_deposit_trust = {
	value = 1
	if = {
		limit = {
			OR = {
				has_modifier = economic_instability
				has_modifier = financial_crash
			}
		}
		value = 0.5
	}
}
zz_ef_deposit_share_target = {
	value = zz_ef_deposit_access
	multiply = zz_ef_deposit_rate_factor
	multiply = zz_ef_deposit_trust
	min = 0.05
	max = 0.8
}
# Monthly move S -> pool (negative: withdrawals, limited by the pool).
zz_ef_deposit_flow = {
	value = zz_ef_savings_base
	add = zz_ef_deposits
	multiply = zz_ef_deposit_share_target
	subtract = zz_ef_deposits
	multiply = zz_ef_deposit_speed
	min = {
		value = investment_pool
		multiply = -1
	}
}
# Interest on deposits, pool -> S, a month (not more than the pool).
zz_ef_deposit_interest = {
	value = zz_ef_deposits
	multiply = zz_ef_deposit_rate
	divide = 12
	max = {
		value = investment_pool
		min = 0
	}
	min = 0
}

# ---- 4. bank credit from the central bank ---------------------------------

# Credit multiplier: 3 at 2% and below, 1.2 at 12% and above.
zz_ef_credit_multiplier = {
	value = zz_ef_money_rate
	subtract = 0.02
	multiply = -18
	add = 3
	min = 1.2
	max = 3
}
# Where the pool is steered: deposits x multiplier.
zz_ef_pool_target = {
	value = zz_ef_deposits
	multiply = zz_ef_credit_multiplier
}
# Bank credit = pool above deposits (display, M2 - M1 split).
zz_ef_bank_credit = {
	value = investment_pool
	subtract = zz_ef_deposits
}
zz_ef_pool_to_deposits = {
	value = investment_pool
	divide = {
		value = zz_ef_deposits
		min = 1
	}
}
# Borrowed from the central bank this month (pool under its target).
zz_ef_cb_borrow = {
	value = zz_ef_pool_target
	subtract = investment_pool
	multiply = zz_ef_borrow_speed
	min = 0
}
# Repaid to the central bank this month (pool over its target), not more
# than the debt.
zz_ef_cb_repay = {
	value = investment_pool
	subtract = zz_ef_pool_target
	multiply = zz_ef_repay_speed
	min = 0
	max = zz_ef_bank_cb_debt
}
# Interest on the banks' debt at the key rate, a month (not more than the
# pool); remitted by the CB to the treasury.
zz_ef_cb_interest = {
	value = zz_ef_bank_cb_debt
	multiply = zz_ef_money_rate
	divide = 12
	max = {
		value = investment_pool
		min = 0
	}
	min = 0
}

# ---- seed (a new game or a running one, once) -----------------------------

# Market share by GDP (the currency good's demand of the market goes to its
# countries by GDP).
zz_ef_market_gdp_share = {
	value = 1
	if = {
		limit = { market.gdp > 0 }
		value = gdp
		divide = market.gdp
	}
	min = 0
	max = 1
}
# Central bank reserves, month (E&F: stockpile_currency_type_1,
# 01_economic_scripted_effects.txt:55549): var_2 = (sell - buy orders of the
# currency good in the market) x 4 + devaluation - revaluation, market owner
# only.
zz_ef_cb_issue_month = {
	value = 0
	if = {
		limit = { market.owner = this }
		value = market.mg:liquidity_currency.market_goods_sell_orders
		multiply = 4
	}
}
zz_ef_cb_demand_month = {
	value = 0
	if = {
		limit = { market.owner = this }
		value = market.mg:liquidity_currency.market_goods_buy_orders
		multiply = 4
	}
}
# Bought by this country's buildings and pops, the rest by the other
# countries of the market.
zz_ef_cb_demand_own = {
	value = zz_ef_cb_demand_month
	multiply = zz_ef_market_gdp_share
}
zz_ef_cb_demand_others = {
	value = zz_ef_cb_demand_month
	subtract = zz_ef_cb_demand_own
	min = 0
}
zz_ef_cb_devaluation_month = {
	value = 0
	if = {
		limit = { has_variable = devaluation_on var:devaluation_on = 1 }
		value = money_supply_to_add_per_month
	}
	min = 0
}
zz_ef_cb_revaluation_month = {
	value = 0
	if = {
		limit = { has_variable = revaluation_on var:revaluation_on = 1 }
		value = money_supply_to_subtract_per_month
		abs = yes
	}
}
# Trade balance of the month in the national currency, from E&F's
# trade_balance_in_gold_delta (gold; E&F's tooltip shows it as "per month";
# its var:..._fix copies are set only under a debt condition and read 0) /
# money_value_in_gold (gold per unit of currency). Positive: exports over
# imports.
zz_ef_trade_month = {
	value = 0
	if = {
		limit = { money_value_in_gold > 0 }
		value = trade_balance_in_gold_delta
		divide = money_value_in_gold
	}
}
# The treasury card's "other": its change over the month minus the whole
# budget (all income - all expenses, week x 4.333) minus the model's own
# payments to it (CB interest remittance).
zz_ef_other_treasury_budget = {
	value = zz_ef_v_d_treasury
	subtract = {
		value = total_income
		subtract = total_expenses
		multiply = zz_ef_weeks_per_month
	}
	subtract = zz_ef_v_f_cb_interest
}
