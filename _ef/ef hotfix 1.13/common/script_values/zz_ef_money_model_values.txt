# EF.39 / EF.44 / EF.48 (2026-09-30) -- the money model as a flow matrix
# between accounts, all in the ENGINE'S money. COUNTRY scope. Per WEEK, as
# the engine's budget counts (the user, 2026-09-30: a monthly step against
# weekly budget lines did not add up); rates are fractions (a year). Steps:
# scripted_effects/zz_ef_money_model.txt;
# the tooltip: tools/regen_ef_money_supply_loc.py. Decisions: План проекта.md,
# EF.48.
#
# Decided by the user 2026-09-30 (evening, after a new game to 1838): every
# account in the engine's own money. E&F labels that money with the national
# currency, but its "central bank reserves" (money_supply_state) and the
# currency-good flows are counts of the liquidity_currency GOOD (x 4), not
# money -- they are shown apart. Pops hold no money in the engine: their
# income goes to taxes, purchases and pool contributions, and what is left
# becomes wealth (an index), i.e. leaves the money -- so the engine does not
# conserve money, and pops are a transit account, not a stock.
#
# MONEY (all exact engine numbers, no model stock)
#   M0 = treasury (gold_reserves > 0)
#   M1 = M0 + businesses' cash (vanilla credit limit - defines)
#   M2 = M1 + the investment pool (banks)
# E&F's money_supply is replaced by M2, so the currency value reads it.
#
# WHAT THE MODEL MOVES
#   Bank credit is borrowed from the central bank at the key rate: the pool is
#   steered to (monthly contributions) x credit months (12 at 2%, 3 at 12%);
#   below it banks borrow 1.15% of the gap a week (~5% a month; debt
#   var:zz_ef_bank_cb_debt), above it they repay as much (at most the debt);
#   interest at the key rate pool -> CB
#   -> treasury (the CB's profit). CB credit creates money, repayment removes
#   it. Everything else is the engine's own and only shown.
# (Steps 2-3 of the afternoon -- pops' savings from flows, deposits and their
# interest -- are gone: without money at pops there is nothing to hold them.)
#############################################

# ---- parameters -----------------------------------------------------------

# month).

	value = 0.065
	if = {
		limit = { has_variable = base_rate_percentage }
		value = var:base_rate_percentage
	}
	min = 0
	max = 0.3
}

# Vanilla `credit` is the credit LIMIT = buildings' cash reserves +
# COUNTRY_MIN_CREDIT_BASE + COUNTRY_MIN_CREDIT_SCALED x GDP (defines; Grey's
# Soft Econ Adjustments, zzz_MoG_SEA_def.txt, sets 10000 / 0.01 in this
# playset, vanilla 100000 / 0.5). The cash part (the 1838 save: equal to the
# sum of the buildings' cash_reserves):
zz_ef_building_cash_estimate = {
	value = credit
	subtract = define:NEconomy|COUNTRY_MIN_CREDIT_BASE
	subtract = {
		value = gdp
		multiply = define:NEconomy|COUNTRY_MIN_CREDIT_SCALED
	}
	min = 0
}
# Since run 15 (1.10 evening) the sum over the buildings, kept weekly in

	}

# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of E&F's money_circulating and
# money_supply (script_values/00_economic_scripted_value.txt). E&F's
# money_supply = money_supply_state (goods) + money_circulating (pop savings
# counter + pool + credit limit + government_loan counter).
# The money in circulation (the user, 1.10 day: M2 - M0): businesses' cash +
# the pool + pops' savings at hand. The treasury and the CB are out of it
# (EF.48 item 11: hoarding deflates, spending inflates). The currency's value
# (E&F money_value_0 = national capacity / money_supply) and the inflation
# read it.
# 1.10 evening (the user, run 12): the money in circulation is M2 -- pops'
# cash, business cash and pops' deposits (zz_ef_agg_m2). The pool beyond the
# deposits (the CB's credit, contributions not yet spent) becomes money when
# the banks spend it.
REPLACE_OR_CREATE:money_circulating = {
	value = zz_ef_agg_m2
	min = 1
}
REPLACE_OR_CREATE:money_supply = {
	value = money_circulating
	min = 1

# Banks' credit from the central bank, steered by the rate against GDP (the
# user, 1.10 evening, run 10): with the target "pool = contributions x months"
# the pool was almost always over it, and in 6 years the CB lent Britain 1.2M
# at M2 ~300M -- the credit channel was off. Now the banks' debt to the CB goes
# to GDP x share: 50% at a 2% key rate, 0 at 12%, linear (27.5% at E&F's
# neutral 6.5%); the money comes into the pool and the engine spends it on
# private construction (and the consumer credit lends from it). 1.10 evening,

zz_ef_cb_credit_gdp_max = 0.5

	value = 0.12
	subtract = zz_ef_money_rate
	divide = 0.10
	min = 0
	max = 1

	value = gdp
	min = 0
	multiply = zz_ef_cb_credit_gdp_max
	multiply = zz_ef_cb_credit_share
	multiply = zz_ef_cb_cover_credit_mult
}
# What the banks need in the pool (1.10 evening, run 15: the pool piled up
# to 0.9 GDP while the banks kept borrowing from the CB): 3 months of what it
# pays out -- the transfer for private construction or the contributions,
# whichever is larger, at least 0.1% of GDP a week.
zz_ef_pool_need = {
	value = zz_ef_v_f_transfer
	min = zz_ef_v_f_contrib
	min = {
		value = gdp
		divide = 1000
	}
	multiply = 13
}
# Pool over its need, in needs (0 = no idle money).
zz_ef_pool_idle = {
	value = 0
	if = {
		limit = { zz_ef_pool_need > 0 }
		value = investment_pool
		divide = zz_ef_pool_need
		subtract = 1
	}
	min = 0
}
# Borrowed from the central bank this week: the debt under its target, and
# only what the pool lacks of its need -- idle deposits are lent first.
zz_ef_cb_borrow = {
	value = zz_ef_cb_credit_target
	subtract = zz_ef_bank_cb_debt
	multiply = zz_ef_borrow_speed
	max = {
		value = zz_ef_pool_need
		subtract = investment_pool
	}
	min = 0
}
# Repaid this week: the debt over its target, or the pool's idle money (the
# banks give back the CB's credit they do not need), not more than the debt
# and the pool holds.
zz_ef_cb_repay = {
	value = zz_ef_bank_cb_debt
	subtract = zz_ef_cb_credit_target
	multiply = zz_ef_repay_speed
	min = {
		value = investment_pool
		subtract = zz_ef_pool_need
		multiply = zz_ef_repay_speed
	}
	min = 0
	max = zz_ef_bank_cb_debt
	max = {
		value = investment_pool
		min = 0
	}
}
# Interest on the banks' debt at the key rate, a week (not more than the
# pool); the CB remits it to the treasury.
zz_ef_cb_interest = {
	value = zz_ef_bank_cb_debt
	multiply = zz_ef_money_rate
	divide = 52
	max = {
		value = investment_pool
		min = 0
	}
	min = 0
}

# monetary-policy marks.

	market = {

		mg:clothes = { add = { value = market_goods_pricier add = 1 multiply = market_goods_buy_orders multiply = g:clothes.base_price } }

# ---- the user's money aggregates (1.10 day; the CB out of them 1.10 evening)
# M0 = the treasury; M1 = M0 + the banks (the pool); M2 = M1 + businesses'
# cash + pops' savings at hand; M3 = M2 + claims on abroad (foreign bonds of
# the private banks and of the treasury). Deposits are part of the pool, so
# pops' savings count only at hand.
# The CB's metal and currency-good stock are its RESERVES -- the cover of the
# money, not money (the user, 1.10 evening, run 10): with them in M0 a payment
# abroad was counted twice -- the money left the pool (or business cash) and
# Hume's mechanism took the same sum off the CB's metal (Britain 1840.1.2:
# bonds -7.7M from the pool, metal -7.4M, M2 -4.4% in a month). Now an outflow
# lowers the money once and the cover by the same sum. The CB issues money
# only as credit to banks; that money is in the pool (M1).

# The central bank's reserves in money: its metal at parity (E&F's
# central_bank_reserves, in the standard's metal, / var:money_value_target_1,
# metal per unit). Metal only (the user, 1.10 evening): E&F's currency-good
# stock (money_supply_state) is counts of a good, shown in the CB's card.
zz_ef_cb_metal_money = {
	value = 0
	if = {
		limit = {
			has_modifier = has_central_bank
			has_variable = money_value_target_1
			var:money_value_target_1 > 0
		}
		value = central_bank_reserves
		divide = zz_ef_cb_valuation
	}
}

# Item 1 (the user, 1.10 night): the silver-to-gold rate (gold per unit of
# silver, E&F's silver_to_gold_rate) from the metals' prices: (silver price /
# gold price) x (gold base price / silver base price) / 15.5 -- 1:15.5 at the
# base prices (the French legal ratio of 1803, near the 1830s market). E&F's
# was world gold / silver production (global_var:global_*_production_fix, 2 /
# 13), clamped to 1:5 .. 1:64: silver currencies read ~3 times their gold
# value, the "average value of money" followed them up and the pound read as
# "weak, -55%" (run 18). Prices over every market, weighted by the metal's
# sell + buy orders there (the world price); 1 if nothing is traded. Kept
# monthly in global_var:zz_ef_s2g (zz_ef_silver_rate_update); E&F's bounds.
zz_ef_world_gold_price_mult = {
	value = 1
	if = {
		limit = { zz_ef_world_gold_orders > 0 }
		value = 0
		every_market = {
			add = {
				value = mg:gold.market_goods_pricier
				add = 1
				multiply = {
					value = mg:gold.market_goods_sell_orders
					add = mg:gold.market_goods_buy_orders
				}
			}
		}
		divide = zz_ef_world_gold_orders
	}
	min = 0.05
}

# Item 4 (the user, 1.10 night): the trade balance from the engine's data. The
# engine counts imports and exports in units of goods (run 19); in money: the
# market's exports - imports of every good x its price there (base price x (1 +
# market_goods_pricier)), a week. Trade is the market's: the market owner
# holds it, the other members 0. All goods, the financial ones (E&F's bonds,
# stocks, funds, currency) too -- a payment abroad either way; they are also
# shown apart (zz_ef_trade_fin_week).
# В1.6 item 1 (5.10): at ONE price -- the good's base price, not the market's (base x (1 + pricier)). A good
# leaving one market and entering another was counted at two prices, the importer's and the exporter's, so the
# world's trade did not add up (run s1b, 1839: +32M gold a year without a counterpart).
# В1.6 (5.10 evening, run s1f): at one price the world's exports were still over its imports by 1-4M a month (the
# world market of 1.13 holds goods between its hubs), and the treasuries' market fees, tolls and piracy (abr, +0.5-1.5M)
# have no payer's line. The world market is a pool: what the importers pay is shared by the exporters and the fees'
# receivers (zz_ef_wtr_kr: min(1, imports / (exports + fees)) of the window before); if they claim less, the importers
# pay as much (zz_ef_wtr_kp). The fees' part not covered goes off the flow (zz_ef_tfees_cut).
zz_ef_trade_net_week = {
	value = 0
	if = {
		limit = { exists = market market.owner = this }
		value = zz_ef_trade_exports_week
		multiply = zz_ef_wtr_kr_v
		subtract = {
			value = zz_ef_trade_imports_week
			multiply = zz_ef_wtr_kp_v
		}
		# В3.3 (2.10): customs union members with their own currency settle
		# their sales inside the market with the owner (generated,
		# zz_ef_customs_union_values.txt)
		subtract = zz_ef_members_trade_sum
	}
	else_if = {
		limit = { zz_ef_cu_member = yes }
		value = zz_ef_member_goods_net
	}
}
zz_ef_trade_fin_week = {
	value = 0
	if = {
		limit = { exists = market market.owner = this }
		market = {
			mg:bond = { add = { value = market_goods_exports subtract = market_goods_imports multiply = g:bond.base_price } }
			mg:manufacture_stock = { add = { value = market_goods_exports subtract = market_goods_imports multiply = g:manufacture_stock.base_price } }
			mg:agricultural_stock = { add = { value = market_goods_exports subtract = market_goods_imports multiply = g:agricultural_stock.base_price } }
			mg:mining_stock = { add = { value = market_goods_exports subtract = market_goods_imports multiply = g:mining_stock.base_price } }
			mg:railroad_stock = { add = { value = market_goods_exports subtract = market_goods_imports multiply = g:railroad_stock.base_price } }
			mg:mutual_funds = { add = { value = market_goods_exports subtract = market_goods_imports multiply = g:mutual_funds.base_price } }
			mg:liquidity_currency = { add = { value = market_goods_exports subtract = market_goods_imports multiply = g:liquidity_currency.base_price } }
		}
	}
}
