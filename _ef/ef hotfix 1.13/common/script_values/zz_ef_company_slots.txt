zz_ef_slots_rating = {
	value = 0
	if = {
		limit = { has_variable = country_credit_note_fixe }
		value = var:country_credit_note_fixe
		divide = 2
		round = yes
	}
	min = 0
	max = 5
}

zz_ef_cap_total = {
	value = 0
	if = { limit = { has_variable = zz_ef_cap_avg_ms } add = var:zz_ef_cap_avg_ms }
	if = { limit = { has_variable = zz_ef_cap_avg_as } add = var:zz_ef_cap_avg_as }
	if = { limit = { has_variable = zz_ef_cap_avg_mn } add = var:zz_ef_cap_avg_mn }
	if = { limit = { has_variable = zz_ef_cap_avg_rr } add = var:zz_ef_cap_avg_rr }
	min = 0
}

zz_ef_slots_cap = {
	value = zz_ef_cap_total
	divide = {
		value = gdp
		min = 1
	}
	multiply = 10
	floor = yes
	min = 0
	max = 10
}

zz_ef_slots_cb = {
	value = 0.12
	subtract = zz_ef_money_rate
	divide = 0.10
	multiply = 5
