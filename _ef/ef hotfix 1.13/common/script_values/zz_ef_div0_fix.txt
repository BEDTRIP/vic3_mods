# 7a. Division by zero in the financial demand values
#
# E&F computes the demand modifiers for its five stock/bond goods like this
# (common/script_values/00_financial_scripted_value.txt):
#
#   target_demand_<good>_ajusted = {
#       value    = target_demand_<good>
#       subtract = market.mg:<good>.market_goods_exports
#       divide   = building_financial_num   # <-- can be 0
#       ...
#   }
#   target_demand_<good>_for_modifier = {
#       value    = target_demand_<good>_ajusted
#       subtract = base_demande_<good>_fix
#       divide   = base_demande_<good>_fix  # <-- can be 0
#       multiply = 100
#   }
#
# The author knows about the first one -- his own comment on that line reads
# "---------------> division par zero possible". building_financial_num is a
# sum of ~90 has_building_financial_centre_<tag> flags, so it is 0 for every
# country that owns no financial centre, which is most of the world early on.
# base_demande_<good>_fix just reads var:base_demande_<good>_fix, which is 0
# until the first financial centre exists.
#
# Fix: clamp the first divisor at 1, and guard the second with a
# `> 0` check so the whole block is skipped instead of dividing.
# Values are otherwise E&F's, line for line.
#
# Scope note: `min = 1` on a `divide = { value = ... }` block clamps the
# divisor, not the result -- so with one or more financial centres the maths
# is bit-identical to E&F's.
#
# Was previously carried by the E&F + PSC compatch. It has nothing to do with
# PSC, so it lives here now.
#############################################

REPLACE_OR_CREATE:target_demand_bond_ajusted = {
	value = target_demand_bond
	subtract = market.mg:bond.market_goods_exports
	divide = { value = building_financial_num min = 1 }

	if = {
		limit = {
			or = {
				is_subject = yes
				not = { market.owner = root }
			}
		}
		multiply = 0.05
	}
	if = { limit = { has_modifier = speculative_share_modifier_2 } multiply = 0.75 }
	if = { limit = { has_modifier = speculative_share_modifier_3 } multiply = 0.5 }
	if = { limit = { has_modifier = speculative_share_modifier_4 } multiply = 0.25 }

	min = 0
}

REPLACE_OR_CREATE:target_demand_bond_for_modifier = {
	value = 0
	if = {
		limit = { base_demande_bond_fix > 0 }
		value = target_demand_bond_ajusted
		subtract = base_demande_bond_fix
		divide = base_demande_bond_fix
		multiply = 100
	}
	max = 25000
}

# 7d. The currency's divisor -- clamped, NOT capped, and NOT guarded
#
#   target_demand_currency_for_modifier = {
#       value    = target_demand_currency_ajusted
#       subtract = base_demande_currency_fix
#       divide   = base_demande_currency_fix   # <-- can be 0
#       multiply = 100
#       #max = 25000                           # <-- author's own, commented out
#   }
#
# WHAT THE NUMBER IS, because two wrong fixes came out of misreading it.
# base_demande_currency_fix is scope:central_bank_scope.modifier:
# goods_output_liquidity_currency_add -- the flat part of the bank's output, which is the
# production method's `goods_output_liquidity_currency_add = 1` workforce-scaled. On a level
# 20 bank that is about 27. The value computed here is fed straight into
# `add_modifier = { name = currency_demande multiplier = ... }`, and currency_demande
# is a MULT. So to take a bank from 27 units to Britain's 100K, the number has to be
# around 370,000, and China needs more. It is a multiplier in the hundreds of
# thousands BY DESIGN.
#
# Which is why there is no cap here. 25000 cut the bank to a fifteenth of what it
# should print, 100000 to a quarter -- both looked like "the formula is broken"
# because they broke it. The author's commented-out 25000 was not a safety net he
# forgot to enable; it would have crippled his own mechanic.
#
# Nor the `if ..._fix > 0` guard the five blocks above use. Their divisor is a
# financial-centre count that grows on its own; this one is what the bank is already
# issuing, and this modifier is what makes it issue. Skipping at zero locks the bank
# at bare production method output forever -- Britain went from 100K to 6.77.
#
# The one thing actually wrong is the divide by zero. It happens on the tick after
# the bank is rebuilt under a new owner, before the production methods are back:
# the add is 0, so the percentage is infinite and the mult is whatever infinity
# becomes. That is the "Britain and China print a million now and then".
#
# Clamping the divisor at 1 -- the way E&F clamps building_financial_num in 7a --
# makes that tick finite instead. It is also harmless: a bank with no add to
# multiply produces nothing however large the mult is, and the next tick has its
# production methods back and divides by the real number again. With a healthy
# divisor the arithmetic is bit-identical to E&F's.
#############################################

REPLACE_OR_CREATE:target_demand_currency_for_modifier = {
	value = target_demand_currency_ajusted
	subtract = base_demande_currency_fix
	divide = { value = base_demande_currency_fix min = 1 }
	multiply = 100
}
