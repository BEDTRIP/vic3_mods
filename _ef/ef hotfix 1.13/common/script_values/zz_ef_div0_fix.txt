# 7a. Division by zero in the financial demand values
#
# E&F computes the demand modifiers for its five stock/bond goods like this
# (common/script_values/00_financial_scripted_value.txt):
#
#   target_demand_<good>_ajusted = {
#       value    = target_demand_<good>
#       subtract = market.mg:<good>.market_goods_exports
#       divide   = building_financial_num   # <-- can be 0
#       ...
#   }
#   target_demand_<good>_for_modifier = {
#       value    = target_demand_<good>_ajusted
#       subtract = base_demande_<good>_fix
#       divide   = base_demande_<good>_fix  # <-- can be 0
#       multiply = 100
#   }
#
# The author knows about the first one -- his own comment on that line reads
# "---------------> division par zero possible". building_financial_num is a
# sum of ~90 has_building_financial_centre_<tag> flags, so it is 0 for every
# country that owns no financial centre, which is most of the world early on.
# base_demande_<good>_fix just reads var:base_demande_<good>_fix, which is 0
# until the first financial centre exists.
#
# Fix: clamp the first divisor at 1, and guard the second with a
# `> 0` check so the whole block is skipped instead of dividing.
# Values are otherwise E&F's, line for line.
#
# Scope note: `min = 1` on a `divide = { value = ... }` block clamps the
# divisor, not the result -- so with one or more financial centres the maths
# is bit-identical to E&F's.
#
# Was previously carried by the E&F + PSC compatch. It has nothing to do with
# PSC, so it lives here now.
#############################################

REPLACE_OR_CREATE:target_demand_bond_ajusted = {
	value = target_demand_bond
	subtract = market.mg:bond.market_goods_exports
	divide = { value = building_financial_num min = 1 }

	if = {
		limit = {
			or = {
				is_subject = yes
				not = { market.owner = root }
			}
		}
		multiply = 0.05
	}
	if = { limit = { has_modifier = speculative_share_modifier_2 } multiply = 0.75 }
	if = { limit = { has_modifier = speculative_share_modifier_3 } multiply = 0.5 }
	if = { limit = { has_modifier = speculative_share_modifier_4 } multiply = 0.25 }

	min = 0
}

REPLACE_OR_CREATE:target_demand_bond_for_modifier = {
	value = 0
	if = {
		limit = { base_demande_bond_fix > 0 }
		value = target_demand_bond_ajusted
		subtract = base_demande_bond_fix
		divide = base_demande_bond_fix
		multiply = 100
	}
	max = 25000
}

# 7d. The same division by zero, in the currency -- and why the guard the other
#     five use is WRONG here
#
# target_demand_currency_for_modifier is built like the five above:
#
#   value    = target_demand_currency_ajusted
#   subtract = base_demande_currency_fix
#   divide   = base_demande_currency_fix   # <-- can be 0
#   multiply = 100
#   #max = 25000                           # <-- author's own, commented out
#
# It is a PERCENTAGE: how far this month's target sits from what the bank is
# already issuing. It reaches the bank as goods_output_spe_uni_c_mult at 0.01 per
# point, so 100 means "double it". Unbounded, with a divisor near zero, it means
# Britain and China printing a million now and then.
#
# THE `if base_demande_..._fix > 0` GUARD FROM 7a-7c MUST NOT BE USED HERE, and
# that was one round of everybody being broke. For bonds and stocks the divisor is
# a financial-centre count that becomes positive on its own. For the currency it is
# var:base_demande_currency_fix, refreshed from base_demande_currency, which reads
# what the central bank is ALREADY issuing. Zero until the bank issues something,
# and the modifier is what makes it issue -- so skipping the block at zero locks
# the bank at its bare production method output forever. Britain went from 100K to
# 6.77.
#
# So the divisor is clamped instead, the way E&F clamps building_financial_num in
# 7a: at zero output the value becomes target x 100, which is enormous but finite,
# and the cap turns it into a fast ramp rather than an explosion. From a standing
# start the bank reaches its target in two or three monthly ticks.
#
# The cap is a percentage, not a quantity -- 100000 is "up to a thousandfold in one
# tick", not "at most 100K currency". It exists only to keep a near-zero divisor
# from producing infinity; it never touches a bank running near its target, where
# the value sits close to zero.
#############################################

REPLACE_OR_CREATE:target_demand_currency_for_modifier = {
	value = target_demand_currency_ajusted
	subtract = base_demande_currency_fix
	divide = { value = base_demande_currency_fix min = 1 }
	multiply = 100
	max = 100000
}
