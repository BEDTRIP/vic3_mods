# EF.30 (2026-09-30, v2 same day) -- the central bank's rate, as values.
# Rates are fractions (0.05 = 5%), as E&F's var:base_rate_percentage.
#
# TARGET. The credit note (var:country_credit_note_fixe, set by E&F's
# country_credit_rating from country_credit_pondered_mean) runs from 0 to
# 12.5: the score part is 0..10 (country_credit_score / 31), the central bank
# institution adds 0.5 per level (5 levels, +2.5), taking loans x0.75, war and
# heavy debt multiply it further down. A rank-1 AI is pinned to 10 by E&F.
# The scale 0..10 maps linearly onto 12%..2%: 1 pp per point; above 10
# (only with the institution, 12.5 is practically out of reach) stays at 2%.
#   note 10+ -> 2%, 8.6 (USA 1836) -> 3.4%, 6 -> 6%, 4 -> 8%, 0 -> 12%.
# (v2 had 12.5 -> 2%, 0.8 pp per point: the top of the scale was unreachable,
# changed 2026-09-30.)
# A country WITHOUT a central bank has no bank to set a rate and a note
# counted from its rank alone (1..4); it is held at 6.5%, E&F's neutral
# zone, where the yearly pool credit is zero.
# The player's policy (var:zz_ef_rate_bias, +-2 pp, the rate buttons) is added
# on top (the bank alone stays within 2..12%, policy may take the target
# 2 pp under the bank, but never below 0.5%).
#
# STEP. Every 3 months (on_actions/zz_ef_cb_rate_on_actions.txt) the rate
# moves towards the target by at most 0.5 pp, or 1 pp when the gap is over
# 5 pp.
#############################################

# The bank's own rate, from the rating alone: 2..12%. Shown as the "policy
# rule rate" (concept_zz_ef_policy_rule_rate): what the bank computes, WITHOUT
# the government's discretionary adjustment. Kept apart from the rate the bank
# steers to (zz_ef_cb_rate_target = rule + adjustment): penalties for the gap
# between the actual rate and this one come later (EF.31).
zz_ef_cb_rate_rating_target = {
	value = 0.065
	if = {
		limit = {
			has_modifier = has_central_bank
			has_variable = country_credit_note_fixe
		}
		value = 0.12
		subtract = {
			value = var:country_credit_note_fixe
			multiply = 0.01
		}
	}
	min = 0.02
	max = 0.12
}

# The target: the bank's rate + the player's policy (+-2 pp), 0.5..12%.
# (2026-09-30: the bank's floor stays 2%, policy can go 2 pp under it, but
# never below 0.5%.)
zz_ef_cb_rate_target = {
	value = zz_ef_cb_rate_rating_target
	if = {
		limit = { has_variable = zz_ef_rate_bias }
		add = var:zz_ef_rate_bias
	}
	min = 0.005
	max = 0.12
}

# Target minus the current rate (positive: the bank will raise).
zz_ef_cb_rate_gap = {
	value = zz_ef_cb_rate_target
	if = {
		limit = { has_variable = base_rate_percentage }
		subtract = var:base_rate_percentage
	}
}

zz_ef_cb_rate_step_max = {
	value = 0.005
	if = {
		limit = {
			OR = {
				zz_ef_cb_rate_gap > 0.05
				zz_ef_cb_rate_gap < -0.05
			}
		}
		value = 0.01
	}
}
zz_ef_cb_rate_step_min = {
	value = zz_ef_cb_rate_step_max
	multiply = -1
}

# What the next quarterly step will do to the rate.
zz_ef_cb_rate_next_step = {
	value = zz_ef_cb_rate_gap
	min = zz_ef_cb_rate_step_min
	max = zz_ef_cb_rate_step_max
}

# Months until the next step (the counter runs 0, 1, 2).
zz_ef_cb_rate_months_to_step = {
	value = 3
	if = {
		limit = { has_variable = zz_ef_cb_rate_month }
		subtract = var:zz_ef_cb_rate_month
	}
}

# The player's policy for display (0 when never used).
zz_ef_cb_rate_bias_value = {
	value = 0
	if = {
		limit = { has_variable = zz_ef_rate_bias }
		value = var:zz_ef_rate_bias
	}
}

# The credit note for display.
zz_ef_cb_rate_note_value = {
	value = 0
	if = {
		limit = { has_variable = country_credit_note_fixe }
		value = var:country_credit_note_fixe
	}
}

# The policy rule rate minus the actual rate (positive: the rule wants a
# higher rate than we have). Colours the rule box (scripted_guis
# zz_ef_cb_rule_above_rate / _below_rate), shown in its tooltip.
zz_ef_cb_rule_gap = {
	value = zz_ef_cb_rate_rating_target
	if = {
		limit = { has_variable = base_rate_percentage }
		subtract = var:base_rate_percentage
	}
}

# Display in percentage points.
zz_ef_cb_rule_gap_pp = {
	value = zz_ef_cb_rule_gap
	multiply = 100
}
zz_ef_cb_rate_next_step_pp = {
	value = zz_ef_cb_rate_next_step
	multiply = 100
}
zz_ef_cb_rate_bias_pp = {
	value = zz_ef_cb_rate_bias_value
	multiply = 100
}
