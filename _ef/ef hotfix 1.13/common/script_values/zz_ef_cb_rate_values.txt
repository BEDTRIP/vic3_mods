# EF.30 (2026-09-30, v2 same day) -- the central bank's rate, as values.
# Rates are fractions (0.05 = 5%), as E&F's var:base_rate_percentage.
#
# TARGET (EF.45 + EF.43, 2026-09-30, replaces the rating-only 12%..2%).
# A Fisher-style rule, decided by the user (План проекта.md, EF.39/EF.45):
#   rule rate = neutral real rate 2.5%
#             + inflation: E&F's consumer goods inflation, 6-month rolling,
#               x2 to a year, held to -2..+6 pp on metal (+-20 on fiat) --
#               the other goods groups E&F sums are too noisy (Britain 1848:
#               military equipment +8.6% in six months);
#             + risk premium from the credit note (var:country_credit_note_fixe,
#               0..12.5): 0.6 pp per point under 10, 0..6 pp -- the rating now
#               prices risk only (EF.43), as spreads over consols did;
#             + metal cover: below 75% +1 pp, below 50% +2 pp, above 125%
#               -0.5 pp (metal standards only; E&F's
#               central_bank_metal_reserves_state_to_ensure_the_conversion_pourcent).
#   Corridor by the monetary standard (ДС.1 will give each its own balance):
#     metal (silver, bimetallic, gold, gold exchange, external) 2..12%;
#     fiat and no system 0.5..25%.
# A country WITHOUT a central bank has no bank to set a rate; it is held at
# 6.5%.
# The player's policy (var:zz_ef_rate_bias, +-2 pp, the rate buttons) is added
# on top; policy may take the target 2 pp under the corridor's floor, never
# below 0.5%.
#
# STEP. Every 3 months (on_actions/zz_ef_cb_rate_on_actions.txt) the rate
# moves towards the target by at most 0.5 pp, or 1 pp when the gap is over
# 5 pp.
#############################################

# The bank's own rate, from the rating alone: 2..12%. Shown as the "policy
# rule rate" (concept_zz_ef_policy_rule_rate): what the bank computes, WITHOUT
# the government's discretionary adjustment. Kept apart from the rate the bank
# steers to (zz_ef_cb_rate_target = rule + adjustment): penalties for the gap
# between the actual rate and this one come later (EF.31).
zz_ef_cb_rate_is_fiat = {
	value = 0
	if = {
		limit = {
			OR = {
				has_law = law_type:law_fiat_standard
				has_law = law_type:law_no_monetary_system
			}
		}
		value = 1
	}
}
zz_ef_cb_rate_floor = {
	value = 0.02
	if = {
		limit = { zz_ef_cb_rate_is_fiat = 1 }
		value = 0.005
	}
}
zz_ef_cb_rate_ceiling = {
	value = 0.12
	if = {
		limit = { zz_ef_cb_rate_is_fiat = 1 }
		value = 0.25
	}
}

# Parts of the rule, as fractions (shown in the rule box tooltip).
zz_ef_cb_rule_neutral = 0.025
# EF.48 item 12 (1.10): the inflation from the money accounting (a year,
# zz_ef_inflation in zz_ef_money_model_values.txt) replaces E&F's consumer
# goods inflation x2.
zz_ef_cb_rule_inflation = {
	value = zz_ef_inflation
	if = {
		limit = { zz_ef_cb_rate_is_fiat = 1 }
		min = -0.2
		max = 0.2
	}
	else = {
		min = -0.02
		max = 0.06
	}
}
zz_ef_cb_rule_rating = {
	value = 0.06
	if = {
		limit = { has_variable = country_credit_note_fixe }
		value = 10
		subtract = var:country_credit_note_fixe
		multiply = 0.006
	}
	min = 0
	max = 0.06
}
# EF.43 item 14 (1.10): the risk premium beyond E&F's rating, which bunches
# the big countries together (Russia = France = 8.6 in 1836, while history
# had Russia borrowing 2-7 pp dearer): a default +3 pp; instability -- the
# radicals' share x 10 (up to +2 pp), legitimacy under 50 +0.5 / under 25
# +1 pp, war +0.5 pp; an illiterate state collects its taxes and keeps its
# books worse -- (40% - literacy) x 2.5, up to +1 pp. Weights decided at
# night -- check.
zz_ef_cb_rule_stability = {
	value = 0
	if = {
		limit = { in_default = yes }
		add = 0.03
	}
	add = {
		value = radical_population_fraction
		multiply = 0.1
		max = 0.02
	}
	if = {
		limit = { legitimacy < 25 }
		add = 0.01
	}
	else_if = {
		limit = { legitimacy < 50 }
		add = 0.005
	}
	if = {
		limit = { is_at_war = yes }
		add = 0.005
	}
	add = {
		value = 0.4
		subtract = literacy_rate
		multiply = 0.025
		min = 0
		max = 0.01
	}
}
# The premium: the rating's part + the stability part, 0..8 pp.
zz_ef_cb_rule_risk = {
	value = zz_ef_cb_rule_rating
	add = zz_ef_cb_rule_stability
	min = 0
	max = 0.08
}
zz_ef_cb_rule_cover = {
	value = 0
	if = {
		limit = { zz_ef_cb_rate_is_fiat = 0 }
		if = {
			limit = { central_bank_metal_reserves_state_to_ensure_the_conversion_pourcent < 0.5 }
			value = 0.02
		}
		else_if = {
			limit = { central_bank_metal_reserves_state_to_ensure_the_conversion_pourcent < 0.75 }
			value = 0.01
		}
		else_if = {
			limit = { central_bank_metal_reserves_state_to_ensure_the_conversion_pourcent > 1.25 }
			value = -0.005
		}
	}
}

# The bank's own rate (the "policy rule rate",
# concept_zz_ef_policy_rule_rate): what the bank computes, WITHOUT the
# government's discretionary adjustment. Kept apart from the rate the bank
# steers to (zz_ef_cb_rate_target = rule + adjustment): penalties for the gap
# between the actual rate and this one come later (EF.31).
# (The name stays from v1, when it was read off the rating alone.)
zz_ef_cb_rate_rating_target = {
	value = 0.065
	if = {
		limit = { has_modifier = has_central_bank }
		value = zz_ef_cb_rule_neutral
		add = zz_ef_cb_rule_inflation
		add = zz_ef_cb_rule_risk
		add = zz_ef_cb_rule_cover
		min = zz_ef_cb_rate_floor
		max = zz_ef_cb_rate_ceiling
	}
}

# The target: the bank's rate + the player's policy (+-2 pp), from 2 pp
# under the corridor's floor (never below 0.5%) to its ceiling.
zz_ef_cb_rate_target_floor = {
	value = zz_ef_cb_rate_floor
	subtract = 0.02
	min = 0.005
}
zz_ef_cb_rate_target = {
	value = zz_ef_cb_rate_rating_target
	if = {
		limit = { has_variable = zz_ef_rate_bias }
		add = var:zz_ef_rate_bias
	}
	min = zz_ef_cb_rate_target_floor
	max = zz_ef_cb_rate_ceiling
}

# Target minus the current rate (positive: the bank will raise).
zz_ef_cb_rate_gap = {
	value = zz_ef_cb_rate_target
	if = {
		limit = { has_variable = base_rate_percentage }
		subtract = var:base_rate_percentage
	}
}

zz_ef_cb_rate_step_max = {
	value = 0.005
	if = {
		limit = {
			OR = {
				zz_ef_cb_rate_gap > 0.05
				zz_ef_cb_rate_gap < -0.05
			}
		}
		value = 0.01
	}
}
zz_ef_cb_rate_step_min = {
	value = zz_ef_cb_rate_step_max
	multiply = -1
}

# What the next quarterly step will do to the rate.
zz_ef_cb_rate_next_step = {
	value = zz_ef_cb_rate_gap
	min = zz_ef_cb_rate_step_min
	max = zz_ef_cb_rate_step_max
}

# Months until the next step (the counter runs 0, 1, 2).
zz_ef_cb_rate_months_to_step = {
	value = 3
	if = {
		limit = { has_variable = zz_ef_cb_rate_month }
		subtract = var:zz_ef_cb_rate_month
	}
}

# The player's policy for display (0 when never used).
zz_ef_cb_rate_bias_value = {
	value = 0
	if = {
		limit = { has_variable = zz_ef_rate_bias }
		value = var:zz_ef_rate_bias
	}
}

# The credit note for display.
zz_ef_cb_rate_note_value = {
	value = 0
	if = {
		limit = { has_variable = country_credit_note_fixe }
		value = var:country_credit_note_fixe
	}
}

# The policy rule rate minus the actual rate (positive: the rule wants a
# higher rate than we have). Colours the rule box (scripted_guis
# zz_ef_cb_rule_above_rate / _below_rate), shown in its tooltip.
zz_ef_cb_rule_gap = {
	value = zz_ef_cb_rate_rating_target
	if = {
		limit = { has_variable = base_rate_percentage }
		subtract = var:base_rate_percentage
	}
}

# Display in percentage points.
zz_ef_cb_rule_gap_pp = {
	value = zz_ef_cb_rule_gap
	multiply = 100
}
zz_ef_cb_rate_next_step_pp = {
	value = zz_ef_cb_rate_next_step
	multiply = 100
}
zz_ef_cb_rate_bias_pp = {
	value = zz_ef_cb_rate_bias_value
	multiply = 100
}

# Parts of the rule in percentage points, for the rule box tooltip.
zz_ef_cb_rule_neutral_pp = {
	value = zz_ef_cb_rule_neutral
	multiply = 100
}
zz_ef_cb_rule_inflation_pp = {
	value = zz_ef_cb_rule_inflation
	multiply = 100
}
zz_ef_cb_rule_risk_pp = {
	value = zz_ef_cb_rule_risk
	multiply = 100
}
zz_ef_cb_rule_cover_pp = {
	value = zz_ef_cb_rule_cover
	multiply = 100
}

# ---- EF.31 items 8-9 (1.10): the price of the government's policy ---------
zz_ef_rate_bias_signed = {
	value = 0
	if = { limit = { has_variable = zz_ef_rate_bias } value = var:zz_ef_rate_bias }
}
# 0.5 pp steps of the policy off the CB's target (rounded to the nearest).
zz_ef_rate_policy_steps = {
	value = zz_ef_rate_bias_signed
	abs = yes
	divide = 0.005
	round = yes
}
zz_ef_rate_policy_ig_steps = {
	value = zz_ef_rate_policy_steps
	max = 5
}
zz_ef_cb_rule_rating_pp = {
	value = zz_ef_cb_rule_rating
	multiply = 100
}
zz_ef_cb_rule_stability_pp = {
	value = zz_ef_cb_rule_stability
	multiply = 100
}
