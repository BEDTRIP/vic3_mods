# EF.17 (4b) (2026-09-24) -- exchange margin 1.5 -> 1.25
#
# How E&F sizes an exchange (00_financial_scripted_value.txt:3595-3610, 4.1.7):
#   total_cost_fcc                    what the exchange pays a week for its
#                                     five inputs (four stocks and bonds)
#   collateral_quantity_mutual_fund = total_cost_fcc / 250
#   target_supply_mutual_fund       = that x 1.5        <-- this file
#   -> target_supply_mutual_fund_for_modifier (7e) turns it into the output
#      multiplier, so the exchange always sells 1.5 x what it buys.
#
# The whole 50% is value added, and value added per worker is what makes an
# exchange the best investment in the country (~2800 per worker seen in
# Petersburg) and what lands in GDP (the EF.17 loop). 1.25 halves that margin;
# the exchange still pays for itself and still follows demand.
#
# Why here and not on the 7e ceiling: capping the output multiplier alone was
# tried the same day and bankrupted every exchange (see 7e). This value moves
# output and cost together.
#
# !! TUNING !! the 1.25. 1.0 = no value added at all (an exchange that only
# pays wages out of nothing -- do not go below ~1.1).
# !! MAINTENANCE !! key-level REPLACE_OR_CREATE of a 3-line E&F value.
#############################################

REPLACE_OR_CREATE:target_supply_mutual_fund = {
	value = collateral_quantity_mutual_fund
	multiply = 1.25	### EF.17 (4b): E&F has 1.5
}

#############################################
# EF.33 (Ф.7 а, night 5-6.10.2026) -- the margin is 1.25 only on paper: x2.5 in fact (London 1847: buys 603K a
# week, sells 1.53M; value added ~18% of Britain's GDP). total_cost_fcc prices E&F's FORMULA demand
# (target_demand_<good>_ajusted), while the exchange's actual purchase is that demand through the input modifier
# capped at +25000% (target_demand_<good>_for_modifier, max = 25000: at most base x 251) -- the larger the market,
# the further the formula runs past what is really bought, and the output follows the formula.
# Now each part of total_cost_fcc is priced on the demand the cap lets through: min(adjusted demand,
# base x 251) -- the same quantity the input modifier buys, so the output is 1.25 x the real purchase (the base
# read only where E&F has set it, as 7a guards it).
# Still needed after Д2.10 (capitalization from listed dividends): the exchange's value added is GDP and profit,
# and the profit now feeds the capitalization.
# !! MAINTENANCE !! key-level REPLACE_OR_CREATE of E&F's five *_price_by_demand_fc (00_financial_scripted_value.txt).
#############################################

REPLACE_OR_CREATE:bond_price_by_demand_fc = {
	value = target_demand_bond_ajusted
	if = {
		limit = { has_variable = base_demande_bond_fix }
		max = { value = var:base_demande_bond_fix multiply = 251 min = 0 }
	}
	multiply = bond_price_brut
}
