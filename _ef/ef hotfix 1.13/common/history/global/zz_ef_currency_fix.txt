### E&F Hotfix [1.13] — currency law corrections
###
### This file is additive: GLOBAL blocks under common/history/global/ stack,
### and "zz_" is processed after "99_ef_history_global_variable.txt", so the
### activate_law calls here land last.

GLOBAL = {
	every_country = {

		########################################################################
		# 1. Wuerttemberg: a typo in E&F.
		#    99_ef_history_global_variable.txt hands WUR the law
		#    law_gulden_south_german_gulde_currency — no trailing "n". No such
		#    law exists, activate_law silently does nothing, and Wuerttemberg
		#    ends up with no currency at all even though the good
		#    gulden_south_german_gulden_c is alive.
		########################################################################
		if = {
			limit = { c:WUR ?= this }
			activate_law = law_type:law_gulden_south_german_gulden_currency
		}

		########################################################################
		# 2. Countries whose currency the E&F author commented out in
		#    ef_00_goods.txt while leaving the activate_law in history. The law
		#    is active but the good is gone: the central bank mints nothing,
		#    buildings pay nothing for liquidity, and the stock
		#    pm_no_currency_type / pm_no_market_liquidity never switch on
		#    because law_no_market_liquidity is what unlocks them.
		#    Move them to the legitimate "no currency" state.
		#
		# 3. Plus the ones we cut ourselves, to free good slots for
		#    Morgenroete and PSC (the 128 goods limit).
		########################################################################
		if = {
			limit = {
				OR = {
					c:LIB ?= this		# Liberia       — dollar_liberian_dollar
					c:COS ?= this		# Costa Rica    — peso_costa_rican_colon
					c:ECU ?= this		# Ecuador       — peso_ecuadorian_peso
					c:ELS ?= this		# El Salvador   — peso_el_salvador_colon
					c:GUA ?= this		# Guatemala     — peso_guatemalan_quetzal
					c:HON ?= this		# Honduras      — peso_honduran_lempira
					c:NIC ?= this		# Nicaragua     — peso_nicaraguan_cordoba
					c:PRG ?= this		# Paraguay      — peso_paraguayan_peso
					c:URU ?= this		# Uruguay       — peso_uruguayan_peso
					c:VNZ ?= this		# Venezuela     — peso_venezuelan_peso
					c:DAI ?= this		# Dai Viet      — spe_dong
					c:HAI ?= this		# Haiti         — dollar_caribbean_dollar  (cut by us)
					c:NZL ?= this		# New Zealand   — dollar_new_zealand_dollar (cut by us)
				}
			}
			activate_law = law_type:law_no_market_liquidity
		}


		########################################################################
		# 4. Initial liquidity_currency grant (minor countries).
		#    E&F's flat 2500 is disabled at the source, so without this block a
		#    country would sit with no currency at all for the whole first
		#    month: the game spreads on_monthly_pulse_country across the days
		#    of the month.
		#
		#    The condition may well not fire: E&F applies no_money_production
		#    from its own pulse, and it may not be there yet when history runs.
		#    In that case nothing happens here and the pulse does the work.
		#
		#    Grants liquidity_currency, not a separate local_currency (unified
		#    02.09.2026, same evening as the split -- see EF.10 in
		#    План проекта.md).
		########################################################################
		if = {
			limit = {
				zz_ef_local_currency_needs_throttle = yes
				zz_ef_local_currency_per_state > 0
			}
			add_modifier = {
				name = zz_ef_local_currency_fix
				multiplier = zz_ef_local_currency_per_state
			}
		}
	}
}
