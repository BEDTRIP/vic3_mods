# 7c. Init for the stockpile state variables E&F reads but does not set
#
# THE ERRORS (error.log: 474 blocks in one session, 270 of them these five)
#   Failed to fetch variable for 'stockpiling_bond_var_state_1' due to not being set
#   Event target link 'var' returned an unset scope
#   Invalid left side during comparison 'var'
#     common/scripted_effects/01_economic_scripted_effects.txt   (x18 each, five variables)
#     <- common/on_actions/00_ef_on_action.txt:96
#     <- common/scripted_effects/00_on_action_main.txt:21
#     <- common/scripted_effects/00_on_action_main.txt:18431
#
# WHERE THEY COME FROM
# `stockpiling_capital_state_transfert` runs on the monthly pulse of any country
# that owns a bank, and gates its financial-centre branch on
#
#   any_scope_state = {
#       var:financial_center_site_var = 0
#       or = {
#           var:stockpiling_bond_var_state_1 > 0
#           var:stockpiling_manufacture_stock_var_state_1 > 0
#           var:stockpiling_agricultural_stock_var_state_1 > 0
#           var:stockpiling_mining_stock_var_state_1 > 0
#           var:stockpiling_railroad_stock_var_state_1 > 0
#       }
#   }
#
# `any_scope_state` walks every state the country owns, so all five have to
# exist on all of them or the read throws three errors per state per month.
#
# GLOBAL blocks are additive and `zz_` is processed last, so this appends to
# whatever E&F sets itself. Every write is guarded by `NOT = { has_variable }`,
# so it only fills gaps and never fights E&F over a value.
#
# !! every_state, NOT every_scope_state !!
# GLOBAL runs in `none` scope. An earlier version of this file used
# `every_scope_state` at the top level, and the entire block silently did
# nothing:
#
#   Error: every_scope_state effect [ Wrong scope for effect: none, expected
#          country, front, state_region, strait, strategic_region, theater ]
#   Script location: common/history/global/zz_ef_init_stockpiling_state_vars.txt:31
#
# `every_scope_state` needs a country or region to iterate. `every_state` is the
# top-level list, and it is what E&F itself uses in
# common/history/global/01_ef_state_global_variable.txt:2. Nesting as
# `every_country = { every_scope_state = { ... } }` also works, but it misses
# states with no owner and repeats the work once per country.
#
# Was previously carried by the E&F + PSC compatch. It has nothing to do with
# PSC, so it lives here now.
#############################################

GLOBAL = {
	every_state = {
		# E&F financial-centre stockpile transfer vars
		if = {
			limit = { NOT = { has_variable = stockpiling_bond_var_state_1 } }
			set_variable = { name = stockpiling_bond_var_state_1 value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = stockpiling_manufacture_stock_var_state_1 } }
			set_variable = { name = stockpiling_manufacture_stock_var_state_1 value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = stockpiling_agricultural_stock_var_state_1 } }
			set_variable = { name = stockpiling_agricultural_stock_var_state_1 value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = stockpiling_mining_stock_var_state_1 } }
			set_variable = { name = stockpiling_mining_stock_var_state_1 value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = stockpiling_railroad_stock_var_state_1 } }
			set_variable = { name = stockpiling_railroad_stock_var_state_1 value = 0 }
		}

		# Common markers used by E&F stockpile transfer code
		if = {
			limit = { NOT = { has_variable = financial_center_site_var } }
			set_variable = { name = financial_center_site_var value = 0 }
		}
		if = {
			limit = { NOT = { has_variable = looted_state } }
			set_variable = { name = looted_state value = 0 }
		}
	}
}
