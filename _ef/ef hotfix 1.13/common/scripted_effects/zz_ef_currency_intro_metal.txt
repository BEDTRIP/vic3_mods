# В2.4 (decided 4.10, М.10 а; 5.10): E&F's metal "from nothing" at a currency's introduction is switched off.
# E&F's introduction_of_<currency> (95 effects, 09_introduction_building_lvl.txt, ~190 edits) adds to the capital's
# gold_state_1 / silver_state_1 (money in E&F's count + its stock + building savings) x the new parity -- a cover far
# over 100% out of nowhere; reset_law_event_currency (the currency's reset after a crisis) zeroes the metal
# (scripted_effects/zz_ef_currency_reset_keep_metal.txt keeps it now). Wrapped around introduction_new_currency
# (zz_ef_cm_bank_spawn.txt):
#   * the first 13 weeks: untouched -- the start's metal is E&F's and the model rescales it to 40% (M.1);
#   * later: the capital's metal is put back as it was before the call;
#   * a FIRST currency (no var:country_already_central_bank before the call) on a metal standard gets the norm --
#     40% of the model's M2 at the new parity, in the standard's metal (gold for bimetallism) -- as the start does
#     for countries without a CB (M.8); logged (EFM|...|cur_intro) and seen by tools/check_flows.py as a jump of the
#     world metal. Accepted 5.10, check: the norm is still metal that nobody gave; with the pops' metal (В8.1,
#     stage 3) it can be bought from them.
#   * 6.10 (М, night of stage 2): the norm is gone -- a new CB starts with no metal and buys it (norm logged as 0).

zz_ef_cur_intro_before = {
	# 5.10 (run s1b): history can call it for a country with no capital yet
	if = {
		limit = { exists = capital }
		if = { limit = { has_variable = zz_ef_cur_intro_first } remove_variable = zz_ef_cur_intro_first }
		if = {
			limit = { NOT = { has_variable = country_already_central_bank } }
			set_variable = { name = zz_ef_cur_intro_first value = yes }
		}
		capital = {
			set_variable = { name = zz_ef_cur_intro_gold value = 0 }
			set_variable = { name = zz_ef_cur_intro_silver value = 0 }
			if = {
				limit = { has_variable = gold_state_1 }
				set_variable = { name = zz_ef_cur_intro_gold value = var:gold_state_1 }
			}
			if = {
				limit = { has_variable = silver_state_1 }
				set_variable = { name = zz_ef_cur_intro_silver value = var:silver_state_1 }
			}
		}
	}
}

zz_ef_cur_intro_after = {
	if = {
		limit = { exists = capital }
		# 6.10 (stage 2, run r1006_132739): from the country's first step on -- E&F's metal at an introduction in the first
		# 13 weeks was left to the week-12 rescale, and the world line of metal read +370M gold and +4.4 bn silver from
		# nothing in one week (17.2.1836); before the first step it is E&F's start, which the first step takes as stock
		if = {
			limit = {
				has_variable = zz_ef_metal_started
			}
			capital = {
				set_variable = { name = zz_ef_cur_intro_g_ef value = 0 }
				if = {
					limit = { has_variable = gold_state_1 }
					set_variable = { name = zz_ef_cur_intro_g_ef value = var:gold_state_1 }
				}
				set_variable = { name = gold_state_1 value = var:zz_ef_cur_intro_gold }
				set_variable = { name = silver_state_1 value = var:zz_ef_cur_intro_silver }
			}
			# М (the user, 6.10): a first currency got 40% of its M2 in metal from nothing (the norm) -- removed; its CB
			# buys metal on the market like every CB (its building's input, zz_ef_cb_metal_purchase_step)
			set_variable = { name = zz_ef_cur_intro_norm value = 0 }
			debug_log = "EFM|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|cur_intro|first [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cur_intro_first_v')|0]|gold_ef [THIS.GetCountry.GetCapital.MakeScope.Var('zz_ef_cur_intro_g_ef').GetValue|0]|gold_kept [THIS.GetCountry.GetCapital.MakeScope.Var('zz_ef_cur_intro_gold').GetValue|0]|norm [THIS.GetCountry.MakeScope.Var('zz_ef_cur_intro_norm').GetValue|0]|parity [THIS.GetCountry.MakeScope.Var('money_value_target_1').GetValue|4]"
			capital = {
				if = { limit = { has_variable = zz_ef_cur_intro_g_ef } remove_variable = zz_ef_cur_intro_g_ef }
			}
		}
		capital = {
			if = { limit = { has_variable = zz_ef_cur_intro_gold } remove_variable = zz_ef_cur_intro_gold }
			if = { limit = { has_variable = zz_ef_cur_intro_silver } remove_variable = zz_ef_cur_intro_silver }
		}
		if = { limit = { has_variable = zz_ef_cur_intro_first } remove_variable = zz_ef_cur_intro_first }
		# В3.2 (5.10): E&F just gave a subject its own currency -- back to the overlord's zone
		if = {
			limit = { is_subject = yes }
			zz_ef_cur_zone_step = yes
		}
	}
}
