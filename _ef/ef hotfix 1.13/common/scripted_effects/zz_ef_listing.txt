# Д2.11 / Д2.12 (5.10.2026, decided by the user) -- the "publicly traded" method of company headquarters
# (EF.22, production_methods/zz_ef_company_hq_publicly_traded.txt) is switched by script, monthly, not by hand
# (the user: "this method must become very important and be switched by script"):
#   * allowed (scripted_triggers/zz_ef_listing_triggers.txt: an exchange, joint-stock law, the technology) --
#     the largest headquarters go public until their levels reach the country's share = the literacy of the
#     rich (var:zz_ef_rich_literacy, zz_ef_stock_issue_update): reading a prospectus took literate investors;
#     the rest are private;
#   * not allowed (no exchange, traditionalism...) -- every public headquarters goes back to private.
# activate_production_method works on a building type in a state, so the switch is per state: two
# headquarters in one state share it (the last one wins).
# Logged for big economies: EFL|date|country|hq H|listed L|share S|lit R|cap C|div D|cap_gdp G.
# COUNTRY scope, from on_actions/zz_ef_capitalization_on_actions.txt before the capitalization update.
#############################################

zz_ef_listing_update = {
	set_local_variable = { name = zz_lst_tot value = zz_ef_hq_levels }
	if = {
		limit = {
			zz_ef_listing_allowed = yes
			local_var:zz_lst_tot > 0
		}
		set_local_variable = { name = zz_lst_lit value = 0 }
		if = {
			limit = { has_variable = zz_ef_rich_literacy }
			set_local_variable = { name = zz_lst_lit value = var:zz_ef_rich_literacy }
		}
		set_local_variable = {
			name = zz_lst_target
			value = {
				value = local_var:zz_lst_tot
				multiply = local_var:zz_lst_lit
			}
		}
		set_local_variable = { name = zz_lst_cum value = 0 }
		ordered_scope_building = {
			limit = {
				is_building_type = building_company_headquarter
				level > 0
			}
			order_by = level
			max = 1000
			check_range_bounds = no
			if = {
				limit = { local_var:zz_lst_cum < local_var:zz_lst_target }
				if = {
					limit = { NOT = { has_active_production_method = pm_company_headquarter_publicly_traded } }
					state = {
						activate_production_method = {
							building_type = building_company_headquarter
							production_method = pm_company_headquarter_publicly_traded
						}
					}
				}
			}
			else_if = {
				limit = { has_active_production_method = pm_company_headquarter_publicly_traded }
				state = {
					activate_production_method = {
						building_type = building_company_headquarter
						production_method = pm_company_headquarter_privately_owned
					}
				}
			}
			change_local_variable = { name = zz_lst_cum add = level }
		}
	}
	else = {
		every_scope_building = {
			limit = {
				is_building_type = building_company_headquarter
				has_active_production_method = pm_company_headquarter_publicly_traded
			}
			state = {
				activate_production_method = {
					building_type = building_company_headquarter
					production_method = pm_company_headquarter_privately_owned
				}
			}
		}
	}
}

zz_ef_listing_log = {
	if = {
		limit = {
			OR = {
				is_player = yes
				gdp > 20000000
			}
		}
		debug_log = "EFL|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|hq [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_hq_levels')|1]|listed [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_listed_hq_levels')|1]|lshare [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_listed_share')|3]|cshare [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_company_share')|3]|lit [THIS.GetCountry.MakeScope.Var('zz_ef_rich_literacy').GetValue|3]|div [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_listed_div_year')|0]|snap [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cap_listed_snapshot')|0]|cap [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_equity_capitalization')|0]|cap_gdp [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cap_to_gdp')|3]|disc [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cap_discount')|4]"
	}
}
