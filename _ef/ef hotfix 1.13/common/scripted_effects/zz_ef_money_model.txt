# EF.39 / EF.48 (2026-09-30) -- the money model, monthly step, in the
# engine's money: bank credit borrowed from the central bank at the key rate,
# its interest to the treasury, and the ledger of each account's change.
# Numbers and the why: script_values/zz_ef_money_model_values.txt. Called from
# on_actions/zz_ef_money_model_on_actions.txt for every country.
#
# E&F's investement_pool_borrowing (01_economic_scripted_effects.txt:135769,
# yearly, market owners) is replaced: its +-2..10% of the money supply a year
# by the rate, gated by the private credit limits, is what filled rich pools
# for decades ahead. Kept from it: the interest the private banks earn on
# the foreign bonds they hold (ai_privat_bank_total_interest), while the pool
# is under the model's target.
#
# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of investement_pool_borrowing.
#############################################

REPLACE_OR_CREATE:investement_pool_borrowing = {
	if = {
		limit = {
			ai_privat_bank_total_interest > 0
			investment_pool < zz_ef_pool_target
		}
		add_investment_pool = ai_privat_bank_total_interest
	}
}

zz_ef_money_model_step = {
	# Gold parity re-anchor, once per change of what the money supply counts.
	# The money supply is the engine's money now (treasury + businesses' cash +
	# pool; since 2026-09-30 evening, version 2 -- E&F's goods counts are out),
	# a fraction of what E&F counted, so every currency is worth several times
	# its gold parity and the metal cover reads 200%+ (Britain 1848: 15.3 vs
	# 7.32). E&F sets the parity itself from the value when a standard is
	# adopted (money_value_target_valided); the same here: parity = today's
	# value, cover back to ~100%. Metal standards with a central bank only.
	if = {
		limit = { NOT = { has_variable = zz_ef_parity_version } }
		set_variable = { name = zz_ef_parity_version value = 0 }
	}
	if = {
		limit = { var:zz_ef_parity_version < 2 }
		set_variable = { name = zz_ef_parity_version value = 2 }
		if = {
			limit = {
				has_modifier = has_central_bank
				has_variable = money_value_target_1
				OR = {
					has_law = law_type:law_gold_standard
					has_law = law_type:law_silver_standard
					has_law = law_type:law_bimetallism_standard
				}
			}
			set_variable = { name = zz_ef_parity_before value = var:money_value_target_1 }
			set_variable = { name = money_value_target_1 value = money_value_0 }
		}
	}

	# Bank credit from the central bank: borrow when the pool is under its
	# target, repay when over (at most the debt); interest at the key rate goes
	# pool -> CB -> treasury (the CB's profit).
	if = {
		limit = { NOT = { has_variable = zz_ef_bank_cb_debt } }
		set_variable = { name = zz_ef_bank_cb_debt value = 0 }
	}
	set_variable = { name = zz_ef_f_contrib value = zz_ef_pool_contrib_month }
	set_variable = { name = zz_ef_f_cb_interest value = zz_ef_cb_interest }
	set_variable = { name = zz_ef_f_cb_borrow value = zz_ef_cb_borrow }
	set_variable = { name = zz_ef_f_cb_repay value = zz_ef_cb_repay }
	add_investment_pool = var:zz_ef_f_cb_borrow
	change_variable = { name = zz_ef_bank_cb_debt add = var:zz_ef_f_cb_borrow }
	add_investment_pool = {
		value = var:zz_ef_f_cb_repay
		multiply = -1
	}
	change_variable = { name = zz_ef_bank_cb_debt subtract = var:zz_ef_f_cb_repay }
	add_investment_pool = {
		value = var:zz_ef_f_cb_interest
		multiply = -1
	}
	add_treasury = var:zz_ef_f_cb_interest

	# Month-on-month change of each account (0 in the first month).
	zz_ef_money_ledger_delta = { ACC = pool VALUE = investment_pool }
	zz_ef_money_ledger_delta = { ACC = buildings VALUE = zz_ef_building_cash }
	zz_ef_money_ledger_delta = { ACC = treasury VALUE = zz_ef_treasury }
	zz_ef_money_ledger_delta = { ACC = cb VALUE = money_supply_state }
	zz_ef_money_ledger_delta = { ACC = m2 VALUE = money_supply }
	zz_ef_money_ledger_delta = { ACC = bonds VALUE = zz_ef_bank_bonds }

	# EF.31: the rate's share of private construction, refreshed monthly.
	remove_modifier = zz_ef_rate_private_construction
	add_modifier = {
		name = zz_ef_rate_private_construction
		multiplier = zz_ef_rate_construction_mult
	}
}

# var:zz_ef_d_<ACC> = VALUE now - VALUE a month ago; var:zz_ef_prev_<ACC> keeps
# this month's value.
zz_ef_money_ledger_delta = {
	if = {
		limit = { has_variable = zz_ef_prev_$ACC$ }
		set_variable = { name = zz_ef_d_$ACC$ value = $VALUE$ }
		change_variable = { name = zz_ef_d_$ACC$ subtract = var:zz_ef_prev_$ACC$ }
	}
	else = {
		set_variable = { name = zz_ef_d_$ACC$ value = 0 }
	}
	set_variable = { name = zz_ef_prev_$ACC$ value = $VALUE$ }
}
