# EF.39 / EF.48 (2026-09-30) -- the money model, in the engine's money, on
# the budget's own rhythm: a WEEKLY step (bank credit from the central bank
# at the key rate, its interest to the treasury, the ledger of each account's
# change, M0/M1/M2 history) and a monthly one (E&F's currency good, which
# E&F moves monthly; EF.31's modifier). Numbers and the why:
# script_values/zz_ef_money_model_values.txt. Started from
# on_actions/zz_ef_money_model_on_actions.txt.
#
# The engine has no weekly pulse for countries (script_docs on_actions.log:
# monthly and slower only), so the week is a chain: the weekly on_action
# re-triggers itself 7 days later (trigger_event = { on_action days = 7 },
# as LLWA and PSC do). The chain starts right after the engine's weekly
# budget tick: a daily probe (at most 8 days) waits until the treasury
# (gold_reserves - principal) changes. Then every 7-day window holds exactly
# one tick, and the budget read at the step is the tick in the window.
#
# E&F's investement_pool_borrowing (01_economic_scripted_effects.txt:135769,
# yearly, market owners) is replaced: its +-2..10% of the money supply a year
# by the rate, gated by the private credit limits, is what filled rich pools
# for decades ahead. Kept from it: the interest the private banks earn on
# the foreign bonds they hold (ai_privat_bank_total_interest), while the pool
# is under the model's target.
#
# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of investement_pool_borrowing.
#############################################

# Ф.2 (the user, 4.10): that interest is now paid -- by the issuer's treasury, every January
# (zz_ef_privbank_interest_pay, the monthly step) -- so the holder gets it always, not only while
# its pool is under the target (EF.34: "money from nothing" in the banks' card).
REPLACE_OR_CREATE:investement_pool_borrowing = {
	if = {
		limit = {
			ai_privat_bank_total_interest > 0
		}
		add_investment_pool = ai_privat_bank_total_interest
	}
}

# Ф.2 (4.10): the issuer pays the year's interest on its bonds held by foreign private banks (E&F's
# ai_entral_bank_debt_privat_bank_interest_refund_per_year_1..25: the holders' interest on bond N, summed
# over its buyer list) out of its treasury -- the engine's money is one unit for every country, so the
# sum is the holders'. E&F only counted it against the "gold loaned" and paid the holders from nothing.
# The treasury's change lands in the treasury's E&F moves (tr_other), a payment abroad.
zz_ef_privbank_interest_pay = {
	set_variable = { name = zz_ef_f_pbi_paid value = zz_ef_privbank_interest_due }
	if = {
		limit = { var:zz_ef_f_pbi_paid > 0 }
		add_treasury = {
			value = var:zz_ef_f_pbi_paid
			multiply = -1
		}
		if = {
			limit = {
				OR = {
					is_player = yes
					gdp > 20000000
				}
			}
			debug_log = "EFM|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|privbank_interest|paid [THIS.GetCountry.MakeScope.Var('zz_ef_f_pbi_paid').GetValue|0]"
		}
	}
}

# Е.5 (5.10, Д.2): the issuer paid the interest on its bonds held by other treasuries as an expense line -- replaced
# by the bond ledger (В5.5, 5.10: scripted_effects/zz_ef_bond_ledger.txt), which pays the holder out of the issuer's pool.

# ---- the weekly chain -------------------------------------------------------

# Monthly: (re)start the chain if it is not running (new game, new country,
# a save that lost its delayed on_actions). var:zz_ef_week_alive lives 10
# days and every step renews it.
zz_ef_money_week_start = {
	# Business cash at ~0.3 GDP: now in base_values (static_modifiers/
	# zz_ef_business_cash.txt); the monthly modifier of 330ba50 goes.
	if = {
		limit = { has_modifier = zz_ef_business_cash }
		remove_modifier = zz_ef_business_cash
	}
	if = {
		limit = {
			NOT = { has_variable = zz_ef_week_alive }
			NOT = { has_variable = zz_ef_week_probe }
		}
		set_variable = { name = zz_ef_week_probe value = 0 days = 10 }
		set_variable = { name = zz_ef_week_probe_value value = zz_ef_budget_position }
		trigger_event = { on_action = zz_ef_money_week_probe days = 1 }
	}
}

# Daily until the treasury changes (the budget tick), at most 8 days.
zz_ef_money_week_probe_step = {
	if = {
		limit = {
			OR = {
				NOT = { zz_ef_budget_position = var:zz_ef_week_probe_value }
				var:zz_ef_week_probe >= 7
			}
		}
		remove_variable = zz_ef_week_probe
		remove_variable = zz_ef_week_probe_value
		trigger_event = { on_action = zz_ef_money_model_weekly }
	}
	else = {
		change_variable = { name = zz_ef_week_probe add = 1 }
		set_variable = { name = zz_ef_week_probe_value value = zz_ef_budget_position }
		trigger_event = { on_action = zz_ef_money_week_probe days = 1 }
	}
}

# ---- the weekly step --------------------------------------------------------

zz_ef_money_model_step = {
	set_variable = { name = zz_ef_week_alive value = yes days = 10 }
	# Business cash: the sum over the buildings, once a week (run 15: the
	# estimate from the credit limit -- credit - base - 1.7 x GDP -- swung by
	# millions a week, M2 by +-30%).
	set_variable = { name = zz_ef_bcash value = zz_ef_building_cash_sum }
	# В1.3 probe (3.10): who gets the business cash jump of 1 January
	zz_ef_probe_newyear_cash = yes
	# UI.7 (3.10): the CB's metal rescaled this step, in money (zz_ef_metal_rescale_step)
	set_variable = { name = zz_ef_f_cb_rescale value = 0 }
	# foreign currency in the reserves (night 2, item 2), before the rescale
	zz_ef_fx_metal_update = yes
	# П.8 (4.10): each foreign currency's weekly change, for E&F's table (generated, zz_ef_clearing.txt)
	zz_ef_cbfx_week_step = yes
	# В2.5 (4.10): CB states taken from a country send their metal back to it
	zz_ef_cb_state_owner_step = yes
	# Е.5 (5.10, Д.2): the issuer pays the week's interest to the treasuries holding its bonds
	# В5.5 (5.10): bonds as parts of the engine's debt (generated, scripted_effects/zz_ef_bond_ledger.txt); replaces
	# Е.5's issuer-pays expense line (zz_ef_bond_interest_step)
	zz_ef_bond_ledger_step = yes
	if = {
		limit = { NOT = { has_variable = zz_ef_weeks_run } }
		set_variable = { name = zz_ef_weeks_run value = 0 }
	}
	if = {
		limit = { var:zz_ef_weeks_run < 100 }
		change_variable = { name = zz_ef_weeks_run add = 1 }
	}
	# M.8 (4.10): subjects' and CB-less countries' E&F metal to 40% of their money, once at week 12
	if = {
		limit = { var:zz_ef_weeks_run >= 12 }
		zz_ef_subject_metal_step = yes
	}

	# Gold parity re-anchor, once per change of what the money supply counts.
	# The money supply is the engine's money now (treasury + businesses' cash +
	# pool; since 2026-09-30 evening, version 2 -- E&F's goods counts are out),
	# Version 6 (1.10 evening): the national capacity x 2.5 -- parity = value
	# with the metal covering 40% of the money (zz_ef_capacity_per_metal).
	# Version 5 (1.10 day): the money in circulation now holds pops' savings
	# at hand too (the user's M2 - M0).
	# Version 4 (1.10, item 11): the treasury left the money supply.
	# Version 3 (1.10 night, EF.48 item 6): E&F's trade balance left the
	# national capacity (the CB's metal moves with the flows abroad instead).
	# a fraction of what E&F counted, so every currency is worth several times
	# its gold parity and the metal cover reads 200%+ (Britain 1848: 15.3 vs
	# 7.32). E&F sets the parity itself from the value when a standard is
	# adopted (money_value_target_valided); the same here: parity = today's
	# value, cover back to ~100%. Metal standards with a central bank only.
	# Version 7 (1.10 evening, the user): a NEW game keeps E&F's historical
	# parity (history/global/99_ef_history_global_variable.txt: the pound 7.32
	# -- grams of gold in a sovereign, the franc 0.75, ...); our re-anchors put
	# the pound at 60-80. E&F gives the CB metal = parity x its own money (a
	# counter ~10 times our money in circulation), i.e. ~1000% cover of ours.
	# So the CB's metal is rescaled once: 40% cover of the money in
	# circulation at the historical parity (zz_ef_cover_normal); with the
	# capacity x 2.5 the value starts at the parity. Older saves keep their
	# re-anchored parity (versions <= 6) -- test with a new game.
	# Twice: at the first step against the money expected (at least 0.45 GDP
	# -- run 15: at the first step the engine had not filled the buildings'
	# cash yet, Russia's M2 was 2M, France's 0, and the metal was cut to
	# nothing), and at week 12 against the money itself, if it is over 10% of
	# GDP by then (else a week later).
	if = {
		limit = { NOT = { has_variable = zz_ef_parity_version } }
		set_variable = { name = zz_ef_parity_version value = 7 }
		set_variable = { name = zz_ef_metal_rescale_due value = 11 }
		zz_ef_metal_rescale_step = yes
	}
	else_if = {
		limit = { has_variable = zz_ef_metal_rescale_due }
		change_variable = { name = zz_ef_metal_rescale_due subtract = 1 }
		if = {
			limit = {
				var:zz_ef_metal_rescale_due <= 0
				zz_ef_agg_m2_to_gdp > 0.1
			}
			remove_variable = zz_ef_metal_rescale_due
			zz_ef_metal_rescale_step = yes
		}
	}
	if = {
		limit = { var:zz_ef_parity_version < 6 }
		set_variable = { name = zz_ef_parity_version value = 6 }
		if = {
			limit = {
				has_modifier = has_central_bank
				has_variable = money_value_target_1
				OR = {
					has_law = law_type:law_gold_standard
					has_law = law_type:law_silver_standard
					has_law = law_type:law_bimetallism_standard
				}
			}
			set_variable = { name = zz_ef_parity_before value = var:money_value_target_1 }
			set_variable = { name = money_value_target_1 value = money_value_0 }
		}
	}

	# The budget's week, as the engine counted it at its last tick.
	set_variable = { name = zz_ef_f_budget value = zz_ef_budget_week }
	set_variable = { name = zz_ef_f_contrib value = investment_pool_gross_income }
	set_variable = { name = zz_ef_f_transfer value = zz_ef_pool_transfer_week }

	# Bank credit from the central bank: the debt goes to GDP x a share by the
	# rate (zz_ef_cb_credit_target, 1.10 evening), repaid from the pool when
	# over it (at most the debt); interest at the key rate goes
	# pool -> CB -> treasury (the CB's profit).
	if = {
		limit = { NOT = { has_variable = zz_ef_bank_cb_debt } }
		set_variable = { name = zz_ef_bank_cb_debt value = 0 }
	}
	set_variable = { name = zz_ef_f_cb_interest value = zz_ef_cb_interest }
	set_variable = { name = zz_ef_f_cb_borrow value = zz_ef_cb_borrow }
	set_variable = { name = zz_ef_f_cb_repay value = zz_ef_cb_repay }
	add_investment_pool = var:zz_ef_f_cb_borrow
	change_variable = { name = zz_ef_bank_cb_debt add = var:zz_ef_f_cb_borrow }
	add_investment_pool = {
		value = var:zz_ef_f_cb_repay
		multiply = -1
	}
	change_variable = { name = zz_ef_bank_cb_debt subtract = var:zz_ef_f_cb_repay }
	add_investment_pool = {
		value = var:zz_ef_f_cb_interest
		multiply = -1
	}
	add_treasury = var:zz_ef_f_cb_interest

	# Д4.5 / EF.38 (the user, 4.10): the treasury over the engine's limit (gold_reserves_limit; over it
	# the engine's returns fall off and the surplus burns, E&F GOLD_RESERVE_RETURNS_FACTOR) buys the
	# banks' bonds -- the excess goes into the pool (stays in circulation as bank money). Before the
	# ledger: a known flow treasury -> pool (zz_ef_v_f_tr_pool in both accounts' "other").
	set_variable = { name = zz_ef_f_tr_pool value = 0 }
	if = {
		limit = { gold_reserves > gold_reserves_limit }
		set_variable = { name = zz_ef_f_tr_pool value = gold_reserves }
		change_variable = { name = zz_ef_f_tr_pool subtract = gold_reserves_limit }
		add_treasury = {
			value = var:zz_ef_f_tr_pool
			multiply = -1
		}
		add_investment_pool = var:zz_ef_f_tr_pool
	}
	# EF.48 item 5 (business credit; repayment in money, 1.10 evening).
	zz_ef_business_credit_step = yes

	# The CB coins mined gold / silver (1.10 night).
	zz_ef_mint_step = yes

	# History rings (1.10, the user's M0..M3 -- var:zz_ef_agg0..3): a week ago
	# (w1..4), 13 blocks of 4 weeks (q1..13, a year), 20 blocks of 13 weeks
	# (r1..20, 5 years); plus the money in circulation and GDP for the inflation
	# (q only). Filled from last week's value; each macro skips a quantity not
	# yet recorded. The old m0/m1/m2 rings (definitions before 1.10) are left
	# to rot in old saves.
	if = {
		limit = { NOT = { has_variable = zz_ef_ring_q } }
		set_variable = { name = zz_ef_ring_q value = 0 }
		set_variable = { name = zz_ef_ring_r value = 0 }
	}
	# Once: the year ring of the money in circulation takes over the old M3
	# ring (until 1.10 day M3 was the same sum: businesses' cash + the pool +
	# savings at hand), so the inflation does not start from zero again.
	if = {
		limit = {
			has_variable = zz_ef_m3_q1
			NOT = { has_variable = zz_ef_circ_q1 }
		}
		if = { limit = { has_variable = zz_ef_m3_q1 } set_variable = { name = zz_ef_circ_q1 value = var:zz_ef_m3_q1 } }
		if = { limit = { has_variable = zz_ef_m3_q2 } set_variable = { name = zz_ef_circ_q2 value = var:zz_ef_m3_q2 } }
		if = { limit = { has_variable = zz_ef_m3_q3 } set_variable = { name = zz_ef_circ_q3 value = var:zz_ef_m3_q3 } }
		if = { limit = { has_variable = zz_ef_m3_q4 } set_variable = { name = zz_ef_circ_q4 value = var:zz_ef_m3_q4 } }
		if = { limit = { has_variable = zz_ef_m3_q5 } set_variable = { name = zz_ef_circ_q5 value = var:zz_ef_m3_q5 } }
		if = { limit = { has_variable = zz_ef_m3_q6 } set_variable = { name = zz_ef_circ_q6 value = var:zz_ef_m3_q6 } }
		if = { limit = { has_variable = zz_ef_m3_q7 } set_variable = { name = zz_ef_circ_q7 value = var:zz_ef_m3_q7 } }
		if = { limit = { has_variable = zz_ef_m3_q8 } set_variable = { name = zz_ef_circ_q8 value = var:zz_ef_m3_q8 } }
		if = { limit = { has_variable = zz_ef_m3_q9 } set_variable = { name = zz_ef_circ_q9 value = var:zz_ef_m3_q9 } }
		if = { limit = { has_variable = zz_ef_m3_q10 } set_variable = { name = zz_ef_circ_q10 value = var:zz_ef_m3_q10 } }
		if = { limit = { has_variable = zz_ef_m3_q11 } set_variable = { name = zz_ef_circ_q11 value = var:zz_ef_m3_q11 } }
		if = { limit = { has_variable = zz_ef_m3_q12 } set_variable = { name = zz_ef_circ_q12 value = var:zz_ef_m3_q12 } }
		if = { limit = { has_variable = zz_ef_m3_q13 } set_variable = { name = zz_ef_circ_q13 value = var:zz_ef_m3_q13 } }
	}
	zz_ef_ring_w_push = { M = agg0 }
	zz_ef_ring_w_push = { M = agg1 }
	zz_ef_ring_w_push = { M = agg2 }
	zz_ef_ring_w_push = { M = agg3 }
	change_variable = { name = zz_ef_ring_q add = 1 }
	if = {
		limit = { var:zz_ef_ring_q >= 4 }
		set_variable = { name = zz_ef_ring_q value = 0 }
		zz_ef_ring_q_push = { M = agg0 }
		zz_ef_ring_q_push = { M = agg1 }
		zz_ef_ring_q_push = { M = agg2 }
		zz_ef_ring_q_push = { M = agg3 }
		zz_ef_ring_q_push = { M = circ }
		zz_ef_ring_q_push = { M = gdp }
		# prices settle in the first two months of a game (run 17: the USA's index
		# 144 -> 99 by spring 1836, read as -37% a year): the year ring of prices
		# starts with week 9
		if = {
			limit = { var:zz_ef_weeks_run >= 9 }
			zz_ef_ring_q_push = { M = price }
		}
	}
	change_variable = { name = zz_ef_ring_r add = 1 }
	if = {
		limit = { var:zz_ef_ring_r >= 13 }
		set_variable = { name = zz_ef_ring_r value = 0 }
		zz_ef_ring_r_push = { M = agg0 }
		zz_ef_ring_r_push = { M = agg1 }
		zz_ef_ring_r_push = { M = agg2 }
		zz_ef_ring_r_push = { M = agg3 }
	}

	# UI.7 (3.10): what moves the CB's account (its metal at parity), for the
	# CB card -- each a week, in money; the rest is E&F's own metal operations.
	# Revaluation: the reserves of last step at the new parity minus at the old
	# one = cbm_prev x (parity_prev / parity_now - 1) -- devaluation laws and
	# re-anchors (run e0_3: the USA 2.06 -> 11.8 -> 29.9, cbm +99M / -86M).
	# Hume: the metal zz_ef_cb_hume_step added last week (after last week's
	# ledger, so it is in this week's change), at the parity.
	set_variable = { name = zz_ef_f_cb_reval value = 0 }
	set_variable = { name = zz_ef_f_cb_hume_m value = 0 }
	set_variable = { name = zz_ef_f_cb_stock value = 0 }
	if = {
		limit = {
			has_variable = money_value_target_1
			var:money_value_target_1 > 0
		}
		if = {
			limit = {
				has_variable = zz_ef_prev_cbm
				has_variable = zz_ef_prev_cbval
			}
			set_variable = { name = zz_ef_f_cb_reval value = var:zz_ef_prev_cbval }
			change_variable = { name = zz_ef_f_cb_reval divide = zz_ef_cb_valuation }
			change_variable = { name = zz_ef_f_cb_reval subtract = 1 }
			change_variable = { name = zz_ef_f_cb_reval multiply = var:zz_ef_prev_cbm }
		}
		if = {
			limit = {
				has_variable = zz_ef_f_hume
				zz_ef_cb_metal_money > 0
			}
			set_variable = { name = zz_ef_f_cb_hume_m value = var:zz_ef_f_hume }
			change_variable = { name = zz_ef_f_cb_hume_m divide = zz_ef_cb_valuation }
		}
		# UI.11 (5.10): E&F's CB building stored metal since the last step (its monthly stockpiling,
		# scripted_effects/zz_ef_cb_stockpile_track.txt), at the parity -- a line of the CB card
		if = {
			limit = { has_variable = zz_ef_cb_stock_acc }
			set_variable = { name = zz_ef_f_cb_stock value = var:zz_ef_cb_stock_acc }
			change_variable = { name = zz_ef_f_cb_stock divide = zz_ef_cb_valuation }
		}
		set_variable = { name = zz_ef_prev_cbval value = zz_ef_cb_valuation }
	}
	else_if = {
		limit = { has_variable = zz_ef_prev_cbval }
		remove_variable = zz_ef_prev_cbval
	}
	if = {
		limit = { has_variable = zz_ef_cb_stock_acc }
		remove_variable = zz_ef_cb_stock_acc
	}
	# Week-on-week change of each account (0 in the first week).
	zz_ef_money_ledger_delta = { ACC = pool VALUE = investment_pool }
	zz_ef_money_ledger_delta = { ACC = buildings VALUE = zz_ef_building_cash }
	zz_ef_money_ledger_delta = { ACC = tc VALUE = zz_ef_tc_cash }
	zz_ef_money_ledger_delta = { ACC = treasury VALUE = zz_ef_treasury }
	zz_ef_money_ledger_delta = { ACC = bonds VALUE = zz_ef_bank_bonds }
	zz_ef_money_ledger_delta = { ACC = tbonds VALUE = zz_ef_treasury_bonds }
	zz_ef_money_ledger_delta = { ACC = m2 VALUE = zz_ef_m2 }
	zz_ef_money_ledger_delta = { ACC = cbm VALUE = zz_ef_cb_money }
	# П.7 (4.10, В.4): abroad = the net foreign position (bonds + the CB's foreign currency - our currency abroad)
	zz_ef_money_ledger_delta = { ACC = abroad VALUE = zz_ef_abroad_net }
	zz_ef_money_ledger_delta = { ACC = agg0 VALUE = zz_ef_agg_m0 }
	zz_ef_money_ledger_delta = { ACC = agg1 VALUE = zz_ef_agg_m1 }
	zz_ef_money_ledger_delta = { ACC = agg2 VALUE = zz_ef_agg_m2 }
	zz_ef_money_ledger_delta = { ACC = agg3 VALUE = zz_ef_agg_m3 }
	zz_ef_money_ledger_delta = { ACC = circ VALUE = zz_ef_circulation }
	zz_ef_money_ledger_delta = { ACC = gdp VALUE = gdp }
	zz_ef_money_ledger_delta = { ACC = price VALUE = zz_ef_price_index }
	# night 2, item 6: building levels the government owns (privatization)
	zz_ef_money_ledger_delta = { ACC = govlv VALUE = zz_ef_gov_levels }
	# 4.10 (the money tooltip): pops' savings and deposits, for the cash-at-hand and deposits lines
	zz_ef_money_ledger_delta = { ACC = savings VALUE = zz_ef_pop_savings }
	zz_ef_money_ledger_delta = { ACC = deposits VALUE = zz_ef_pop_deposits }

	# Sandbox log (tools/run_vic3_sandbox.ps1 reads debug.log): one line a
	# week for the player and the big economies.
	if = {
		limit = {
			OR = {
				is_player = yes
				gdp > 20000000
			}
		}
		debug_log = "EFW|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|m0 [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_agg_m0')|0]|m1 # Pops' savings (decided 2026-09-30: savings = the money the engine lost,
# merged -- pops' income left over, which vanilla turns into wealth, and
# payments abroad inside the market). var:zz_ef_pop_savings (S) holds all,
# var:zz_ef_pop_deposits (D) the part in banks; cash at hand = S - D.
# Deposits are real engine money again: add_investment_pool. D moves 5% of
# the gap a week towards S x the deposit share (by the key rate); interest at
# the deposit rate is paid from the pool into D. The operations fall into the
# next week's window and are taken out of that week's leak.
zz_ef_pop_savings_step = {
	if = {
		limit = { NOT = { has_variable = zz_ef_pop_savings } }
		# pops' starting cash (1.10 evening): M0 is cash at hand, it cannot
		# start at zero; zz_ef_pop_start_savings
		set_variable = { name = zz_ef_pop_savings value = zz_ef_pop_start_savings }
		# Е.7 (5.10, Д.4): over the cash norm -- deposits from the start, the money in the banks' pool
		set_variable = { name = zz_ef_pop_deposits value = zz_ef_pop_start_deposits }
		add_investment_pool = var:zz_ef_pop_deposits
	}
	set_variable = { name = zz_ef_f_inflow value = zz_ef_v_f_leak }
	change_variable = { name = zz_ef_f_inflow multiply = -1 }
	# The first weeks of a game are the engine settling (run 14: Britain's
	# treasury 21 -> 39M in the second week, read as pops spending 20M of
	# savings they never had): no inflow while zz_ef_weeks_run < 3.
	if = {
		limit = {
			OR = {
				NOT = { has_variable = zz_ef_weeks_run }
				var:zz_ef_weeks_run < 3
			}
		}
		set_variable = { name = zz_ef_f_inflow value = 0 }
	}
	change_variable = { name = zz_ef_pop_savings add = var:zz_ef_f_inflow }
	# В1.2 (3.10): levels sold to companies -- the price to the sellers' savings
	# (zz_ef_v_f_buyout); the pool's loss stays out of the leak.
	set_variable = { name = zz_ef_f_buyout_got value = 0 }
	if = {
		limit = {
			has_variable = zz_ef_weeks_run
			var:zz_ef_weeks_run >= 3
		}
		set_variable = { name = zz_ef_f_buyout_got value = zz_ef_v_f_buyout }
		change_variable = { name = zz_ef_pop_savings add = var:zz_ef_f_buyout_got }
	}

	# В1.6 / ИВ.1 (the user 5.10 evening, (a)): the treasury's event spending and income (the budget's additional
	# expenses / income -- expeditions, "developing the can opener", DLC events; E&F's interest to its own CB not
	# counted) pair with pops' savings: what the treasury spent on an event is an order at home -- into the savings;
	# what it got from one -- out of them (at most what they hold). The engine itself only destroys / creates it.
	set_variable = { name = zz_ef_f_event_sav value = 0 }
	if = {
		limit = {
			has_variable = zz_ef_weeks_run
			var:zz_ef_weeks_run >= 3
		}
		set_variable = {
			name = zz_ef_f_event_sav
			value = {
				value = zz_ef_v_f_aint
				multiply = -1
				min = {
					value = var:zz_ef_pop_savings
					min = 0
					multiply = -1
				}
			}
		}
		change_variable = { name = zz_ef_pop_savings add = var:zz_ef_f_event_sav }
	}
	set_variable = { name = zz_ef_f_dep_in value = zz_ef_dep_in }
	set_variable = { name = zz_ef_f_dep_out value = zz_ef_dep_out }
	add_investment_pool = var:zz_ef_f_dep_in
	change_variable = { name = zz_ef_pop_deposits add = var:zz_ef_f_dep_in }
	add_investment_pool = {
		value = var:zz_ef_f_dep_out
		multiply = -1
	}
	change_variable = { name = zz_ef_pop_deposits subtract = var:zz_ef_f_dep_out }

	set_variable = { name = zz_ef_f_dep_int value = zz_ef_dep_interest }
	add_investment_pool = {
		value = var:zz_ef_f_dep_int
		multiply = -1
	}
	change_variable = { name = zz_ef_pop_deposits add = var:zz_ef_f_dep_int }
	change_variable = { name = zz_ef_pop_savings add = var:zz_ef_f_dep_int }
}

# EF.48 item 6 (the user, 1.10): the CB and abroad are one account -- the CB
# settles every payment abroad. Its reserves: metal (E&F's var:gold_state_1 /
# silver_state_1 on the CB's state, re-read by E&F's storing_gold_1 every
# month, so a change there stays) and E&F's currency-good stock. The week's
# net flow abroad (zz_ef_ext_net_week: budget lines with other countries,
# trade centres' cash, the pool's unexplained change, foreign bonds) moves
# the metal at the currency's value (Hume's price-specie flow): an outflow
# pays out metal, an inflow brings it in. Metal standards with a central
# bank only (fiat: nothing moves, the rate does). Metal and money fall by
# the same fraction, so an outflow alone does not move the currency's value.
zz_ef_cb_hume_step = {
	set_variable = { name = zz_ef_f_ext_net value = zz_ef_ext_net_week }
	set_variable = { name = zz_ef_f_hume value = 0 }
	set_variable = { name = zz_ef_f_hume_money value = 0 }
	# П.6 (the user, 4.10, В.2 / В.3): the world clearing -- the outflow pays the house in metal (by trust in
	# the currency) and in the own currency, the inflow takes a slice of what the house holds; every country
	# with a CB, fiat too (pays in currency). Was: the metal moved by the country's own flow with nobody on the
	# other side (generated, scripted_effects/zz_ef_clearing.txt; sets var:zz_ef_f_hume, the CB's metal, native).
	zz_ef_clr_step = yes
	if = {
		limit = {
			has_modifier = has_central_bank
			OR = {
				has_law = law_type:law_gold_standard
				has_law = law_type:law_silver_standard
				has_law = law_type:law_bimetallism_standard
			}
		}
		# M.3 (the user, 4.10): Hume at full strength -- the metal that came in is coined: the CB
		# issues as much money at the parity; the metal that went out takes as much money out of
		# circulation. The engine has no effect on a building's cash, so the money goes into / out of
		# the banks (the pool: the exporters' receipts deposited; the importers' payments drawn),
		# at most the pool on the way out. The pool operation falls into next week's window, as the
		# deposits do (var:zz_ef_w_hume_money in the pool's and the leak's accounts).
		# Check (night 4.10): the engine already pays exporters for the goods sold abroad and takes
		# the importers' money, so part of this may count twice -- switch: zz_ef_hume_money_on.
		if = {
			limit = {
				has_variable = money_value_target_1
				var:money_value_target_1 > 0
				NOT = { var:zz_ef_f_hume = 0 }

# EF.48 item 4: consumer credit, in the receiver after the savings (its pool
# operations fall into the next week's window, like the deposits). Values:
# script_values/zz_ef_money_model_values.txt.
zz_ef_consumer_credit_step = {
	if = {
		limit = {
			has_modifier = has_central_bank
			has_variable = zz_ef_dependents
			var:zz_ef_dependents > 0
		}
		if = {
			limit = { NOT = { has_variable = zz_ef_cc_debt } }
			set_variable = { name = zz_ef_cc_debt value = 0 }
		}
		set_variable = { name = zz_ef_f_cc_issue value = zz_ef_cc_issue }
		set_variable = { name = zz_ef_f_cc_repay value = zz_ef_cc_repay }
		set_variable = { name = zz_ef_f_cc_int value = zz_ef_cc_interest }
		set_variable = { name = zz_ef_f_cc_net value = var:zz_ef_f_cc_issue }
		change_variable = { name = zz_ef_f_cc_net subtract = var:zz_ef_f_cc_repay }
		change_variable = { name = zz_ef_f_cc_net subtract = var:zz_ef_f_cc_int }
		# pops cannot pay more than the floor (values file): cut the repayment,
		# then the interest, which is added to the debt
		set_variable = { name = zz_ef_cc_short value = zz_ef_cc_floor_week }
		change_variable = { name = zz_ef_cc_short subtract = var:zz_ef_f_cc_net }
		if = {
			limit = { var:zz_ef_cc_short > 0 }
			if = {
				limit = { var:zz_ef_cc_short > var:zz_ef_f_cc_repay }
				change_variable = { name = zz_ef_cc_short subtract = var:zz_ef_f_cc_repay }
				set_variable = { name = zz_ef_f_cc_repay value = 0 }
				change_variable = { name = zz_ef_f_cc_int subtract = var:zz_ef_cc_short }
				change_variable = { name = zz_ef_cc_debt add = var:zz_ef_cc_short }
			}
			else = {
				change_variable = { name = zz_ef_f_cc_repay subtract = var:zz_ef_cc_short }
			}
			set_variable = { name = zz_ef_f_cc_net value = zz_ef_cc_floor_week }
		}
		remove_variable = zz_ef_cc_short
		change_variable = { name = zz_ef_cc_debt add = var:zz_ef_f_cc_issue }
		change_variable = { name = zz_ef_cc_debt subtract = var:zz_ef_f_cc_repay }
		# pool side
		add_investment_pool = {
			value = var:zz_ef_f_cc_net
			multiply = -1
		}
	}
}

# EF.48 version 7 (1.10 evening): the CB's metal rescaled to cover
# zz_ef_cover_normal (40%) of the money in circulation at the historical
# parity. Metal standards with a CB only.
zz_ef_metal_rescale_step = {
	if = {
		limit = {
			has_modifier = has_central_bank
			has_variable = money_value_target_1
			var:money_value_target_1 > 0
			OR = {
				has_law = law_type:law_gold_standard
				has_law = law_type:law_silver_standard
				has_law = law_type:law_bimetallism_standard
			}
			zz_ef_cb_cover_ref > 0
		}
		set_variable = { name = zz_ef_metal_rescale value = zz_ef_cover_normal }
		change_variable = { name = zz_ef_metal_rescale divide = zz_ef_cb_cover_ref }
		# UI.7 (3.10): the CB card shows the rescale as a flow -- not at the
		# first step, where the account has no change yet (run e0_5: +1 bn "other")
		if = {
			limit = { has_variable = zz_ef_prev_cbm }
			set_variable = { name = zz_ef_f_cb_rescale value = var:zz_ef_metal_rescale }
			change_variable = { name = zz_ef_f_cb_rescale subtract = 1 }
			change_variable = { name = zz_ef_f_cb_rescale multiply = zz_ef_cb_money }
		}
		# every CB state, as E&F's gold_state_native_for_stockpile sums them
		every_scope_state = {
			limit = {
				has_modifier = central_bank_historic_place
			}
			if = {
				limit = { has_variable = gold_state_1 }
				change_variable = { name = gold_state_1 multiply = owner.var:zz_ef_metal_rescale }
			}
			if = {
				limit = { has_variable = silver_state_1 }
				change_variable = { name = silver_state_1 multiply = owner.var:zz_ef_metal_rescale }
			}
			# night 2, item 2 (2.10): E&F's currency stocks with the metal
			# (generated, scripted_effects/zz_ef_fx_stock_rescale.txt)
			zz_ef_fx_stock_rescale = yes
		}
		if = {
			limit = { has_variable = zz_ef_fx_metal }
			change_variable = { name = zz_ef_fx_metal multiply = var:zz_ef_metal_rescale }
		}
	}
}

# Night 2, item 3 (2.10): a seller redeems a currency in crisis for the
# issuer's metal (generated sell_<currency>_currency_crisis,
# scripted_effects/zz_ef_fx_crisis.txt). SELLER scope; var:zz_ef_crisis_due =
# the holdings at the issuer's parity, in the issuer's metal. The issuer pays
# from its CB capital's stock, at most what it holds; the seller gets it in
# its own standard's metal at the market rate (silver_to_gold_rate). Was:
# the issuer's metal set to 0, the sellers' stocks to 0, nobody paid.
zz_ef_crisis_redeem = {
	set_variable = { name = zz_ef_crisis_paid value = 0 }
	set_variable = { name = zz_ef_crisis_got value = 0 }
	if = {
		limit = {
			scope:overrated_currency_country_owner = { has_law = law_type:law_silver_standard }
			scope:central_bank_site_overrated_currency_owner = { has_variable = silver_state_1 }
		}
		set_variable = {
			name = zz_ef_crisis_paid
			value = {
				value = var:zz_ef_crisis_due
				max = scope:central_bank_site_overrated_currency_owner.var:silver_state_1
				min = 0
			}
		}
		scope:central_bank_site_overrated_currency_owner = {
			change_variable = { name = silver_state_1 subtract = scope:seller_root.var:zz_ef_crisis_paid }
		}
		set_variable = { name = zz_ef_crisis_got value = var:zz_ef_crisis_paid }
		if = {
			limit = { NOT = { has_law = law_type:law_silver_standard } }
			change_variable = { name = zz_ef_crisis_got multiply = silver_to_gold_rate }
		}
	}
	else_if = {
		limit = {
			scope:overrated_currency_country_owner = { NOT = { has_law = law_type:law_fiat_standard } }
			scope:central_bank_site_overrated_currency_owner = { has_variable = gold_state_1 }
		}
		set_variable = {
			name = zz_ef_crisis_paid
			value = {
				value = var:zz_ef_crisis_due
				max = scope:central_bank_site_overrated_currency_owner.var:gold_state_1
				min = 0
			}
		}
		scope:central_bank_site_overrated_currency_owner = {
			change_variable = { name = gold_state_1 subtract = scope:seller_root.var:zz_ef_crisis_paid }
		}
		set_variable = { name = zz_ef_crisis_got value = var:zz_ef_crisis_paid }
		if = {
			limit = { has_law = law_type:law_silver_standard }
			change_variable = { name = zz_ef_crisis_got multiply = gold_to_silver_rate }
		}
	}
	scope:central_bank_site_root = {
		if = {
			limit = { owner = { has_law = law_type:law_silver_standard } }
			if = {
				limit = { NOT = { has_variable = silver_state_1 } }
				set_variable = { name = silver_state_1 value = 0 }
			}
			change_variable = { name = silver_state_1 add = scope:seller_root.var:zz_ef_crisis_got }
		}
		else = {
			if = {
				limit = { NOT = { has_variable = gold_state_1 } }
				set_variable = { name = gold_state_1 value = 0 }
			}
			change_variable = { name = gold_state_1 add = scope:seller_root.var:zz_ef_crisis_got }
		}
	}
	if = {
		limit = { var:zz_ef_crisis_due > 0 }
		debug_log = "EFC|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|due [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_crisis_due')|0]|paid [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_v_crisis_paid')|0]"
		scope:overrated_currency_country_owner = {
			debug_log = "EFC|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|issuer"
		}
	}
}

# EF.48 item 5 -- business credit with repayment (the user, 1.10 evening:
# "repayment in any case, the point is a realistic money turnover"). Until
# now only values existed and the debt was never kept (always 0). Private
# construction paid with the banks' borrowed money (the CB's credit and pops'
# deposits, by their share of the pool) becomes the businesses' debt; due a
# week: 1/260 of it (5 years) + interest (key + 3 pp). It is paid from the
# buildings' contributions to the pool -- real money, business cash -> pool,
# out of M1. When the contributions fall short, zz_ef_debt_service raises the
# owners' contribution share (profits go to the bank instead of dividends),
# a step a week up to +0.3; when they cover the due by 20% it steps down.
# What is still unpaid is added to the debt.
zz_ef_business_credit_step = {
	if = {
		limit = { NOT = { has_variable = zz_ef_bc_debt } }
		set_variable = { name = zz_ef_bc_debt value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = zz_ef_bc_svc } }
		set_variable = { name = zz_ef_bc_svc value = 0 }
	}
	set_variable = { name = zz_ef_f_bc_new value = zz_ef_bc_new }
	set_variable = { name = zz_ef_f_bc_int value = zz_ef_bc_interest }
	set_variable = { name = zz_ef_f_bc_due value = zz_ef_bc_due }
	set_variable = { name = zz_ef_f_bc_paid value = zz_ef_bc_paid }
	change_variable = { name = zz_ef_bc_debt add = var:zz_ef_f_bc_new }
	change_variable = { name = zz_ef_bc_debt add = var:zz_ef_f_bc_int }
	change_variable = { name = zz_ef_bc_debt subtract = var:zz_ef_f_bc_paid }
	if = {
		limit = { var:zz_ef_bc_debt < 0 }
		set_variable = { name = zz_ef_bc_debt value = 0 }
	}
	# the controller of the owners' contribution share
	if = {
		limit = {
			var:zz_ef_f_bc_due > zz_ef_v_f_contrib
			var:zz_ef_bc_svc < zz_ef_bc_svc_max
		}
		change_variable = { name = zz_ef_bc_svc add = 1 }
	}
	else_if = {
		limit = {
			var:zz_ef_bc_svc > 0
			zz_ef_bc_contrib_slack > 0
		}
		change_variable = { name = zz_ef_bc_svc subtract = 1 }
	}
	remove_modifier = zz_ef_debt_service
	if = {
		limit = { var:zz_ef_bc_svc > 0 }
		add_modifier = {
			name = zz_ef_debt_service
			multiplier = var:zz_ef_bc_svc
		}
	}
}

# The CB coins half of the week's mined gold / silver (values file,
# zz_ef_mint_*): metal into every CB state's stock -- the first one, the
# capital's -- and the same money into pops' savings (the mines' owners).

	change_variable = { name = zz_ef_f_mint add = zz_ef_mint_silver_money }

		limit = { var:zz_ef_f_mint > 0 }
		set_variable = { name = zz_ef_f_mint_gold value = zz_ef_mint_gold_metal }
		set_variable = { name = zz_ef_f_mint_silver value = zz_ef_mint_silver_metal }

			limit = {
				has_modifier = central_bank_historic_place

				limit = { NOT = { has_variable = gold_state_1 } }

			}

				limit = { NOT = { has_variable = silver_state_1 } }
				set_variable = { name = silver_state_1 value = 0 }

			change_variable = { name = silver_state_1 add = owner.var:zz_ef_f_mint_silver }
		}

		# (zz_ef_mint_treasury_share: the treasury's brassage, the rest to the

			limit = { has_variable = zz_ef_pop_savings }

			change_variable = { name = zz_ef_pop_savings subtract = var:zz_ef_f_mint_tr }
		}
	}
	else = {
		set_variable = { name = zz_ef_f_mint_tr value = 0 }
	}
}
