# EF.39 / EF.48 (2026-09-30) -- the money model, in the engine's money, on
# the budget's own rhythm: a WEEKLY step (bank credit from the central bank
# at the key rate, its interest to the treasury, the ledger of each account's
# change, M0/M1/M2 history) and a monthly one (E&F's currency good, which
# E&F moves monthly; EF.31's modifier). Numbers and the why:
# script_values/zz_ef_money_model_values.txt. Started from
# on_actions/zz_ef_money_model_on_actions.txt.
#
# The engine has no weekly pulse for countries (script_docs on_actions.log:
# monthly and slower only), so the week is a chain: the weekly on_action
# re-triggers itself 7 days later (trigger_event = { on_action days = 7 },
# as LLWA and PSC do). The chain starts right after the engine's weekly
# budget tick: a daily probe (at most 8 days) waits until the treasury
# (gold_reserves - principal) changes. Then every 7-day window holds exactly
# one tick, and the budget read at the step is the tick in the window.
#
# E&F's investement_pool_borrowing (01_economic_scripted_effects.txt:135769,
# yearly, market owners) is replaced: its +-2..10% of the money supply a year
# by the rate, gated by the private credit limits, is what filled rich pools
# for decades ahead. Kept from it: the interest the private banks earn on
# the foreign bonds they hold (ai_privat_bank_total_interest), while the pool
# is under the model's target.
#
# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of investement_pool_borrowing.
#############################################

REPLACE_OR_CREATE:investement_pool_borrowing = {
	if = {
		limit = {
			ai_privat_bank_total_interest > 0
			investment_pool < zz_ef_pool_target
		}
		add_investment_pool = ai_privat_bank_total_interest
	}
}

# ---- the weekly chain -------------------------------------------------------

# Monthly: (re)start the chain if it is not running (new game, new country,
# a save that lost its delayed on_actions). var:zz_ef_week_alive lives 10
# days and every step renews it.
zz_ef_money_week_start = {
	if = {
		limit = {
			NOT = { has_variable = zz_ef_week_alive }
			NOT = { has_variable = zz_ef_week_probe }
		}
		set_variable = { name = zz_ef_week_probe value = 0 days = 10 }
		set_variable = { name = zz_ef_week_probe_value value = zz_ef_budget_position }
		trigger_event = { on_action = zz_ef_money_week_probe days = 1 }
	}
}

# Daily until the treasury changes (the budget tick), at most 8 days.
zz_ef_money_week_probe_step = {
	if = {
		limit = {
			OR = {
				NOT = { zz_ef_budget_position = var:zz_ef_week_probe_value }
				var:zz_ef_week_probe >= 7
			}
		}
		remove_variable = zz_ef_week_probe
		remove_variable = zz_ef_week_probe_value
		trigger_event = { on_action = zz_ef_money_model_weekly }
	}
	else = {
		change_variable = { name = zz_ef_week_probe add = 1 }
		set_variable = { name = zz_ef_week_probe_value value = zz_ef_budget_position }
		trigger_event = { on_action = zz_ef_money_week_probe days = 1 }
	}
}

	# EF.48 item 5: business credit, accounting only (values file).
	if = {

	change_variable = { name = zz_ef_bc_debt add = var:zz_ef_f_bc_int }
	change_variable = { name = zz_ef_bc_debt subtract = var:zz_ef_f_bc_paid }

		limit = { var:zz_ef_bc_debt < 0 }
		set_variable = { name = zz_ef_bc_debt value = 0 }
	}

	# Week-on-week change of each account (0 in the first week).
	zz_ef_money_ledger_delta = { ACC = pool VALUE = investment_pool }
	zz_ef_money_ledger_delta = { ACC = buildings VALUE = zz_ef_building_cash }
	zz_ef_money_ledger_delta = { ACC = tc VALUE = zz_ef_tc_cash }
	zz_ef_money_ledger_delta = { ACC = treasury VALUE = zz_ef_treasury }
	zz_ef_money_ledger_delta = { ACC = bonds VALUE = zz_ef_bank_bonds }
	zz_ef_money_ledger_delta = { ACC = m0 VALUE = zz_ef_m0 }

	# The GUI bridge (gui/zz_ef_money_hook.gui): the budget's lines outside
	# the accounts (minting, additional income and expenses, diplomacy,
	# treaties, power bloc, supply network fees, tolls, piracy) are GUI-only;
	# the hook reads them for every country in the list and hands them back
	# through scripted_guis/zz_ef_money_hook.txt, which computes the leak.
	# Last week's values are dropped, so a week without the GUI reads 0.
	# The deposit operations the receiver made after the previous step fell
	# into the window just closed: moved to var:zz_ef_w_dep_* for the leak.
	zz_ef_money_window_roll = { F = dep_in }
	zz_ef_money_window_roll = { F = dep_out }
	zz_ef_money_window_roll = { F = dep_int }
	zz_ef_money_window_roll = { F = cc_issue }
	zz_ef_money_window_roll = { F = cc_repay }
	zz_ef_money_window_roll = { F = cc_int }
	zz_ef_money_window_roll = { F = cc_net }
	if = { limit = { has_variable = zz_ef_f_ext } remove_variable = zz_ef_f_ext }
	if = { limit = { has_variable = zz_ef_f_leak } remove_variable = zz_ef_f_leak }
	set_variable = { name = zz_ef_hook_pending value = yes days = 6 }
	if = {
		limit = {
			NOT = { is_target_in_global_variable_list = { name = zz_ef_hook_countries target = this } }
		}
		add_to_global_variable_list = { name = zz_ef_hook_countries target = this days = 6 }
	}
}

# Called by the GUI bridge with scope:ext = the week's budget lines outside
# the accounts (settled, GetTrendValue). Once per week: the pending var is
# consumed (on_start may fire twice). The leak = what left the engine's
# money unexplained: the change of M2 minus the outside lines, the CB's
# credit and the private banks' foreign bonds. Negative = money went to pops
# (their income left over, wealth) or abroad inside the market.
zz_ef_money_hook_receive = {
	# Diagnostics: every call, also the ones that find nothing pending.

		remove_list_global_variable = { name = zz_ef_hook_countries target = this }

		set_variable = { name = zz_ef_f_pool_other value = zz_ef_pool_other_week }
		set_variable = { name = zz_ef_f_abr value = scope:abr }
		set_variable = { name = zz_ef_f_leak value = zz_ef_leak_week }

# var:zz_ef_w_<F> = var:zz_ef_f_<F> (0 if unset); var:zz_ef_f_<F> = 0.
zz_ef_money_window_roll = {
	set_variable = { name = zz_ef_w_$F$ value = 0 }
	if = {
		limit = { has_variable = zz_ef_f_$F$ }
		set_variable = { name = zz_ef_w_$F$ value = var:zz_ef_f_$F$ }
	}
	set_variable = { name = zz_ef_f_$F$ value = 0 }
}

# Pops' savings (decided 2026-09-30: savings = the money the engine lost,
# merged -- pops' income left over, which vanilla turns into wealth, and
# payments abroad inside the market). var:zz_ef_pop_savings (S) holds all,
# var:zz_ef_pop_deposits (D) the part in banks; cash at hand = S - D.
# Deposits are real engine money again: add_investment_pool. D moves 5% of
# the gap a week towards S x the deposit share (by the key rate); interest at
# the deposit rate is paid from the pool into D. The operations fall into the
# next week's window and are taken out of that week's leak.
zz_ef_pop_savings_step = {
	if = {
		limit = { NOT = { has_variable = zz_ef_pop_savings } }
		set_variable = { name = zz_ef_pop_savings value = 0 }
		set_variable = { name = zz_ef_pop_deposits value = 0 }
	}
	set_variable = { name = zz_ef_f_inflow value = zz_ef_v_f_leak }
	change_variable = { name = zz_ef_f_inflow multiply = -1 }
	change_variable = { name = zz_ef_pop_savings add = var:zz_ef_f_inflow }

	set_variable = { name = zz_ef_f_dep_in value = zz_ef_dep_in }
	set_variable = { name = zz_ef_f_dep_out value = zz_ef_dep_out }
	add_investment_pool = var:zz_ef_f_dep_in
	change_variable = { name = zz_ef_pop_deposits add = var:zz_ef_f_dep_in }
	add_investment_pool = {
		value = var:zz_ef_f_dep_out
		multiply = -1
	}
	change_variable = { name = zz_ef_pop_deposits subtract = var:zz_ef_f_dep_out }

	set_variable = { name = zz_ef_f_dep_int value = zz_ef_dep_interest }
	add_investment_pool = {
		value = var:zz_ef_f_dep_int
		multiply = -1
	}
	change_variable = { name = zz_ef_pop_deposits add = var:zz_ef_f_dep_int }
	change_variable = { name = zz_ef_pop_savings add = var:zz_ef_f_dep_int }
}

# EF.48 item 6 (the user, 1.10): the CB and abroad are one account -- the CB
# settles every payment abroad. Its reserves: metal (E&F's var:gold_state_1 /
# silver_state_1 on the CB's state, re-read by E&F's storing_gold_1 every
# month, so a change there stays) and E&F's currency-good stock. The week's
# net flow abroad (zz_ef_ext_net_week: budget lines with other countries,
# trade centres' cash, the pool's unexplained change, foreign bonds) moves
# the metal at the currency's value (Hume's price-specie flow): an outflow
# pays out metal, an inflow brings it in. Metal standards with a central
# bank only (fiat: nothing moves, the rate does). Metal and money fall by
# the same fraction, so an outflow alone does not move the currency's value.
zz_ef_cb_hume_step = {
	set_variable = { name = zz_ef_f_ext_net value = zz_ef_ext_net_week }
	set_variable = { name = zz_ef_f_hume value = 0 }
	if = {
		limit = {
			has_modifier = has_central_bank
			OR = {
				has_law = law_type:law_gold_standard
				has_law = law_type:law_silver_standard
				has_law = law_type:law_bimetallism_standard
			}
		}
		set_variable = { name = zz_ef_f_hume value = zz_ef_hume_metal_week }
		every_scope_state = {
			limit = {
				has_modifier = central_bank_historic_place
				is_capital = yes
			}
			if = {
				limit = { owner = { has_law = law_type:law_silver_standard } }
				if = {
					limit = { NOT = { has_variable = silver_state_1 } }
					set_variable = { name = silver_state_1 value = 0 }
				}
				change_variable = { name = silver_state_1 add = owner.var:zz_ef_f_hume }
			}
			else = {
				if = {
					limit = { NOT = { has_variable = gold_state_1 } }
					set_variable = { name = gold_state_1 value = 0 }
				}
				change_variable = { name = gold_state_1 add = owner.var:zz_ef_f_hume }
			}
		}
	}
}

# EF.48 item 4: consumer credit, in the receiver after the savings (its pool
# operations fall into the next week's window, like the deposits). Values:
# script_values/zz_ef_money_model_values.txt.
zz_ef_consumer_credit_step = {
	if = {
		limit = {
			has_modifier = has_central_bank
			has_variable = zz_ef_dependents
			var:zz_ef_dependents > 0
		}
		if = {
			limit = { NOT = { has_variable = zz_ef_cc_debt } }
			set_variable = { name = zz_ef_cc_debt value = 0 }
		}
		set_variable = { name = zz_ef_f_cc_issue value = zz_ef_cc_issue }
		set_variable = { name = zz_ef_f_cc_repay value = zz_ef_cc_repay }
		set_variable = { name = zz_ef_f_cc_int value = zz_ef_cc_interest }
		change_variable = { name = zz_ef_cc_debt add = var:zz_ef_f_cc_issue }
		change_variable = { name = zz_ef_cc_debt subtract = var:zz_ef_f_cc_repay }
		set_variable = { name = zz_ef_f_cc_net value = var:zz_ef_f_cc_issue }
		change_variable = { name = zz_ef_f_cc_net subtract = var:zz_ef_f_cc_repay }
		change_variable = { name = zz_ef_f_cc_net subtract = var:zz_ef_f_cc_int }
		# pool side
		add_investment_pool = {
			value = var:zz_ef_f_cc_net
			multiply = -1
		}
		# pops side: the dependents' surcharge for the coming week
		remove_modifier = zz_ef_consumer_credit
		add_modifier = {
			name = zz_ef_consumer_credit
			multiplier = zz_ef_cc_wage_add
		}
	}
}

# ---- the monthly step -------------------------------------------------------

zz_ef_money_model_monthly_step = {
	# EF.48 item 4: dependents for the consumer credit's surcharge.
	if = {
		limit = { has_modifier = has_central_bank }
		set_variable = { name = zz_ef_dependents value = zz_ef_dependents_count }
	}

	# E&F's currency good moves monthly (stockpile_currency_type_1).
	zz_ef_money_ledger_delta = { ACC = cb VALUE = money_supply_state }

	# EF.31: the rate's share of private construction.
	remove_modifier = zz_ef_rate_private_construction
	add_modifier = {
		name = zz_ef_rate_private_construction
		multiplier = zz_ef_rate_construction_mult
	}
}

# var:zz_ef_d_<ACC> = VALUE now - VALUE at the previous step; var:zz_ef_prev_<ACC>
# keeps this step's value.
zz_ef_money_ledger_delta = {
	if = {
		limit = { has_variable = zz_ef_prev_$ACC$ }
		set_variable = { name = zz_ef_d_$ACC$ value = $VALUE$ }
		change_variable = { name = zz_ef_d_$ACC$ subtract = var:zz_ef_prev_$ACC$ }
	}
	else = {
		set_variable = { name = zz_ef_d_$ACC$ value = 0 }
	}
	set_variable = { name = zz_ef_prev_$ACC$ value = $VALUE$ }
}
