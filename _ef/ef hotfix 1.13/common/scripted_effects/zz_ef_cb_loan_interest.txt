# В5.4 (5.10): the interest on the treasury's loan from its OWN central bank (E&F's "credit at the central bank":
# the player's button, zz_ef_cb_loan, and ai_credit_at_central_bank) is E&F's modifier interest_at_the_central_bank,
# country_expenses_add x the multiplier, a week -- the budget's "additional expenses". The money model took the
# additional income - expenses whole as primary income with abroad (zz_ef_f_aint, the bridge's g_aint): a payment to
# our own CB went into the balance of payments and the clearing. The multiplier is not readable from script, so it is
# kept in var:zz_ef_cbl_int when the modifier is set; the bridge takes it out of the abroad line and shows it as
# treasury -> CB (tools/regen_ef_money_supply_loc.py, zz_ef_v_cbl_int).
# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of E&F's two effects (01_financial_scripted_effects.txt): the bodies are
# E&F's as of 04.07.2026 plus the variable.
REPLACE_OR_CREATE:credit_at_central_bank_interest_per_week = {
	remove_modifier = interest_at_the_central_bank
	set_variable = { name = zz_ef_cbl_int value = credit_at_central_bank_interest_per_month_by_inflation }
	add_modifier = {
		name = interest_at_the_central_bank
		multiplier = var:zz_ef_cbl_int
	}
}
REPLACE_OR_CREATE:credit_at_central_bank_interest_per_week_off = {
	remove_modifier = interest_at_the_central_bank
	remove_variable = zz_ef_cbl_int
}
