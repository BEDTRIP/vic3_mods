# В3.6 (the user, 3.10; decided 4.10) -- the reference currency is chosen by geopolitics too: a
# great power with a credit rating BBB or better (E&F's country_credit_note_fixe >= 6; was >= 8 until 7.10) and,
# among those, the biggest reserves in gold (national_capacity_in_gold -- since M.1 the CB's
# reserves themselves). No candidate -- the reference stays where it is ("Britain is the best,
# but who cares if it is in default"). E&F took the biggest reserves of all.
# Also: both lists are cleared first (E&F only added to them, year after year).
# The candidate trigger: scripted_triggers/zz_ef_reference_currency_triggers.txt.
# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of E&F's national_capacity_variable_list
# (scripted_effects/08_list_effect.txt:202): E&F's body plus the three changes above.
#############################################

REPLACE_OR_CREATE:national_capacity_variable_list = {
	clear_variable_list = national_capacity_variable_list
	clear_variable_list = reference_currency_list_1
	# 6.10 day (В1 check): the yearly pick's great powers, for the log -- E&F's rating note (candidate: 6+), cover
	# (candidate: 0.25+), reserves in gold (the order), standard (1 gold, 2 silver, 3 bimetallism, 4 gold exchange),
	# the reference now. Line EFE.
	every_country = {
		limit = { country_rank >= rank_value:great_power }
		debug_log = "EFE|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|note [THIS.GetCountry.MakeScope.Var('country_credit_note_fixe').GetValue|0]|cover [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_cb_cover')|3]|cap_g [THIS.GetCountry.MakeScope.ScriptValue('national_capacity_in_gold')|0]|std [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_std_code')|0]|cb [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_has_cb_v')|0]|ref [THIS.GetCountry.MakeScope.ScriptValue('zz_ef_is_reference_v')|0]"
	}
    every_country = {
        limit = {
            has_modifier = has_central_bank
			not = {
				or = {
					market_owner_is_root_univ_with_buiding = yes
					is_subject_custom_trigger = yes
				}
			}
        }
        root = {
            add_to_variable_list = {
                name = national_capacity_variable_list
                target = prev
            }
        }
    }
    root  = {
        if = {
            clear_variable_list = list_before_order
            every_in_list = {
                variable = national_capacity_variable_list
                root = {
                    add_to_variable_list = {
                        name = list_before_order
                        target = prev
                    }
                }
            }
            clear_global_variable_list = national_capacity_variable_list_ordered_1
            ordered_in_list = {
                variable = list_before_order
                order_by = national_capacity_in_gold
                min = 0
                max = 10
                check_range_bounds = no
                root = {
                    add_to_global_variable_list = {
                        name = national_capacity_variable_list_ordered_1
                        target = prev
                    }
                    if = {
                        limit = { prev = { zz_ef_reference_candidate = yes } }
                        add_to_variable_list = {
                            name = reference_currency_list_1
                            target = prev
                        }
                    }
                }
            }
            clear_variable_list = reference_currency_country_1
            # 6.10 day (run r1006_172327): France (rating 7, cover 0.46, reserves 15.3M) and Russia (13.1M) were candidates
            # at 1837.12.29, Britain 10.1M, and the reference stayed with Britain -- the log of the candidates and the pick
            every_in_list = {
                variable = reference_currency_list_1
                debug_log = "EFE|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|cand|cap_g [THIS.GetCountry.MakeScope.ScriptValue('national_capacity_in_gold')|0]"
            }
            ordered_in_list = {
                variable = reference_currency_list_1
                order_by = national_capacity_in_gold
                # В1 (7.10): the FIRST -- positions count from 0 (vanilla: position = 0 for the top); E&F's 1 took the
                # second biggest reserves, and with one candidate nobody, so the reference never moved (France, AA,
                # cover 0.4-0.6, stayed behind Britain at 0.006 cover through 1842 in run r1006_141714)
                # 6.10 day: position alone (vanilla's form for the top); was with min = 0 / max = 1 as well -- a range
                position = 0
                check_range_bounds = no
                root = {
                    add_to_variable_list = {
                        name = reference_currency_country_1
                        target = prev
                    }
                }
            } 

			every_in_list = {
				variable = reference_currency_country_1
				debug_log = "EFE|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetCountry.GetNameNoFormatting]|pick"
			}
			if = {
				limit = { any_in_list = { variable = reference_currency_country_1 } }
			every_country = {
				limit = {
					has_modifier = global_monetary_reference
				}
				save_scope_as = target_country
			}
            every_in_list = {
                variable = national_capacity_variable_list 
				remove_modifier = global_monetary_reference
            } 

            every_in_list = {
                variable = reference_currency_country_1 
				add_modifier = {
					name = global_monetary_reference
				}

				if = {
					limit = { 
						not = { 
							scope:target_country = this 
						}
					} 
					#Event for player
					capital= {
						save_scope_as = capital_state
					}
					save_scope_as = target_country

					every_country = {
						limit = {
							is_player = yes 
						}
						trigger_event = 00_ef_economic_event.33
					}   
				} 
			}
            }
        }
    }
}
