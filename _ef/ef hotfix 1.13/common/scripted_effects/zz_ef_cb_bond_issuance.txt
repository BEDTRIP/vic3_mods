#############################################
# EF.15 (2026-09-24) -- central bank bond output follows national debt / GDP.
# The why and the formula: common/script_values/zz_ef_cb_bond_issuance_values.txt.
#
# Called once a month per country, from the central-bank branch of
# update_modifiers_bc_fc_ns (zz_ef_monthly_pulse_dedup.txt) -- that branch
# already runs only for countries that own a building_bank.
#
# COUNTRY scope. The ratio is stored on the country, the building reads it
# through owner.var:.
#
# var:credit_at_central_bank is part of the debt and is set by E&F's own
# credit effects. A country that never touched them has no variable; for it
# the shift is 0 (E&F's output unchanged) rather than a flood of
# "Failed to fetch variable" in error.log.
#############################################

zz_ef_cb_bond_issuance_update = {
	if = {
		limit = { has_variable = credit_at_central_bank }
		set_variable = { name = zz_ef_cb_bond_low value = zz_ef_cb_bond_output_low }
		set_variable = { name = zz_ef_cb_bond_high value = zz_ef_cb_bond_output_high }
	}
	else = {
		set_variable = { name = zz_ef_cb_bond_low value = 0 }
		set_variable = { name = zz_ef_cb_bond_high value = 0 }
	}
	every_scope_state = {
		limit = { has_building = building_bank }
		every_scope_building = {
			limit = { is_building_type = building_bank }
			remove_modifier = zz_ef_cb_bond_issuance_low
			remove_modifier = zz_ef_cb_bond_issuance_high
			if = {
				limit = { owner.var:zz_ef_cb_bond_low > 0 }
				add_modifier = {
					name = zz_ef_cb_bond_issuance_low
					multiplier = owner.var:zz_ef_cb_bond_low
				}
			}
			else_if = {
				limit = { owner.var:zz_ef_cb_bond_high > 0 }
				add_modifier = {
					name = zz_ef_cb_bond_issuance_high
					multiplier = owner.var:zz_ef_cb_bond_high
				}
			}
		}
	}
}
