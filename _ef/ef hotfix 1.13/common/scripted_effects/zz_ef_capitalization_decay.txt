# EF.23 (2026-09-24) -- market capitalization decays 2% a month
#
# WHAT E&F DOES
# Capitalization in the E&F menu (country_indice, 00_financial_scripted_value.txt
# :3119) = the five stockpiling_<type>_var_state_1 variables of the country's
# financial_center_historic_place state(s) x price. Each month every market
# owner runs stockpile_finacial_product (00_on_action_main.txt:509), which adds
# 4 x that market's buy orders of each financial product to those variables
# (01_financial_scripted_effects.txt:800-980). Nothing ever takes them down
# again except losing the capital to occupation (halved). So capitalization
# was the integral of demand since 1836: whoever grew a market first stays
# ahead for good -- "Russia never catches Germany".
#
# The author meant a ceiling: each _1 effect is gated on
# `stockpiling_<type>_var_1 < 1000000`. That line is dead twice over: written
# without `var:` it does not read the variable at all (error.log 24.09:
# "Variable 'stockpiling_manufacture_stock_var_1' is set but is never used"),
# and the country variable it names is only ever set to 0 at game start
# (00_ef_stockpile_global_variable.txt:407) -- the accumulator is the STATE
# variable _var_state_1. Adding `var:` would change nothing.
#
# WHAT THIS DOES
# Same wrapper, same five calls, then every counter in the historic place is
# multiplied by 0.98. With inflow I a month the counter settles at 50 x I:
# capitalization follows today's market (half-life ~34 months) instead of its
# history.
#
# Capitalization is read by the menu and by the half-yearly index history
# (country_indice_value_*, financial_center_ef_on_half_yearly_pulse_country);
# it does not feed production, prices or AI. The index can now fall.
#
# The state variables are seeded for every state by 7h/7i
# (zz_ef_stockpile_state_var_seed.txt), so no has_variable guard here.
#
# !! MAINTENANCE !! Key-level REPLACE_OR_CREATE of a 7-line E&F effect
# (01_financial_scripted_effects.txt:793, 4.1.7). Re-diff after an E&F update.
# !! TUNING !! the 0.98.
#############################################

REPLACE_OR_CREATE:stockpile_finacial_product = {
	stockpiling_bond_1 = yes
	stockpiling_manufacture_stock_1 = yes
	stockpiling_agricultural_stock_1 = yes
	stockpiling_mining_stock_1 = yes
	stockpiling_railroad_stock_1 = yes

	### EF.23
	every_scope_state = {
		limit = { has_modifier = financial_center_historic_place }
		change_variable = { name = stockpiling_bond_var_state_1 multiply = 0.98 }
		change_variable = { name = stockpiling_manufacture_stock_var_state_1 multiply = 0.98 }
		change_variable = { name = stockpiling_agricultural_stock_var_state_1 multiply = 0.98 }
		change_variable = { name = stockpiling_mining_stock_var_state_1 multiply = 0.98 }
		change_variable = { name = stockpiling_railroad_stock_var_state_1 multiply = 0.98 }
	}
}
