zz_pb_ef_css_private_ban_sgui = {
	scope = country
	is_shown = {
		has_variable = base_rate_percentage
		NOT = { has_variable = zz_pb_ef_css_private_ban }
	}
	is_valid = {
		custom_tooltip = {
			text = zz_pb_ef_css_private_ban_cooldown_tt
			NOT = { has_variable = zz_pb_ef_css_private_ban_cooldown }
		}
	}
	effect = {
		set_variable = zz_pb_ef_css_private_ban
		set_variable = {
			name = zz_pb_ef_css_private_ban_cooldown
			months = 6
		}
	}
}
