#############################################
# EF.30 -- the player's rate buttons, now with a price and a bias.
#
# E&F's base_rate_increase / base_rate_reduce (00_financial_scripted_guis.txt
# :3668) moved the rate +-0.5 pp for free, and the AI regulator then undid it.
# Now each press moves the rate at once, shifts the central bank's target by
# the same 0.5 pp (var:zz_ef_rate_bias, kept within +-2 pp, so the bank does not
# walk it back) and costs zz_ef_cb_rate_button_cost of treasury. The validity
# conditions are E&F's; the bias limit is added. Same names, so the E&F GUI
# needs no change.
#############################################

REPLACE_OR_CREATE:base_rate_increase = {
	is_shown = {
	}
	is_valid = {
		var:base_rate_percentage < 0.12
		not = { var:zz_ef_rate_bias >= 0.02 }
		not = {
			or = {
				has_modifier = rise_base_rate
				has_modifier = down_base_rate
				has_law = law_type:law_devaluation
				has_modifier = foreign_exchange_controls
			}
		}
	}
	effect = {
		add_treasury = {
			value = zz_ef_cb_rate_button_cost
			multiply = -1
		}
		change_variable = { name = base_rate_percentage add = 0.005 }
		change_variable = { name = zz_ef_rate_bias add = 0.005 }
		add_modifier = { name = rise_base_rate months = 6 }
	}
}

REPLACE_OR_CREATE:base_rate_reduce = {
	is_shown = {
	}
	is_valid = {
		var:base_rate_percentage > 0.02
		not = { var:zz_ef_rate_bias <= -0.02 }
		not = {
			or = {
				has_modifier = rise_base_rate
				has_modifier = down_base_rate
				has_law = law_type:law_revaluation
				has_modifier = foreign_exchange_controls
			}
		}
	}
	effect = {
		add_treasury = {
			value = zz_ef_cb_rate_button_cost
			multiply = -1
		}
		change_variable = { name = base_rate_percentage add = -0.005 }
		change_variable = { name = zz_ef_rate_bias add = -0.005 }
		add_modifier = { name = down_base_rate months = 6 }
	}
}
