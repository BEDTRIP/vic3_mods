#############################################
# pb+ef: switch off E&F's "Overbuilt Economy" bubble under PSC.
#
# WHY THE MECHANIC DOES NOT APPLY HERE
# E&F grows `speculative_share_2` every month while
#   building_ef_private_construction_lvl > building_urban_center_lvl_by_base_rate
# and turns it into `overbuilt_economy_modifier` on the sectors. In pb+ef that
# script_value is remapped (zz_pb_ef_remap_pcs_values.txt) onto
# `building_construction_sector`, i.e. onto the country's real construction
# capacity. PSC already charges for over-building the honest way: sectors no
# longer emit construction points, they emit wood/iron/steel/arc_welded
# construction goods, and set_construction_point_demand bills the treasury and
# the investment pool for them. Stacking E&F's debuff on top charges twice for
# the same thing.
#
# WHY WE DO NOT JUST REMAP E&F's DRAIN
# The counter's only sink in E&F is speculative_share_13_button:
#   random_scope_state = {
#       limit = { has_building = building_ef_private_construction }
#       remove_building = building_ef_private_construction }
# `remove_building` takes the WHOLE building, it has no level argument. In E&F
# that costs a side contraption; under PSC a state holds exactly one
# building_construction_sector and its level IS that state's entire
# construction capacity -- PSC's own script says so:
#   construction_regulator_level = {
#       if = { limit = { has_building = building_construction_sector }
#              b:building_construction_sector = { value = level } } }
# A 1:1 remap would have the AI bulldoze a full 20-50 level sector once a
# month, and PSC's regulator would keep converting goods it no longer has for
# up to four more weeks (set_construction_regulator_level only shrinks at
# `var:week_count >= 4`). So the drain stays disabled and we hold the counter
# at zero instead.
#
# SIDE EFFECTS, ON PURPOSE
# - speculative_share_9..12 stay usable: their `possible` requires
#   `var:speculative_share_2 = 0`.
# - speculative_share_13 never becomes visible: its `visible` requires
#   `var:speculative_share_2 > 0`.
# - the JE progress bar reads 0 instead of sitting parked at a clamped 100%.
#
# on_actions are additive, so this adds to E&F's hooks instead of replacing
# them -- no need to touch financial_center_je_2 at all.
#############################################

#############################################
# EF.18 (2026-09-24) -- a construction brake that actually brakes
#
# "PSC already charges for over-building" above turned out not to hold in
# play: 3473 construction sector levels against a ceiling of 3656 on a 1867
# save, no brake anywhere. And E&F's own brake was never one:
# overbuilt_economy_modifier has building_throughput_add commented out and
# only raises manufacture stock output; E&F also hangs it on
# building_ef_private_construction, which does not exist under PSC.
# Turning the counter back on would therefore brake nothing -- and bring
# back speculative_share_13 (the whole-building drain) and lock 9..12.
#
# So the counter stays at zero (above) and the brake is ours:
#   overbuild % = (construction sector levels - ceiling) / ceiling x 100,
#                 0 .. 50
#   every building_construction_sector of the country gets
#   zz_pb_ef_overbuilt_brake (building_throughput_add = -0.01) x that.
# The ceiling is E&F's (urban centre levels x base_rate_to_pcs_factor), so
# the central bank rate still steers it. No drain, no removed levels: an
# overbuilt country builds slower, it does not lose sectors.
# Proportional to the overbuild, not a counter that grows each month: it
# switches on and off with the cause.
#
# Only worth having with EF.17 in the E&F Hotfix: the ceiling counts urban
# centres, and exchanges used to add 5 urbanization per level.
#
# Cost: two per-state sums a month per country (script_values file); the
# per-building pass runs only for countries overbuilt this month or last.
#############################################

on_monthly_pulse_country = {
	on_actions = {
		zz_pb_ef_psc_overbuilt_off
	}
}

zz_pb_ef_psc_overbuilt_off = {
	effect = {
		if = {
			limit = {
				has_variable = speculative_share_2
				var:speculative_share_2 > 0
			}
			set_variable = { name = speculative_share_2 value = 0 }

			# Housekeeping for saves carried over from an earlier pb+ef build,
			# where the debuff was applied to the unified sector.
			every_scope_state = {
				every_scope_building = {
					limit = {
						is_building_type = building_construction_sector
					}
					remove_modifier = overbuilt_economy_modifier
				}
			}
		}

		### EF.18 -- our own brake, see the second header below.
		if = {
			limit = { has_variable = base_rate_percentage }
			set_variable = { name = zz_pb_ef_brake_pct value = zz_pb_ef_overbuilt_brake_pct }
		}
		else = {
			set_variable = { name = zz_pb_ef_brake_pct value = 0 }
		}
		if = {
			limit = {
				OR = {
					var:zz_pb_ef_brake_pct > 0
					has_variable = zz_pb_ef_brake_applied
				}
			}
			every_scope_state = {
				limit = { has_building = building_construction_sector }
				every_scope_building = {
					limit = { is_building_type = building_construction_sector }
					remove_modifier = zz_pb_ef_overbuilt_brake
					if = {
						limit = { owner.var:zz_pb_ef_brake_pct > 0 }
						add_modifier = {
							name = zz_pb_ef_overbuilt_brake
							multiplier = owner.var:zz_pb_ef_brake_pct
						}
					}
				}
			}
			if = {
				limit = { var:zz_pb_ef_brake_pct > 0 }
				set_variable = zz_pb_ef_brake_applied
			}
			else = {
				remove_variable = zz_pb_ef_brake_applied
			}
		}
	}
}
