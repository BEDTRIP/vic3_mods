# pb+ef: cosmetic only -- give the regulator's conversion PMs the construction
# icons instead of PSC's "construction bureaucrats" icon.
#
# Why the whole PM is restated: there is no way to patch a single field of an
# existing PM. INJECT: appends keys, and `texture` already exists, so it would
# leave a duplicate. So the four PMs are copied from PSC verbatim and only the
# texture line differs.
#
# Why we bother at all: the budget/overview panel reads AccessConstructionSpeeds,
# which is tied to NDefines.NCountry.CONSTRUCTION_CAMP_BUILDING. PSC repoints
# that define at building_construction_regulator, so the UI shows the regulator's
# PMs where the player expects the construction sector's.
#
# !! MAINTENANCE !!
# These four are byte-for-byte copies of PSC/common/production_methods/
# zz_PSC_construction.txt apart from `texture`. If PSC changes an input amount
# or the bureaucrat count, this file silently reverts it. RE-DIFF AGAINST PSC
# ON EVERY PSC UPDATE. Verified in sync with PSC 1.3.7 on 2026-08-19.
#
# REPLACE_OR_CREATE, not REPLACE: these keys do not exist in vanilla, PSC creates
# them. Plain REPLACE: would throw four errors into error.log if PSC is absent.
#############################################

REPLACE_OR_CREATE:pm_wood_point_conversion = {
	texture = "gfx/interface/icons/production_method_icons/wooden_buildings.dds"
	is_default = yes
	is_hidden_when_unavailable = yes

	country_modifiers = {
		workforce_scaled = {
			country_construction_add = 1
		}
	}

	building_modifiers = {
		workforce_scaled = {
			goods_input_wood_construction_add = 10
		}
		level_scaled = {
			building_employment_bureaucrats_add = 20
		}
	}
	unlocking_production_methods = {
		pm_wood_point_conversion
	}
}
