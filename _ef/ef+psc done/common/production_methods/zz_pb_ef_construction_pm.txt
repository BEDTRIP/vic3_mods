#############################################
# pb+ef: PSC's construction-goods output + E&F's stock issuance in one PM set.
#
# PSC replaces the four vanilla construction PMs so they emit
# goods_output_*_construction instead of country_construction_add. E&F's own
# private-sector PMs emitted manufacture_stock. Since the compatch collapses
# both mods onto one building, both outputs have to live in the same method.
#
# The stock numbers are E&F's, halved -- the commented values on the right are
# what E&F used on its own (cheaper, higher-yield) private sector. Its building
# gave double the construction for a quarter of the goods, all off-budget,
# which is the whole reason this compatch exists; keeping its stock yield
# unchanged on top of PSC's full-price sector would just move the free lunch.
#
# !! MAINTENANCE !!
# Everything except the goods_output_manufacture_stock_add lines is a verbatim
# copy of PSC/common/production_methods/zz_PSC_construction.txt. If PSC retunes
# an input or an employment number, this file silently reverts it.
# RE-DIFF AGAINST PSC ON EVERY PSC UPDATE.
# Verified in sync with PSC 1.3.7 on 2026-08-19 (only the four stock lines differ).
#############################################

	texture = "gfx/interface/icons/production_method_icons/wooden_buildings.dds"
	
	is_default = yes

	building_modifiers = {
		workforce_scaled = {
			# input goods
			goods_input_fabric_add = 25
			goods_input_wood_add = 75

			# output goods (PSC)
			goods_output_wood_construction_add = 20

			# output goods (E&F)
			goods_output_manufacture_stock_add = 20 #50
		}

		level_scaled = {
			building_employment_engineers_add = 40
			building_employment_clerks_add = 100
			building_employment_laborers_add = 860
		}
		unscaled = {
			building_laborers_mortality_mult = 0.1
		}
	}
}
