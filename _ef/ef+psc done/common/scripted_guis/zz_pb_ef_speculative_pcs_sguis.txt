#############################################
# pb+ef: player-side speculation buttons, remapped to the unified sector.
#
# E&F 04.07.2026 split these in two: common/scripted_buttons/00_ef_buttons.txt
# is the AI branch (visible = { ... is_ai = yes }, ai_chance = 1000, fired off
# the journal entry) and common/scripted_guis/00_financial_scripted_guis.txt is
# the player branch, clicked from gui/scripted_widgets/00_ef_custom_widgets.gui
# via GetScriptedGui('speculative_share_N_button').Execute.
#
# The earlier pb+ef build only patched the scripted_buttons copy, so the player
# branch still ran
#   start_privately_funded_building_construction = building_ef_private_construction
# against a building this compatch disables (potential = { always = no } in
# zz_pb_ef_disable_ef_private_construction.txt) -- the button lit up and built
# nothing. This file covers the other half.
#
# Two edits per button, nothing else:
#   1. build building_construction_sector instead of the disabled E&F type;
#   2. the cap check. E&F asks "is there room for one more sector", but the
#      effect builds 5/10/15/20 in one go, so we ask for that many free slots
#      via pcs_construction_sector_headroom_by_base_rate
#      (zz_pb_ef_remap_pcs_values.txt).
#
# Keep in sync with 00_financial_scripted_guis.txt on every E&F update.
#############################################

REPLACE_OR_CREATE:speculative_share_9_button = { #count = 5/12mois
	name = "speculative_share_9_button"
	desc = "speculative_share_9_button_desc"

	# is_shown = {
	# 	var:speculative_share_1 = 0
	# 	var:speculative_share_2 = 0
	# }
	is_valid = {
		NOT = {
			has_modifier = speculative_share_modifier_5
		}
		has_modifier = has_financial_center
		custom_tooltip = {
			text = speculative_share_5_button_tt_1
			var:speculative_share_1 < 20
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_1
			var:speculative_share_2 = 0
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_2
			pcs_construction_sector_headroom_by_base_rate >= 4
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_3
			base_rate_percentage_decimal < 12
			base_rate_percentage_decimal > 1.10
		}
		#is_player = yes
	}

	effect = {
		while = {
			count = 5
			random_scope_state = {
				start_privately_funded_building_construction = building_construction_sector
			}
		}
		add_modifier = {
			name = speculative_share_modifier_5
			months = 12
		}
		add_modifier = {
			name = speculative_share_modifier_6 #Monetary Expansion Lock
			months = 12
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_effect_1
		}
		# custom_tooltip = {
		# 	text = speculative_share_9_button_tt_effect_2
		# }
		if = { #protection de l'IA
			limit = {
				is_player = yes
			}
			change_variable = {
				name = base_rate_percentage
				subtract =  0.01
			}
			add_modifier = {
				name = down_base_rate
				months = 6
			}
		}
	}
}
