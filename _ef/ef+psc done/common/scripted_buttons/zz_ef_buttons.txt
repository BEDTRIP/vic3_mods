#############################################
# pb+ef: AI-side speculation buttons, remapped to the unified sector.
#
# Mirrors zz_pb_ef_speculative_pcs_sguis.txt for the other half of E&F's split
# (see that file's header). This is the branch the AI fires: E&F 04.07.2026
# gates these on `is_ai = yes` and gives them ai_chance = 1000, so they are
# pressed automatically off financial_center_je_2 -- the player never sees them.
#
# NOTE: an earlier pb+ef build rewrote `visible` back to
# `{ var:speculative_share_1 < 20 }`, which was E&F v4.1.1 behaviour and made
# the AI buttons show up for the player on top of the player's own widget
# copies. We now keep E&F's current gating verbatim and change only:
#   1. build building_construction_sector instead of the disabled E&F type;
#   2. ask pcs_construction_sector_headroom_by_base_rate for as many free slots
#      as the effect actually builds.
#
# speculative_share_13_button is deliberately NOT remapped -- see
# common/on_actions/zz_pb_ef_psc_overbuilt_off.txt for why.
#
# Keep in sync with 00_ef_buttons.txt on every E&F update.
#############################################

REPLACE_OR_CREATE:speculative_share_9_button = { #count = 5/12mois
	name = "speculative_share_9_button"
	desc = "speculative_share_9_button_desc"

	visible = {
		var:speculative_share_1 = 0
		var:speculative_share_2 = 0
		is_ai = yes
	}
	possible = {
		NOT = {
			has_modifier = speculative_share_modifier_5
		}
		has_modifier = has_financial_center
		custom_tooltip = {
			text = speculative_share_5_button_tt_1
			var:speculative_share_1 < 20
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_1
			var:speculative_share_2 = 0
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_2
			pcs_construction_sector_headroom_by_base_rate >= 4
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_3
			base_rate_percentage_decimal < 12
			base_rate_percentage_decimal > 1.10
		}
		#is_player = yes
	}

	effect = {
		while = {
			count = 5
			random_scope_state = {
				start_privately_funded_building_construction = building_construction_sector
			}
		}
		add_modifier = {
			name = speculative_share_modifier_5
			months = 12
		}
		add_modifier = {
			name = speculative_share_modifier_6 #Monetary Expansion Lock
			months = 12
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_effect_1
		}
		# custom_tooltip = {
		# 	text = speculative_share_9_button_tt_effect_2
		# }
		if = { #protection de l'IA
			limit = {
				is_player = yes
			}
			change_variable = {
				name = base_rate_percentage
				subtract =  0.01
			}
			add_modifier = {
				name = down_base_rate
				months = 6
			}
		}
	}

	ai_chance = {
		value = 1000
	}
}
