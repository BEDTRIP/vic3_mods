zz_ld_probe_rebuild = {
	if = {
		limit = {
			game_date >= 1836.3.1
			NOT = { has_global_variable = zz_p_phase }
		}
		set_global_variable = { name = zz_p_phase value = 0 }
		zz_ld_probe_rebuild_log_before = yes
		s:STATE_LANCASHIRE.region_state:GBR ?= {
			remove_building = building_textile_mill
			create_building = {
				building = building_textile_mill
				add_ownership = {
					country = {
						country = c:GBR
						levels = 13
					}
				}
				reserves = 1
				activate_production_methods = { pm_dye_workshops pm_no_luxury_clothes pm_traditional_looms }
			}
		}
		zz_ld_probe_rebuild_log_after = yes
	}
	else_if = {
		limit = {
			has_global_variable = zz_p_phase
			global_var:zz_p_phase < 4
		}
		change_global_variable = { name = zz_p_phase add = 1 }
		if = { limit = { global_var:zz_p_phase = 1 } zz_ld_probe_rebuild_log_m1 = yes }
		else_if = { limit = { global_var:zz_p_phase = 2 } zz_ld_probe_rebuild_log_m2 = yes }
		else_if = { limit = { global_var:zz_p_phase = 3 } zz_ld_probe_rebuild_log_m3 = yes }
		else = { zz_ld_probe_rebuild_log_m4 = yes }
	}
}
