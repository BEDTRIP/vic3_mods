### ComPatch HC+GoB+MoH x The Great Revision -- the Ottomans at 1836
###
### Same shape as the China file: TGR and Gates of the Bosphorus both ship
### common/history/countries/tur - ottoman empire.txt, GoB is later and takes the
### whole file, and TGR's addition -- the Imperial Arsenal company -- disappears
### with it.  Silent.
###
### Merged below: GoB's file with TGR's add_company block appended.
###
### VARIANTS: needs TGR for the company; without it, drop just that append.
### Victorian Century reworks this same file too -- if VC is part of the
### composition, load the standalone HC+GoB+MoH x TGR x VC compatch
### (_HC+GoB+MoH/hc+vc done) after this addon; it REPLACEs this same path,
### layering VC's law_madrasa activation and two amendments (verified-safe
### additions, not a competing rework -- see that compatch's README) on top of
### the exact body below.

COUNTRIES = {
	c:TUR ?= {
		effect_starting_technology_tier_4_tech = yes
		
		add_technology_researched = academia
		add_technology_researched = law_enforcement
		add_technology_researched = mandatory_service
		add_technology_researched = urban_planning
		add_technology_researched = line_infantry
		
		set_tax_level = medium
		set_military_wage_level = low
		set_government_wage_level = low
		
		add_taxed_goods = g:tobacco
		add_taxed_goods = g:wine
		add_taxed_goods = g:liquor

		activate_law = law_type:law_monarchy
		activate_law = law_type:law_imperial_divan
		activate_law = law_type:law_scribal_bureaucrats
		activate_law = law_type:law_millet_system
		activate_law = law_type:law_subjecthood
		activate_law = law_type:law_traditionalism
		activate_law = law_type:law_censorship
		activate_law = law_type:law_land_based_taxation
		activate_law = law_type:law_debt_slavery
		activate_law = law_type:law_migration_controls
		active_law:lawgroup_army_model ?= {
			add_amendment = {
				type = amendment_gbbf_cizye_tax
				sponsor = prev.ig:ig_landowners
			}
		}
		active_law:lawgroup_education_system ?= {
			add_amendment = {
				type = amendment_gbbf_elifba
				sponsor = prev.ig:ig_devout
			}
		}
		active_law:lawgroup_land_reform ?= {
			add_amendment = {
				type = amendment_gbbf_chiftlik
				sponsor = prev.ig:ig_landowners
			}
		}
		active_law:lawgroup_slavery ?= {
			add_amendment = {
				type = amendment_gbbf_esir_pazari
				sponsor = prev.ig:ig_landowners
			}
		}

		# The Sick Man of Europe
		add_journal_entry = {
			type = je_turkfm_sick_man_main
		}
		turkfm_initialize_sick_man = yes

		# The Reorganization of the State
		add_journal_entry = {
			type = je_turkfm_tanzimat_main
		}
		turkfm_initialize_tanzimat_effect = yes
		
		# The Tanzimat Reforms
		add_journal_entry = {
			type = je_turkfm_tanzimat_reforms
		}

		# The Lands of the Black Mountain
		add_journal_entry = {
			type = je_tur_raided
		}

		# Starting Conditions
		trigger_event = {
			id = turkfm_sick_man.001 # The Sick Man of Europe
		}
		add_modifier = {
			name = modifier_turkfm_imperial_aspiriations
			months = gbbf_normal_modifier_time
		}
		add_modifier = {
			name = modifier_turkfm_janissary_legacy
			months = gbbf_long_modifier_time
			is_decaying = yes
		}
		# this is to insulate the Ottomans from the first month SoL cliff... I don't like it, but its the best tool I have rn
		if = {
			limit = {
				has_game_rule = gbbf_six_month_grace
			}
			add_modifier = {
				name = modifier_turkfm_ottoman_legacy
				months = gbbf_even_shorter_modifier_time
			}
		}

		add_company = company_type:company_imperial_arsenal
		company:company_imperial_arsenal = {
			set_company_establishment_date = 1832.1.1
			set_company_state_region = s:STATE_EASTERN_THRACE
		}
	}
}
