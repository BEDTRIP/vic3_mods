########################################
# E&F + The Great Revision (TGR) - Technology compatibility
#
# Vic3 1.13 database prefixes patch PER SUB-BLOCK, not per entry:
# REPLACE:key = { modifier = { ... } } swaps ONLY the `modifier` block and leaves
# every field the mod did not list alone. Proof inside E&F itself: all 14 of its
# REPLACE:pm_company_headquarter_* entries carry nothing but `building_modifiers`,
# yet E&F standalone keeps vanilla textures and law gating on those PMs.
#
# So the only real conflict is where BOTH mods list the same sub-block.
# On these techs TGR adds its trade keys via `modifier`, and E&F (loaded later)
# lists `modifier` too - so TGR's keys are wiped. Everything else survives on
# its own and needs no patch here.
#
# We re-add the wiped keys with INJECT: instead of restating E&F's whole tech
# definition. Restating is what the previous version of this file did, and it
# drifted: it still carried E&F's pre-4.07.2026 `gdp_view >= 1` (missing `var:`,
# which logs an error) and re-enabled unlocking_technologies that the E&F author
# had deliberately commented out. INJECT touches only what is missing.
#
# INJECT: and not TRY_INJECT: on purpose - these are vanilla techs that cannot
# disappear, and if one ever does we want the log line rather than a silent no-op.
#
# Not patched, checked 21.08.2026:
#   currency_standards               - E&F has no `modifier` block, TGR's INJECT survives
#   international_exchange_standards - E&F's modifier is a superset of TGR's
#   modern_financial_instruments     - same
########################################

# TGR_TRADE_society.txt: INJECT:banking
INJECT:banking = {
	modifier = {
		state_export_advantage_mult = 0.05
	}
}

# TGR_LOANS_society.txt (minting) + TGR_TRADE_society.txt (export/capacity).
# E&F keeps only its own country_loan_interest_rate_add / central bank investment.
INJECT:central_banking = {
	modifier = {
		country_minting_mult = 0.1
		state_export_advantage_mult = 0.05
		state_max_trade_advantage_from_capacity_add = 0.05
	}
}

# state_building_trade_center_max_level_add is the backbone of TGR's trade rework:
# six techs give +10 each. E&F wipes it on four of them. Without this the state
# Trade Center ceiling ends the game ~40 levels short and the AI market starves.
INJECT:mutual_funds = {
	modifier = {
		state_building_trade_center_max_level_add = 10
	}
}

INJECT:corporate_charters = {
	modifier = {
		state_building_trade_center_max_level_add = 10
		state_export_advantage_mult = 0.05
	}
}

INJECT:joint_stock_companies = {
	modifier = {
		state_market_access_price_impact = 0.05
		state_building_trade_center_max_level_add = 10
	}
}

INJECT:investment_banks = {
	modifier = {
		state_max_trade_advantage_from_capacity_add = 0.05
		state_building_trade_center_max_level_add = 10
	}
}
