# ComPatch: The Great Revision + Kuromi AI + Tech & Res
# Load order (see README): CMF -> TGR -> KAI -> T&R -> this patch.
#
# ai_strategy_resource_expansion: TGR and KAI both REPLACE it and T&R does not touch
# it at all, so KAI's body is what the game actually ends up with -- and KAI's
# goods_stances list predates the T&R resource goods. We re-issue KAI's strategy
# verbatim and append the seven T&R resources, otherwise the AI never forms an
# export stance on them (no error in the log; the stance is simply absent).
#
# Removed 2026-08-23: ai_strategy_colonial_extraction. T&R 1.6 now ships the merged
# version itself (kai_has_high_supply triggers plus the same seven T&R goods) and
# loads after KAI, so the patch entry only re-stated what T&R already does.
#
# Deliberately NOT patched: ai_strategy_industrial_expansion (KAI vs T&R),
# ai_strategy_plantation_economy and ai_strategy_placate_population (TGR vs KAI).
# In all three both authors write a complete, deliberately different strategy --
# merging them would be a rebalance, not a compatibility fix. T&R's
# industrial_expansion does keep the industry_banned / extraction_economy gate that
# KAI had, so nothing structural is lost. See conflicts_tgr_vs_kai_report.md.

REPLACE:ai_strategy_resource_expansion = {
	icon = "gfx/interface/icons/ai_strategy_icons/resource_expansion.dds"

	type = administrative

	building_group_weights = {
		bg_rubber = 1.5
		bg_oil_extraction = 1.5
		bg_mining = 1.5
		bg_logging = 1.5
	}

	goods_stances = {
		# --- KAI, verbatim ---
		coal = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = coal } } }
		iron = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = iron } } }
		oil = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = oil } } }
		lead = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = lead } } }
		wood = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = wood } } }
		hardwood = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = hardwood } } }
		sulfur = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = sulfur } } }
		rubber = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = rubber } } }

		# --- T&R resource goods, same seven T&R itself uses in colonial_extraction ---
		good_uranium = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = good_uranium } } }
		bauxite = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = bauxite } } }
		copper = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = copper } } }
		commonores = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = commonores } } }
		advancedores = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = advancedores } } }
		rare_earths = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = rare_earths } } }
		gas = { stance = wants_export trigger = { kai_has_high_supply = { GOODS = gas } } }
	}

	possible = {
		always = yes
	}

	weight = {
		value = 20

		# Scaling with literacy 10-30%
		multiply = {
			value = literacy_rate
			add = -0.10
			divide = 0.20
			min = 0
			max = 1
		}
	}
}
