########################################
# E&F + The Great Revision (TGR) - Technology compatibility
#
# E&F restates ten vanilla techs with REPLACE: (ef_technology.txt: banking,
# currency_standards, central_banking, mutual_funds, corporate_charters,
# investment_banks, international_exchange_standards, joint_stock_companies,
# postal_savings, modern_financial_instruments). REPLACE: replaces the WHOLE
# entry (rules, settled 24.09.2026 -- the header of this file used to claim
# per-sub-block patching, that was wrong). E&F loads after TGR, so every TGR
# INJECT/REPLACE_OR_CREATE on those ten techs is wiped outright.
#
# We re-add TGR's trade keys with INJECT: instead of restating E&F's whole tech
# definition. Restating is what the first version of this file did, and it
# drifted: it still carried E&F's pre-4.07.2026 `gdp_view >= 1` (missing `var:`,
# which logs an error) and re-enabled unlocking_technologies that the E&F author
# had deliberately commented out. INJECT touches only what is missing.
#
# INJECT: and not TRY_INJECT: on purpose - these are vanilla techs that cannot
# disappear, and if one ever does we want the log line rather than a silent no-op.
#
# Only TGR's trade keys come back. TGR's ai_weight overrides (9999 on banking
# and currency_standards) and the urbanization/charter keys of its
# joint_stock_companies / investment_banks bodies stay E&F's.
#
# Not patched:
#   international_exchange_standards - E&F's modifier is a superset of TGR's
#   modern_financial_instruments     - same
#   (checked 21.08.2026)
#
# state_max_trade_advantage_from_capacity_add -- TGR's intent, restored
# 2026-09-25 (TGR.2 tail). Vanilla gives +0.05 on banking, central_banking,
# mutual_funds, international_exchange_standards. TGR 1.3.10 keeps those bodies
# (TGR_LOANS_society) and then puts -0.05 on banking and central_banking
# (TGR_TRADE_society) while adding +0.05 on investment_banks,
# corporate_management and macroeconomics: the bonus moves from the early
# banking techs to the late ones. E&F's whole-entry REPLACE of banking brings
# back vanilla's +0.05 and drops TGR's -0.05, and this file used to add +0.05
# to central_banking. Now: banking gets TGR's -0.05 back (net 0), and
# central_banking gets nothing here (E&F's body has no such line: net 0).
# investment_banks keeps +0.05 below; corporate_management and macroeconomics
# are not touched by E&F, TGR's own bodies stand.
########################################

# TGR_TRADE_society.txt: INJECT:banking
INJECT:banking = {
	modifier = {
		state_export_advantage_mult = 0.05
		state_max_trade_advantage_from_capacity_add = -0.05 ### TGR.2 tail, 2026-09-25
	}
}

# TGR.2 (24.09.2026): currency_standards was listed above as "E&F has no
# `modifier` block, TGR's INJECT survives". Under whole-entry REPLACE it does
# not: E&F's body wipes TGR's +10 Trade Center levels and export advantage.
# That was the one tech of TGR's ten +10s missing in game (90 instead of 100).
INJECT:currency_standards = {
	modifier = {
		state_building_trade_center_max_level_add = 10
		state_export_advantage_mult = 0.05
	}
}

# TGR_LOANS_society.txt (minting) + TGR_TRADE_society.txt (export/capacity).
# E&F keeps only its own country_loan_interest_rate_add / central bank investment.
INJECT:central_banking = {
	modifier = {
		country_minting_mult = 0.1
		state_export_advantage_mult = 0.05
		### TGR.2 tail (2026-09-25): no state_max_trade_advantage_from_capacity_add --
		### TGR's +0.05 (TGR_LOANS) and -0.05 (TGR_TRADE) cancel out.
	}
}

# state_building_trade_center_max_level_add is the backbone of TGR's trade rework:
# ten techs give +10 each. E&F wipes it on five of them (the four below and
# currency_standards above). Without this the state Trade Center ceiling ends the
# game 50 levels short and the AI market starves.
INJECT:mutual_funds = {
	modifier = {
		state_building_trade_center_max_level_add = 10
	}
}

INJECT:corporate_charters = {
	modifier = {
		state_building_trade_center_max_level_add = 10
		state_export_advantage_mult = 0.05
	}
}

INJECT:joint_stock_companies = {
	modifier = {
		state_market_access_price_impact = 0.05
		state_building_trade_center_max_level_add = 10
	}
}

INJECT:investment_banks = {
	modifier = {
		state_max_trade_advantage_from_capacity_add = 0.05
		state_building_trade_center_max_level_add = 10
	}
}
