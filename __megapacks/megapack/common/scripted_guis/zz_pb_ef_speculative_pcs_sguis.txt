# pb+ef: player-side stimulus buttons 9-12 of E&F's Financial Stability
# Office journal (financial_center_je_2), clicked from
# gui/scripted_widgets/zz_pb_ef_fso_widgets.gui.
#
# E&F 04.07.2026 split these in two: common/scripted_buttons/00_ef_buttons.txt
# is the AI branch (switched off here, scripted_buttons/zz_ef_buttons.txt) and
# common/scripted_guis/00_financial_scripted_guis.txt the player branch.
#
# REWORKED 2026-09-25 (after the 1850 run, decided by the user). E&F's button
# cut the base rate by 1-4% AND started 5/10/15/20 construction sectors; the
# sectors never got built, and the button needed zero overcapacity. Now each
# button is a pure rate cut with a price:
#   button   cut   penalty   allowed while the penalty is below
#     9      -1%     +10       90
#    10      -2%     +20       80
#    11      -3%     +30       70
#    12      -4%     +40       60
# so a press never takes the penalty past 100. Kept from E&F: the 12-month
# cooldown (speculative_share_modifier_5), the monetary expansion lock (_6),
# the bubble below 20, the rate window (above the cut + 0.1%, below 12%), the
# down_base_rate modifier and, for 10-12, the sum locked in the government
# fund (government_loan). Button 9 now cuts the rate for whoever presses it
# (E&F cut only for players there; this sgui is player-only anyway).
# Texts: localization/*/replace/zz_pb_ef_psc_l_*.yml.
#
# Keep in sync with 00_financial_scripted_guis.txt on every E&F update.
#############################################

REPLACE_OR_CREATE:speculative_share_9_button = { ### stimulus -1%: +10 overcapacity penalty
	name = "speculative_share_9_button"
	desc = "speculative_share_9_button_desc"

	is_valid = {
		NOT = {
			has_modifier = speculative_share_modifier_5
		}
		has_modifier = has_financial_center
		custom_tooltip = {
			text = speculative_share_5_button_tt_1
			var:speculative_share_1 < 20
		}
		### 2026-09-25: was "overcapacity = 0" plus room for 5 more sectors.
		### Now: the penalty after the press stays within 100.
		custom_tooltip = {
			text = speculative_share_9_button_tt_1
			var:speculative_share_2 < 90
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_3
			base_rate_percentage_decimal < 12
			base_rate_percentage_decimal > 1.10
		}
	}

	effect = {
		### 2026-09-25: no sectors started (the private queue never built
		### them) and no "expansion block" line.
		add_modifier = {
			name = speculative_share_modifier_5
			months = 12
		}
		add_modifier = {
			name = speculative_share_modifier_6 #Monetary Expansion Lock
			months = 12
		}
		custom_tooltip = {
			text = speculative_share_9_button_tt_effect_1
		}
		change_variable = {
			name = base_rate_percentage
			subtract = 0.01
		}
		add_modifier = {
			name = down_base_rate
			months = 6
		}
		### The price of cheap credit: +10 points of the overcapacity penalty
		### per 1% of the cut. It then goes back to its target by the usual
		### 2 / 1 points a month (on_actions/zz_pb_ef_overbuild_counter.txt).
		change_variable = {
			name = speculative_share_2
			add = 10
		}
		if = {
			limit = { var:speculative_share_2 > 100 }
			set_variable = {
				name = speculative_share_2
				value = 100
			}
		}
	}
}
