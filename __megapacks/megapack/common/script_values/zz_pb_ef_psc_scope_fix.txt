# MEGAPACK ONLY VERSION of the E&F + PSC patch's file of the same name.
# The patch's own copy has four tiers and no concrete; this one has five.
# Re-apply this merge whenever either patch is re-synced into the megapack.
#
# Two patches define calculate_state_construction_base_price:
#
#   z_PSC_TR_construction_values.txt   (PSC + T&R)  REPLACE:
#       five tiers -- wood / iron / steel / arc welded / concrete -- but reads
#       b:building_construction_regulator in every state, including states that
#       have no regulator. That is the startup spam
#       `has_active_production_method [ Wrong scope for trigger: none, expected building ]`
#
#   zz_pb_ef_psc_scope_fix.txt         (E&F + PSC)  REPLACE_OR_CREATE:
#       guards the lookup with has_building, but only knows PSC's four tiers.
#
# z_ sorts before zz_, so inside the megapack this file is the one the game
# keeps -- and on its own it would drop the concrete tier back to arc welded
# prices for the entire late game, silently. Hence the merge: T&R's ladder,
# E&F+PSC's guard.

REPLACE_OR_CREATE:calculate_state_construction_base_price = {
	# Default to the regulator's default PM (pm_wood_point_conversion) when the
	# regulator is missing or not initialized yet.
	value = wood_construction_good_base_price

	# Only inspect production methods if the regulator building exists here.
	if = {
		limit = { has_building = building_construction_regulator }
		b:building_construction_regulator = {
			if = {
				limit = { has_active_production_method = pm_wood_point_conversion }
				value = wood_construction_good_base_price
			}
			else_if = {
				limit = { has_active_production_method = pm_iron_point_conversion }
				value = iron_construction_good_base_price
			}
			else_if = {
				limit = { has_active_production_method = pm_steel_point_conversion }
				value = steel_construction_good_base_price
			}
			else_if = {
				limit = { has_active_production_method = pm_arc_welded_point_conversion }
				value = arc_welded_construction_good_base_price
			}
			else = {
				# pm_concrete_point_conversion -- PSC + T&R
				value = concrete_construction_good_base_price
			}
		}
	}
}
